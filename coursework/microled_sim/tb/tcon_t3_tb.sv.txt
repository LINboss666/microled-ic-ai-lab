`timescale 1ns / 1ps
// tcon_t3_tb — single-module verification of the Data Driver digital core (T3).
//
// WHAT IS BEING TESTED
//   rtl/data_driver.v: 384 channels, raw 384x8 + mapped 384x9 dual banks,
//   256x9 dual-bank gray LUT with a commit interlock, ONE shared PWM counter
//   with 384 comparators, and the BLANK/FAULT hardware-priority gate.
//
// WHAT IS NOT BEING TESTED
//   The analog current source (8 PAM tiers, I_PAM0..7, trim, 15 uA compliance),
//   the array RC, optical gray levels, SPI, interrupts, UVLO/OTP. A digital PASS
//   here says NOTHING about analog PVT/SOA/LED load.
//
// STIMULUS PROVENANCE
//   The pin timing is generated HERE from the frozen V1.3 numbers, not taken from
//   the TCON: slot 976/977, BLANK from cycle 768, LATCH 828..831 (40 ns),
//   PWM_START 840..843 (40 ns), PWM_TICK every 12 cycles (120 ns), DCLK 20 ns,
//   and one column per slot:
//       slot s   : window opens at the LATCH falling edge, win_nb bytes
//       slot s+1 : LATCH rising edge commits column s
//       slot s+2 : window displays column s
//   i.e. the accepted rx=s / disp=s-1 / lit=s-2 pipeline.
//
// INDEPENDENT ORACLE
//   The accepted behavioural model `data_model` (T2B-BR) is instantiated next to
//   the new RTL and fed the same D0/DCLK/LATCH pins. Its display bank must equal
//   the RTL's raw display bank, and its self-decoded (frame, column) tag must
//   equal the column the TB actually sent. It shares no internal signal with the
//   RTL, so this is a differential check, not a mirror.
module tcon_t3_tb;

    // ---------------- clocks ----------------------------------------------
    reg clk  = 1'b0;
    reg dclk = 1'b0;
    always #5 clk = ~clk;                       // 10 ns, 100 MHz
    always @(posedge clk) dclk <= ~dclk;        // 20 ns, 50 MHz

    // ---------------- DUT pins --------------------------------------------
    reg         reset_n   = 1'b0;
    reg         fault_n   = 1'b1;
    reg         blank, latch, pwm_start, pwm_tick;
    wire        din_valid;

    reg         cfg_hybrid     = 1'b1;
    reg         cfg_lut_we     = 1'b0;
    reg  [7:0]  cfg_lut_addr   = 8'd0;
    reg  [8:0]  cfg_lut_data   = 9'd0;
    reg         cfg_lut_commit = 1'b0;

    // ---------------- slot time base (frozen V1.3 numbers) ----------------
    integer cyc      = 0;
    integer slot_len = 976;
    integer frac     = 576;
    integer slot_idx = 0;
    reg     stim_en  = 1'b0;
    reg     slot_rst = 1'b0;

    always @(posedge clk) begin
        if (!reset_n || slot_rst) begin
            cyc <= 0; slot_len <= 976; frac <= 576; slot_idx <= 0;
        end else if (cyc == slot_len - 1) begin
            cyc      <= 0;
            slot_idx <= slot_idx + 1;
            if ((frac + 576) >= 1024) begin
                slot_len <= 977; frac <= frac + 576 - 1024;
            end else begin
                slot_len <= 976; frac <= frac + 576;
            end
        end else begin
            cyc <= cyc + 1;
        end
    end

    // the frozen slot waveform, REGISTERED at negedge clk (run03 precheck).
    // A combinational decode changes every pin at the same instant as a DCLK
    // rising edge (dclk is generated from posedge clk), which is a setup
    // violation by construction and makes the DUT (pre-edge sampling) and the
    // data_model oracle (post-edge +1 ns sampling) read DIFFERENT values.
    // Driving every DUT-facing pin from negedge clk gives 5 ns of setup and
    // 5 ns of hold around every DCLK edge, so both models see identical bytes.
    // stim_en=0 forces the safe idle state.
    always @(negedge clk) begin
        if (!reset_n || !stim_en) begin
            blank     <= 1'b1;
            latch     <= 1'b0;
            pwm_start <= 1'b0;
            pwm_tick  <= 1'b0;
        end else begin
            blank     <= (cyc >= 768);
            latch     <= (cyc >= 828) && (cyc < 832);          // 40 ns
            pwm_start <= (cyc >= 840) && (cyc < 844);          // 40 ns
            pwm_tick  <= (cyc < 768) && ((cyc % 12) == 0);     // 64 ticks / window
        end
    end

    // ---------------- per-slot payload plan -------------------------------
    // mode 0 = self-identifying tag payload (matches data_model's expectation)
    // mode 1 = ramp: channel i carries byte (i & 255), so ONE column exercises
    //          all 256 LUT entries at once
    localparam integer NSLOT = 16;
    integer pcol  [0:NSLOT-1];
    integer pmode [0:NSLOT-1];
    integer pnb   [0:NSLOT-1];
    integer pfrm  [0:NSLOT-1];

    integer cur_col = 0, cur_mode = 0, cur_nb = 384, cur_frm = 1;
    integer win_col = 0, win_mode = 0, win_nb = 384, win_frm = 1;
    integer sent    = 0;
    reg     win     = 1'b0;
    reg     latch_d = 1'b0;

    // at the last cycle of a slot, arm the plan for the NEXT slot
    always @(posedge clk) begin
        if (!reset_n || slot_rst) begin
            cur_col <= 0; cur_mode <= 0; cur_nb <= 384; cur_frm <= 1;
        end else if ((cyc == slot_len - 1) && ((slot_idx + 1) < NSLOT)) begin
            cur_col  <= pcol [slot_idx + 1];
            cur_mode <= pmode[slot_idx + 1];
            cur_nb   <= pnb  [slot_idx + 1];
            cur_frm  <= pfrm [slot_idx + 1];
        end
    end

    // window bookkeeping (run03 precheck + phase-drift fix).
    //
    // WHY THE PHASE MATTERS: slot_len alternates 976/977 (odd!), so the LATCH
    // fall - which is registered at negedge clk - drifts by 10 ns against the
    // negedge-dclk grid. In one phase the fall lands BEFORE the negedge dclk
    // that precedes the arming posedge, in the other phase AFTER it. If `sent`
    // simply incremented at every negedge while !latch, the second phase would
    // advance `sent` one step too early and the byte stream would be shifted by
    // one for exactly those columns - which is precisely why run03 showed
    // col0/col1 agreeing with the oracle and col2 disagreeing.
    //
    // FIX: `win` opens when the LATCH RISE is detected (long before the fall, so
    // it is always set in time), and `sent` only starts counting at the ARMING
    // POSEDGE (the first posedge where the DUT sees latch fall), flagged by
    // `started`. din_valid = win && !latch && sent<win_nb is therefore 1 at the
    // arming edge in BOTH phases, and the DUT (pre-edge sample) and the oracle
    // (post-edge +1 ns sample) read the same byte for the whole window.
    // ---------------------------------------------------------------------
    // SINGLE-WRITER RULE (reviewer RUN04 pre-check #3): every TB state
    // variable is assigned in exactly ONE always block.
    //   posedge-dclk block : latch_p, started, and the WHOLE window census
    //                        (win_edges + snapshot + opened/closed/dropped)
    //   negedge-dclk block : latch_d, win, sent, win_col/mode/nb/frm
    // `started` used to be set at posedge dclk and cleared at negedge dclk
    // (two writers). It is now a pure posedge-dclk state machine:
    //     LATCH rise  (latch && !latch_p) -> started <= 0   (new window prep)
    //     arming edge (!latch && latch_p) -> started <= 1   (byte 0 consumed)
    // so the byte stream can only start after the DUT has really taken byte 0,
    // in BOTH 976 and 977 slot phases.
    // ---------------------------------------------------------------------
    reg latch_p = 1'b0;
    reg started = 1'b0;
    integer win_edges = 0;
    integer win_edges_seen = 0;
    integer win_edges_min = 1000000;
    integer win_edges_max = 0;
    integer win_open_384   = 0;   // windows OPENED with a 384-byte plan
    integer win_closed_384 = 0;   // ... closed NORMALLY with exactly 384 edges
    integer win_dropped    = 0;   // closed by reset / superseded (excluded)
    reg     win_q    = 1'b0;
    integer win_nb_q = 0;
    reg     win_norm = 1'b0;

    always @(posedge dclk) begin
        latch_p <= latch;
        if (latch && !latch_p)      started <= 1'b0;   // LATCH rise: new window
        else if (!latch && latch_p) started <= 1'b1;   // arming edge: byte 0

        // window census, entirely in this domain: the counted edges and the
        // snapshot share the same clock, so a window's lifetime is counted 1:1
        // with the DCLK edges on which a byte was really presented.
        if (win) begin
            if (din_valid) win_edges = win_edges + 1;
        end else if (win_q && !win) begin
            if (win_norm) begin
                if (win_nb_q == 384) begin
                    if (win_edges < win_edges_min) win_edges_min = win_edges;
                    if (win_edges > win_edges_max) win_edges_max = win_edges;
                    win_edges_seen = win_edges_seen + 1;
                    win_closed_384 = win_closed_384 + 1;
                end
            end else begin
                win_dropped = win_dropped + 1;   // reset / superseded: excluded
            end
            win_edges = 0;
        end
        win_q    <= win;
        win_nb_q <= win_nb;   // RUN04 fix: this line was lost in the run03
                              // refactor, so win_nb_q stayed 0 and the
                              // win_nb_q==384 guard silently discarded every
                              // normally-closed window (closed384=0 despite
                              // the probes proving sent=384 closes)
    end

    always @(negedge dclk) begin
        if (!reset_n) begin
            latch_d <= 1'b0; win <= 1'b0; sent <= 0; win_norm <= 1'b0;
            win_col <= 0; win_mode <= 0; win_nb <= 384; win_frm <= 1;
        end else begin
            latch_d <= latch;
            if (latch && !latch_d) begin               // LATCH rise: prepare
                win      <= 1'b1;
                win_norm <= 1'b0;
                sent     <= 0;
                win_col  <= cur_col;
                win_mode <= cur_mode;
                win_nb   <= cur_nb;
                win_frm  <= cur_frm;
                if (cur_nb == 384) win_open_384 = win_open_384 + 1;
            end else if (win && started) begin          // byte stream running
                if (sent < win_nb) sent <= sent + 1;
                else begin win <= 1'b0; win_norm <= 1'b1; end
            end
        end
    end

    function automatic [7:0] exp_byte(input integer col, input integer frm,
                                      input integer idx, input integer md);
        integer r;
        begin
            if (md == 0) begin
                if (idx == 0)      exp_byte = col[7:0];
                else if (idx == 1) exp_byte = {col[9:8], frm[5:0]};
                else begin
                    r = 2 * idx;                       // chip 0 -> rows 0,2,4..
                    exp_byte = ((col & 255) + 13*(r & 255) + 7*(frm & 255)
                                + 8'h5A) & 255;
                end
            end else begin
                exp_byte = idx & 255;
            end
        end
    endfunction

    // din is combinational from `sent`, so it is always valid at the DCLK edge
    wire [7:0] din = (win && !latch && (sent < win_nb))
                   ? exp_byte(win_col, win_frm, sent, win_mode) : 8'h00;
    assign din_valid = (win && !latch && (sent < win_nb));

    // ---------------- DUT ------------------------------------------------
    wire [383:0]   ch_en;
    wire [1151:0]  ch_pam;
    wire [7:0]     tick_cnt;
    wire           col_valid;
    wire [9:0]     bytes_rcvd;
    wire           lut_valid;
    wire           lut_bank;
    wire           lut_cmt_bad;
    wire [5:0]     err_flags;

    data_driver #(.NCH(384), .PWM_MAX(63), .CNTW(8)) dut (
        .reset_n (reset_n), .dclk (dclk), .din (din), .din_valid (din_valid),
        .latch (latch), .blank (blank), .fault_n (fault_n),
        .pwm_start (pwm_start), .pwm_tick (pwm_tick),
        .cfg_hybrid (cfg_hybrid), .cfg_lut_we (cfg_lut_we),
        .cfg_lut_addr (cfg_lut_addr), .cfg_lut_data (cfg_lut_data),
        .cfg_lut_commit (cfg_lut_commit),
        .ch_en (ch_en), .ch_pam (ch_pam), .tick_cnt (tick_cnt),
        .col_valid (col_valid), .bytes_rcvd (bytes_rcvd),
        .lut_valid (lut_valid), .lut_bank (lut_bank),
        .lut_cmt_bad (lut_cmt_bad), .err_flags (err_flags)
    );

    // ---------------- DUT #2: the V1.3 self-delimiting receiver -------------
    // Same pins as `dut`, but USE_DIN_VALID=0: it must produce the same bytes
    // from LATCH + its own 384-edge counter, and must ignore din_valid.
    wire [383:0]   ch_en0;
    wire [1151:0]  ch_pam0;
    wire [7:0]     tick_cnt0;
    wire           col_valid0;
    wire [9:0]     bytes_rcvd0;
    wire           lut_valid0;
    wire           lut_bank0;
    wire           lut_cmt_bad0;
    wire [5:0]     err_flags0;

    data_driver #(.NCH(384), .PWM_MAX(63), .CNTW(8), .USE_DIN_VALID(0)) dut0 (
        .reset_n (reset_n), .dclk (dclk), .din (din),
        // V1.3 段0548: the chip has no external valid pin. Tying the legacy
        // port to a constant is exactly how the final wrapper uses it; if the
        // =0 mode secretly depended on it, this instance would receive nothing.
        .din_valid (1'b0),
        .latch (latch), .blank (blank), .fault_n (fault_n),
        .pwm_start (pwm_start), .pwm_tick (pwm_tick),
        .cfg_hybrid (cfg_hybrid), .cfg_lut_we (cfg_lut_we),
        .cfg_lut_addr (cfg_lut_addr), .cfg_lut_data (cfg_lut_data),
        .cfg_lut_commit (cfg_lut_commit),
        .ch_en (ch_en0), .ch_pam (ch_pam0), .tick_cnt (tick_cnt0),
        .col_valid (col_valid0), .bytes_rcvd (bytes_rcvd0),
        .lut_valid (lut_valid0), .lut_bank (lut_bank0),
        .lut_cmt_bad (lut_cmt_bad0), .err_flags (err_flags0)
    );

    // ---------------- independent oracle ----------------------------------
    wire        or_valid, or_corrupt;
    wire [9:0]  or_col;
    wire [5:0]  or_frame;
    wire [8:0]  or_capcnt;
    wire        or_capbad;
    wire [15:0] or_ncommit;

    data_model #(.CHIP(0)) u_oracle (
        .reset_n (reset_n), .dclk (dclk), .din (din), .latch (latch),
        .blank (blank),
        .disp_valid (or_valid), .disp_corrupt (or_corrupt),
        .disp_col (or_col), .disp_frame (or_frame),
        .cap_cnt (or_capcnt), .cap_bad (or_capbad), .n_commit (or_ncommit)
    );

    // ---------------- gray table + assertions -----------------------------
    reg [8:0] lut_tab [0:255];
    integer   assert_count = 0, pass_count = 0, fail_count = 0;

    task automatic chk(input string nm, input logic cond);
        begin
            assert_count = assert_count + 1;
            if (cond) begin
                pass_count = pass_count + 1;
                $display("PASS %s", nm);
            end else begin
                fail_count = fail_count + 1;
                $display("FAIL %s  (t=%0t cyc=%0d slot=%0d)", nm, $time, cyc, slot_idx);
            end
        end
    endtask

    // ---------------- safety monitors -------------------------------------
    integer n_blank_viol = 0, n_novalid_viol = 0, n_on_samples = 0;
    always @(negedge clk) begin
        if (reset_n && stim_en) begin
            if (blank && (|ch_en))      n_blank_viol   = n_blank_viol + 1;
            if (!col_valid && (|ch_en)) n_novalid_viol = n_novalid_viol + 1;
            if (|ch_en)                 n_on_samples   = n_on_samples + 1;
        end
    end

    // per-channel on-time counters, counted once per PWM_TICK
    integer on_cnt [0:383];
    integer i_on;
    integer count_slot = -1;
    reg     counting   = 1'b0;
    always @(posedge pwm_tick) begin
        #1;
        if (counting && reset_n && (slot_idx == count_slot))
            for (i_on = 0; i_on < 384; i_on = i_on + 1)
                if (ch_en[i_on]) on_cnt[i_on] = on_cnt[i_on] + 1;
    end

    // ---------------- T3-H directed monitors -------------------------------
    // B1 (reviewer P0): while lut_valid==0, a COMPLETE 384-byte non-zero column
    // is loaded and displayed. Every observable DCLK rising edge in that window
    // must show ch_en === 384'b0 in FOUR-STATE exact form: an X on ch_en counts
    // as a violation exactly like a 1 would ("do not let X pass as off",
    // task book §B1). ch_en is a top-level DUT pin here, not an internal net.
    reg     t3h_w1 = 1'b0;
    integer t3h_s1 = 0, t3h_v1 = 0;
    always @(posedge dclk) begin
        if (t3h_w1) begin
            t3h_s1 = t3h_s1 + 1;
            if (ch_en !== {384{1'b0}}) t3h_v1 = t3h_v1 + 1;
        end
    end

    // B3: the NATURAL rising edge of BLANK during an active, LUT-valid display
    // must kill the array without waiting for any clock edge (checked 1 ns
    // later - the reference clock period is 10 ns, so no clock can have helped).
    reg     t3h_w3 = 1'b0;
    integer t3h_blank_edges = 0, t3h_blank_viol = 0;
    always @(posedge blank) begin
        if (t3h_w3) begin
            #1;
            t3h_blank_edges = t3h_blank_edges + 1;
            if ((|ch_en) !== 1'b0) t3h_blank_viol = t3h_blank_viol + 1;
        end
    end

    always @(posedge clk) begin
        if (reset_n && stim_en && (slot_idx < 4) && (latch !== latch_d)) begin
            $display("DBG_LATCH t=%0t latch=%0d slot=%0d cyc=%0d cur_nb=%0d win=%0d sent=%0d dut_wcnt=%0d colv=%0d orc=%0d",
                     $time, latch, slot_idx, cyc, cur_nb, win, sent, dut.wcnt, col_valid, or_ncommit);
        end
    end

    // ---------------- helpers ---------------------------------------------
    task automatic goto_slot(input integer s);
        begin
            while (!((slot_idx == s) && (cyc == 0))) @(posedge clk);
        end
    endtask

    task automatic run_to_window_end;
        begin
            @(posedge clk);
            while (cyc != 768) @(posedge clk);
        end
    endtask

    task automatic lut_write(input integer a, input integer d);
        begin
            @(negedge dclk);
            cfg_lut_we = 1'b1; cfg_lut_addr = a[7:0]; cfg_lut_data = d[8:0];
            @(negedge dclk);
            cfg_lut_we = 1'b0;
            @(negedge dclk);
        end
    endtask

    task automatic lut_commit_pulse;
        begin
            @(negedge dclk);
            cfg_lut_commit = 1'b1;
            @(negedge dclk);
            cfg_lut_commit = 1'b0;
            @(negedge dclk);
        end
    endtask

    function automatic integer e_eff(input integer b);
        integer c;
        begin
            c = lut_tab[b][5:0];
            if (c == 0)       e_eff = 0;
            else if (c == 63) e_eff = 64;
            else              e_eff = c;
        end
    endfunction

    // ---------------- T3-H2: a SECOND, different LUT image ------------------
    // Table B halves every PWM code (black stays black, full stays full so the
    // black/full commit rule still holds). It exists only to prove that after a
    // bank switch the display bank really comes from the NEW image; it is not a
    // calibrated table and is never used for a functional pass criterion.
    reg [8:0] lut_b [0:255];

    function automatic integer e_eff_b(input integer b);
        integer c;
        begin
            c = lut_b[b][5:0];
            if (c == 0)       e_eff_b = 0;
            else if (c == 63) e_eff_b = 64;
            else              e_eff_b = c;
        end
    endfunction

    // ---- T3-H2 RUN02: ONE-EDGE PRE/POST commit observer --------------------
    // The old observer counted `lut_valid` 0->1, which is blind to a second
    // bank switch (1->1). This one attributes a LUT bank flip to the exact
    // DCLK edge it happens on: PRE values are read in the active region (before
    // the NBA update), POST values 1 ns later. No `#0`, no lut_valid counting.
    reg     pre_bank = 1'b0, pre_lt = 1'b0, pre_ltd = 1'b0, pre_armed = 1'b0;
    integer pre_wcnt = 0;
    reg     post_bank = 1'b0;
    reg [1:0] post_colv = 2'b00;
    integer commit_events = 0;
    reg     last_pre_ok = 1'b0;
    reg [1:0] last_post_colv = 2'b11;
    reg     last_both_bank_same = 1'b0;
    wire    latch_obs = latch;   // observation alias (keeps the TB-side pin
                                 // driver audit free of false positives)

    always @(posedge dclk) begin
        pre_bank  = lut_bank;
        pre_lt    = latch_obs;
        pre_ltd   = dut.latch_d;
        pre_armed = dut.armed;
        pre_wcnt  = dut.wcnt;
        #1;                                  // after the NBA update of this edge
        post_bank = lut_bank;
        if (reset_n && (post_bank !== pre_bank)) begin
            commit_events = commit_events + 1;
            // a legal LATCH commit is being made on this very edge when the
            // pre-edge state already satisfied armed && wcnt==NCH && rising
            last_pre_ok = (pre_lt === 1'b1) && (pre_ltd === 1'b0) &&
                          (pre_armed === 1'b1) && (pre_wcnt == 384);
            last_post_colv = {col_valid0, col_valid};
            last_both_bank_same = (lut_bank === lut_bank0);
        end
    end

    // ---- continuous "must stay dark" observer (COMMIT -> legal reload) ----
    // Armed right after a COMMIT flip, disarmed by the FIRST col_valid rise,
    // which is exactly the "new table, 384 bytes, LATCH commit" re-enable.
    reg     dark_watch = 1'b0, dark_done = 1'b0;
    integer dark_viol = 0, dark_samples = 0, dark_blank_low = 0;
    always @(negedge clk) begin
        #0.1;
        if (dark_watch && !dark_done && reset_n) begin
            if (col_valid === 1'b1) begin
                dark_done = 1'b1;            // legal reload ended the interval
            end else begin
                dark_samples = dark_samples + 1;
                if (blank === 1'b0) dark_blank_low = dark_blank_low + 1;
                if ((col_valid  !== 1'b0) || (col_valid0 !== 1'b0)) dark_viol = dark_viol + 1;
                if (((|ch_en)   !== 1'b0) || ((|ch_en0)  !== 1'b0)) dark_viol = dark_viol + 1;
            end
        end
    end

    // ---- bounded waits (never hang; the chk after them decides) ----------
    task automatic wait_col_valid1(input integer maxc);
        integer n; begin
            n = 0;
            while ((col_valid !== 1'b1) && (n < maxc)) begin @(posedge dclk); n = n + 1; end
        end
    endtask

    task automatic wait_col_valid0(input integer maxc);
        integer n; begin
            n = 0;
            while ((col_valid !== 1'b0) && (n < maxc)) begin @(posedge dclk); n = n + 1; end
        end
    endtask

    task automatic wait_dut0_valid(input integer maxc);
        integer n; begin
            n = 0;
            while ((col_valid0 !== 1'b1) && (n < maxc)) begin @(posedge dclk); n = n + 1; end
        end
    endtask

    task automatic wait_lut_valid1(input integer maxc);
        integer n; begin
            n = 0;
            while ((lut_valid !== 1'b1) && (n < maxc)) begin @(posedge dclk); n = n + 1; end
        end
    endtask

    task automatic wait_commit_events(input integer target, input integer maxc);
        integer n; begin
            n = 0;
            while ((commit_events < target) && (n < maxc)) begin @(posedge dclk); n = n + 1; end
        end
    endtask

    // enter a display window (BLANK low) and settle 30 clocks into it
    task automatic wait_display(input integer maxc);
        integer n; begin
            n = 0;
            while ((blank !== 1'b0) && (n < maxc)) begin @(posedge clk); n = n + 1; end
            repeat (30) @(posedge clk);
        end
    endtask

    task automatic wait_wcnt384(input integer maxc);
        integer n; begin
            n = 0;
            while ((dut0.wcnt !== 10'd384) && (n < maxc)) begin @(posedge dclk); n = n + 1; end
        end
    endtask

    // ---------------- scenario --------------------------------------------
    integer i, k, bad, col_expect, n_pam_bad, n_map_bad;
    integer k0;
    // T3-H2 RUN02 observation bookkeeping
    integer len_obs [0:1];
    reg     bank_at_fault = 1'b0;
    integer f_samples = 0, f_bank_flip = 0, f_on_viol = 0;
    reg     f_pend_seen = 1'b0;

    initial begin
        $display("COURSE_TCON_T3_SCENARIO=DATA_DRIVER_INTERNAL");
        $display("COURSE_TCON_T3_CONTRACT=V13_DATA_DRIVER_DIGITAL");

        // ---- the whole slot plan is fixed BEFORE any stimulus runs ---------
        for (i = 0; i < NSLOT; i = i + 1) begin
            pcol[i] = i; pmode[i] = 0; pnb[i] = 384; pfrm[i] = 1;
        end
        pmode[3] = 1;                     // slot 3 = ramp  -> displayed in slot 5
        pnb[5]   = 383;                   // short column  -> displayed in slot 7
        pnb[7]   = 385;                   // over-long     -> displayed in slot 9
        for (i = 0; i < 384; i = i + 1) on_cnt[i] = 0;

        // ---- the gray LUT image (V1.3 段0236 derivation, see GRAY_CODE_MAPPING)
        $readmemh("t3_lut.hex", lut_tab);
        // T3-H2 table B: same PAM tier, half the PWM width (black/full kept)
        for (i = 0; i < 256; i = i + 1) begin
            if (i == 0)        lut_b[i] = 9'h000;
            else if (i == 255) lut_b[i] = 9'h1FF;
            else               lut_b[i] = {lut_tab[i][8:6], (lut_tab[i][5:0] >> 1)};
        end

        // ================= phase 0: reset =================
        $display("--- P0 reset ---");
        reset_n = 1'b0;
        repeat (200) @(negedge clk);
        chk("D01_reset_no_channel_on", (|ch_en) === 1'b0);
        chk("D02_reset_col_invalid",   col_valid === 1'b0);
        chk("D03_reset_lut_invalid",   lut_valid === 1'b0);
        chk("D04_reset_tick_zero",     tick_cnt === 8'd0);
        reset_n = 1'b1;
        repeat (40) @(negedge clk);
        chk("D05_no_channel_on_after_reset", (|ch_en) === 1'b0);

        // ================= phase 1: LUT banks + commit interlock =============
        $display("--- P1 LUT dual bank / commit interlock ---");
        stim_en = 1'b0;                       // idle: BLANK high, no LATCH
        repeat (20) @(negedge clk);
        chk("D06_idle_blank_high", blank === 1'b1);

        // (a) an illegal image: entry 0 has PWM=5, so "black must be fully off"
        //     is violated -> the commit has to be refused
        for (i = 0; i < 256; i = i + 1) begin
            if (i == 0) lut_write(i, 9'h005);
            else        lut_write(i, lut_tab[i]);
        end
        chk("D07_bad_image_not_active", lut_valid === 1'b0 && lut_bank === 1'b0);
        lut_commit_pulse;
        repeat (10) @(negedge clk);
        chk("D08_commit_refused_black_rule", lut_cmt_bad === 1'b1 && lut_bank === 1'b0);

        // (b) fix entry 0; the deferred commit must now be accepted
        lut_write(0, lut_tab[0]);
        repeat (20) @(negedge clk);
        chk("D09_commit_accepted_after_fix", lut_bank === 1'b1 && lut_valid === 1'b1);
        chk("D10_black_and_full_codes_frozen",
            lut_tab[0] === 9'h000 && lut_tab[255] === 9'h1FF);

        // ================= align the slot counter to slot 0 =================
        slot_rst = 1'b1;
        repeat (2) @(posedge clk);
        slot_rst = 1'b0;
        stim_en  = 1'b1;
        // from here: slot s carries column s, is committed at slot s+1's LATCH
        // and is displayed in slot s+2's window.

        // ================= phase 2: normal columns (tag payload) ============
        $display("--- P2 columns 0..2 with the tag payload ---");
        n_pam_bad = 0; n_map_bad = 0;
        for (k = 0; k < 3; k = k + 1) begin
            col_expect = k;
            goto_slot(k + 2);
            bad = 0;
            for (i = 0; i < 384; i = i + 1)
                if (dut.raw_disp[i] !== exp_byte(col_expect, 1, i, 0)) bad = bad + 1;
            chk($sformatf("D11_col%0d_payload_384_bytes", col_expect), bad == 0);
            chk($sformatf("D12_col%0d_valid", col_expect), col_valid === 1'b1);
            for (i = 0; i < 384; i = i + 1) begin
                if (dut.map_disp[i] !== lut_tab[dut.raw_disp[i]]) n_map_bad = n_map_bad + 1;
                if (ch_pam[i*3 +: 3] !== lut_tab[dut.raw_disp[i]][8:6]) n_pam_bad = n_pam_bad + 1;
            end
            bad = 0;
            for (i = 0; i < 384; i = i + 1)
                if (dut.raw_disp[i] !== u_oracle.dbank[i]) bad = bad + 1;
            chk($sformatf("D13_col%0d_oracle_bank_match", col_expect), bad == 0);
            chk($sformatf("D14_col%0d_oracle_tag_col", col_expect), or_col === col_expect[9:0]);
            chk($sformatf("D15_col%0d_oracle_not_corrupt", col_expect), or_corrupt === 1'b0);
        end
        chk("D16_mapped_bank_follows_lut", n_map_bad == 0);
        chk("D17_pam_code_follows_lut",    n_pam_bad == 0);

        // ================= phase 3: gray engine on the ramp column ===========
        $display("--- P3 gray engine: ramp column exercises all 256 codes ---");
        goto_slot(4);
        for (i = 0; i < 384; i = i + 1) on_cnt[i] = 0;
        count_slot = 5; counting = 1'b1;         // count slot 5's window = column 3
        goto_slot(5);
        run_to_window_end;
        counting = 1'b0; count_slot = -1;
        chk("D18_ramp_column_valid", col_valid === 1'b1);
        bad = 0;
        for (i = 0; i < 384; i = i + 1)
            if (on_cnt[i] != e_eff(i & 255)) bad = bad + 1;
        chk("D19_all_384_on_time_matches_lut", bad == 0);
        chk("D20_black_code_never_on",  on_cnt[0] == 0);
        chk("D21_full_code_64_ticks",   on_cnt[255] == 64);
        chk("D22_tick_counter_saturates", tick_cnt <= 8'd63);
        chk("D23_gray_engine_active",   n_on_samples > 0);

        // ================= phase 4: short / over-long columns ================
        $display("--- P4 illegal column length ---");
        goto_slot(7);                            // display of column 5 (short)
        chk("D24_short_column_not_displayed", col_valid === 1'b0);
        chk("D25_short_column_no_channel",    (|ch_en) === 1'b0);
        chk("D29_short_sets_underrun_flag",   err_flags[0] === 1'b1);
        goto_slot(9);                            // display of column 7 (over-long)
        chk("D26_overlong_column_not_displayed", col_valid === 1'b0);
        chk("D27_overlong_sets_overflow_flag",   err_flags[1] === 1'b1);
        chk("D28_overlong_no_channel",           (|ch_en) === 1'b0);
        goto_slot(10);                           // display of column 8 (recovers)
        chk("D30_valid_column_after_illegal",    col_valid === 1'b1);

        // ================= phase 5: fault hardware priority =================
        $display("--- P5 fault hardware priority ---");
        goto_slot(11);
        chk("D31_display_valid_in_window", col_valid === 1'b1);
        // stay inside the first 12-cycle tick interval: tick_cnt is still 0, so
        // every channel whose code is non-zero must be conducting
        repeat (8) @(negedge clk);
        chk("D32_channel_on_before_fault", (|ch_en) === 1'b1);
        fault_n = 1'b0;
        #1;
        chk("D33_fault_kills_output_same_cycle", (|ch_en) === 1'b0);
        repeat (5) @(negedge clk);
        chk("D34_fault_holds_off", (|ch_en) === 1'b0);
        fault_n = 1'b1;
        repeat (5) @(negedge clk);
        chk("D35_recover_after_fault", (|ch_en) === 1'b1);

        // the oracle's own commit counter is cleared by the reset below, so it
        // must be sampled here and not after P6
        bad = or_ncommit;
        chk("D40_oracle_committed_columns", bad >= 8);

        // ================= phase 6: reset ===================================
        $display("--- P6 reset mid-display ---");
        reset_n = 1'b0;
        #1;
        chk("D36_reset_kills_output", (|ch_en) === 1'b0);
        repeat (20) @(negedge clk);
        chk("D37_reset_clears_valid",  col_valid === 1'b0);
        reset_n = 1'b1;
        stim_en = 1'b0;
        repeat (40) @(negedge clk);

        // ================= phase 7 (T3-H): LUT_VALID output gate ============
        // Reviewer P0 (T3-H task book §B). Runs AFTER P6, so the reset has
        // cleared ok_a/ok_b and lut_valid is 0, and BEFORE the D42 census so
        // every window opened here is included in the bookkeeping.
        //   B1: a COMPLETE 384-byte non-zero column with an UNCOMMITTED LUT
        //       must hold ch_en at exactly 0 on every sampled edge.
        //   B2: after a real 256-entry COMMIT and a RELOAD, the same pins must
        //       conduct again with the LUT's on-times / PAM tiers.
        //   B3: BLANK / FAULT / RESET keep hardware priority in this mode.
        $display("--- P7 T3-H: uncommitted LUT must force ch_en == 0 ---");
        chk("D43_t3h_lut_invalid_after_reset", lut_valid === 1'b0);

        // ---- B1 ------------------------------------------------------------
        for (i = 0; i < 8; i = i + 1) begin
            pcol[i] = 8 + i; pmode[i] = 1; pnb[i] = 384; pfrm[i] = 1;
        end
        slot_rst = 1'b1;
        repeat (2) @(posedge clk);
        slot_rst = 1'b0;
        stim_en  = 1'b1;
        goto_slot(2);                        // column 8 committed, now displayed
        bad = 0;
        for (i = 0; i < 384; i = i + 1) if (dut.raw_disp[i] !== 8'h00) bad = bad + 1;
        chk("D45_t3h_b1_raw_payload_nonzero", bad > 0);
        chk("D44_t3h_b1_complete_column_without_lut",
            (col_valid === 1'b1) && (lut_valid === 1'b0));
        t3h_w1 = 1'b1;
        while (cyc < 700) @(posedge clk);
        t3h_w1 = 1'b0;
        $display("T3H_B1_SAMPLES n=%0d violations=%0d", t3h_s1, t3h_v1);
        chk("D47_t3h_b1_samples_not_vacuous", t3h_s1 > 300);
        chk("D46_t3h_b1_ch_en_exactly_zero",  t3h_v1 == 0);

        // Stopping mid-window leaves the DUT's `armed` at 1, which would make
        // safe_cfg false and REFUSE the commit in B2. A reset clears it. No
        // 384-byte window is aborted here: the byte stream already closed at
        // cyc 624 (the census counts the same window as closed).
        stim_en = 1'b0;
        repeat (10) @(negedge clk);
        reset_n = 1'b0; #20; reset_n = 1'b1;
        repeat (10) @(negedge clk);

        // ---- B2: real COMMIT, then RELOAD a legal column --------------------
        // PLAN ARITHMETIC (important): the plan for slot s is only latched into
        // cur_* at the END of slot s-1, so slot 0 still uses the initial
        // cur_mode=0. Display slot s+2 shows the column loaded in slot s, so a
        // RAMP payload (mode 1) is only guaranteed from slot 1 onwards - hence
        // B2 checks slot 3's window, never slot 2's.
        for (i = 0; i < 256; i = i + 1) lut_write(i, lut_tab[i]);
        lut_commit_pulse;
        repeat (10) @(negedge dclk);
        chk("D48_t3h_b2_lut_committed", lut_valid === 1'b1);

        for (i = 0; i < 8; i = i + 1) begin
            pcol[i] = 12 + i; pmode[i] = 1; pnb[i] = 384; pfrm[i] = 1;
        end
        slot_rst = 1'b1;
        repeat (2) @(posedge clk);
        slot_rst = 1'b0;
        stim_en  = 1'b1;
        for (i = 0; i < 384; i = i + 1) on_cnt[i] = 0;
        count_slot = 3; counting = 1'b1;
        goto_slot(3);                        // ramp column displayed, LUT valid
        chk("D49_t3h_b2_reloaded_column_valid", col_valid === 1'b1);
        bad = 0;
        for (i = 0; i < 384; i = i + 1)
            if (dut.map_disp[i] !== lut_tab[dut.raw_disp[i]]) bad = bad + 1;
        chk("D50_t3h_b2_map_matches_lut", bad == 0);
        bad = 0;
        for (i = 0; i < 384; i = i + 1)
            if (ch_pam[i*3 +: 3] !== lut_tab[dut.raw_disp[i]][8:6]) bad = bad + 1;
        chk("D51_t3h_b2_pam_matches_lut", bad == 0);
        t3h_w3 = 1'b1;                       // watch the natural BLANK rise
        run_to_window_end;                   // all 64 ticks, BLANK at cyc 768
        counting = 1'b0; count_slot = -1;
        #5;
        t3h_w3 = 1'b0;
        chk("D55_t3h_b3_blank_kills_at_rising_edge",
            (t3h_blank_viol == 0) && (t3h_blank_edges >= 1));
        bad = 0;
        for (i = 0; i < 384; i = i + 1) if (on_cnt[i] != e_eff(i & 255)) bad = bad + 1;
        chk("D52_t3h_b2_on_time_matches_lut", bad == 0);
        bad = 0;
        for (i = 0; i < 384; i = i + 1) bad = bad + on_cnt[i];
        chk("D53_t3h_b2_conduction_nonzero", bad > 0);
        chk("D54_t3h_b2_full_code_64_ticks", on_cnt[255] == 64);

        // ---- B3: fault / reset priority with a valid LUT --------------------
        goto_slot(4);                        // ramp column displayed, LUT valid
        while (cyc < 640) @(posedge clk);    // byte window closed at 624
        chk("D56_t3h_b3_display_active", (|ch_en) === 1'b1);
        fault_n = 1'b0; #1;
        chk("D57_t3h_b3_fault_kills_active", (|ch_en) === 1'b0);
        fault_n = 1'b1; #1;
        chk("D58_t3h_b3_fault_release_recovers", (|ch_en) === 1'b1);
        reset_n = 1'b0; #1;
        chk("D59_t3h_b3_reset_kills_active", (|ch_en) === 1'b0);
        reset_n = 1'b1;
        stim_en = 1'b0;
        repeat (20) @(negedge clk);

        // ================= phase 8 (T3-H2) ==================================
        // Reviewer T3-H2. Runs after P7 (reset pulsed -> lut_valid=0, col_valid=0)
        // and before the D42 census, so every window opened here is counted.
        //   dut0 = USE_DIN_VALID=0 (V1.3 段0548 self-delimiting receiver)
        //   commit is always armed INSIDE a display window, so the only safe_cfg
        //   edge is the 828 LATCH-rise edge: the same-edge collision is therefore
        //   deterministic, not accidental.
        $display("--- P8 T3-H2: mode-0 self-delimit + COMMIT/RELOAD interlock ---");
        for (i = 0; i < NSLOT; i = i + 1) begin
            pcol[i] = i; pmode[i] = 1; pnb[i] = 384; pfrm[i] = 1;
        end
        slot_rst = 1'b1;
        repeat (2) @(posedge clk);
        slot_rst = 1'b0;
        stim_en  = 1'b1;

        // ---- (a) boot: a complete column with the LUT never committed -------
        wait_col_valid1(4000);
        wait_display(4000);
        chk("D60_t3h2_boot_column_without_lut",
            (col_valid === 1'b1) && (lut_valid === 1'b0) &&
            (col_valid0 === 1'b1) && (lut_valid0 === 1'b0));
        chk("D61_t3h2_boot_both_chips_dark",
            ((|ch_en) === 1'b0) && ((|ch_en0) === 1'b0));

        // ---- (b) COMMIT table A, forced onto the LATCH-rise edge ------------
        for (i = 0; i < 256; i = i + 1) lut_write(i, lut_tab[i]);
        wait_display(6000);              // land early inside a display window
        k0 = commit_events;
        lut_commit_pulse;                // safe_cfg=0 here, so it must wait
        wait_commit_events(k0 + 1, 2000); // bank must actually FLIP on the 828 edge
        #1;
        chk("D62_t3h2_commit_fires_on_latch_edge",
            (commit_events == k0 + 1) && (last_pre_ok === 1'b1) &&
            (last_post_colv === 2'b00) && (last_both_bank_same === 1'b1));
        chk("D63_t3h2_commit_same_edge_clears_valid",
            (col_valid === 1'b0) && (col_valid0 === 1'b0));

        dark_samples = 0; dark_viol = 0; dark_blank_low = 0;
        dark_watch = 1'b1; dark_done = 1'b0;   // continuous "must stay dark"
        wait_display(4000);
        repeat (150) @(posedge clk);           // dwell inside the dark window
        chk("D64_t3h2_commit_without_reload_window_dark",
            (col_valid === 1'b0) && ((|ch_en) === 1'b0) && ((|ch_en0) === 1'b0) &&
            (dark_viol == 0) && (dark_blank_low > 10));

        // ---- (c) a column LOADED after the switch must re-enable display ----
        wait_col_valid1(4000);
        chk("D65_t3h2_reload_reenables_valid",
            (col_valid === 1'b1) && (lut_valid === 1'b1) &&
            (dark_done === 1'b1) && (dark_viol == 0) && (dark_samples > 50));
        $display("T3H2_DARK_A samples=%0d blank_low=%0d viol=%0d done=%0d",
                 dark_samples, dark_blank_low, dark_viol, dark_done);
        wait_display(4000);
        bad = 0;
        for (i = 0; i < 384; i = i + 1)
            if (dut.map_disp[i] !== lut_tab[i & 255]) bad = bad + 1;
        chk("D66_t3h2_reload_bank_uses_committed_table", bad == 0);
        chk("D67_t3h2_reload_conduction_nonzero",
            ((|ch_en) === 1'b1) && ((|ch_en0) === 1'b1));
        // scenario validity: table B must really differ, otherwise D66 proves
        // nothing about "the display bank came from the NEW image"
        bad = 0;
        for (i = 0; i < 384; i = i + 1)
            if (lut_b[i & 255] !== lut_tab[i & 255]) bad = bad + 1;
        chk("D68_t3h2_table_b_really_differs", bad > 100);

        // ---- (d0, new in RUN02) mode-0 bytes vs the TB's OWN expectation ----
        // Two consecutive display windows = two consecutive LOAD slots, which
        // alternate 976/977. Each byte is compared against exp_byte(), i.e. the
        // TB's prior, never against the other DUT (so "both DUTs wrong the same
        // way" cannot pass). Indices 0 and 383 are called out explicitly.
        @(posedge blank);                     // leave the current window
        for (k = 0; k < 2; k = k + 1) begin
            wait_display(4000);
            len_obs[k] = slot_len;
            bad = 0;
            for (i = 0; i < 384; i = i + 1)
                if (dut0.raw_disp[i] !== exp_byte(0, 1, i, 1)) bad = bad + 1;
            if (dut0.raw_disp[0]   !== exp_byte(0, 1, 0,   1)) bad = bad + 1;
            if (dut0.raw_disp[383] !== exp_byte(0, 1, 383, 1)) bad = bad + 1;
            if (k == 0) chk("D79_t3h2_mode0_bytes_exact_phase1", bad == 0);
            else        chk("D80_t3h2_mode0_bytes_exact_phase2", bad == 0);
            @(posedge blank);
        end
        chk("D81_t3h2_both_slot_phases_covered",
            ((len_obs[0] == 976) && (len_obs[1] == 977)) ||
            ((len_obs[0] == 977) && (len_obs[1] == 976)));
        $display("T3H2_MODE0_PHASES len0=%0d len1=%0d", len_obs[0], len_obs[1]);

        // ---- (d) the 385th DCLK edge must be ignored ------------------------
        // Evaluate inside the load window: the byte stream completes at cyc 624
        // and wcnt is only cleared at the 828 LATCH rise, so any sample in
        // [700,800) has already seen ~76..176 extra DCLK edges that must not
        // have written anything.
        while (!((cyc >= 700) && (cyc < 800))) @(posedge clk);
        chk("D69_t3h2_edge385_ignored",
            (dut0.wcnt === 10'd384) && (dut0.ovf === 1'b0) && (dut.ovf === 1'b0));

        // ---- (e) mode 0 must have received the same 384 bytes with its
        //          din_valid PORT TIED TO 1'b0 (structural, cannot be faked) --
        bad = 0;
        for (i = 0; i < 384; i = i + 1) if (dut0.raw_disp[i] !== dut.raw_disp[i]) bad = bad + 1;
        chk("D70_t3h2_mode0_matches_with_valid_tied_low",
            (col_valid === 1'b1) && (col_valid0 === 1'b1) && (bad == 0));
        bad = 0;
        for (i = 0; i < 384; i = i + 1) if (dut0.raw_disp[i] === 8'h00) bad = bad + 1;
        chk("D71_t3h2_mode0_payload_not_all_zero", bad < 384);

        // ---- (f) COMMIT table B while FAULT is asserted ---------------------
        // This is the real A->B switch: the active image becomes lut_b, which
        // differs from lut_tab on every non-zero code. RUN02 replaced the old
        // (vacuous) old counter-snapshot guard with bank-level evidence.
        for (i = 0; i < 256; i = i + 1) lut_write(i, lut_b[i]);
        fault_n = 1'b0;
        #10;
        k0 = commit_events;
        bank_at_fault = lut_bank;
        lut_commit_pulse;
        f_samples = 0; f_bank_flip = 0; f_pend_seen = 1'b0; f_on_viol = 0;
        repeat (400) begin
            @(posedge dclk);
            f_samples   = f_samples + 1;
            if (lut_bank !== bank_at_fault) f_bank_flip = f_bank_flip + 1;
            if (dut.cmt_pend === 1'b1) f_pend_seen = 1'b1;
            if (((|ch_en) !== 1'b0) || ((|ch_en0) !== 1'b0)) f_on_viol = f_on_viol + 1;
        end
        chk("D72_t3h2_fault_holds_pending_commit",
            (f_samples >= 400) && (f_bank_flip == 0) && (f_pend_seen === 1'b1) &&
            (commit_events == k0) && (f_on_viol == 0));
        $display("T3H2_FAULT_HOLD samples=%0d bank_flip=%0d pend_seen=%0d events=%0d on_viol=%0d",
                 f_samples, f_bank_flip, f_pend_seen, commit_events, f_on_viol);

        fault_n = 1'b1;
        wait_commit_events(k0 + 1, 4000);      // wait for a real BANK FLIP
        #1;
        chk("D73_t3h2_fault_release_fires_commit",
            (commit_events == k0 + 1) && (last_both_bank_same === 1'b1));
        chk("D74_t3h2_fault_commit_invalidates", last_post_colv === 2'b00);

        dark_samples = 0; dark_viol = 0; dark_blank_low = 0;
        dark_watch = 1'b1; dark_done = 1'b0;   // continuous dark interval
        wait_display(4000);
        repeat (150) @(posedge clk);           // dwell inside the dark window
        chk("D75_t3h2_post_fault_window_dark",
            (col_valid === 1'b0) && ((|ch_en) === 1'b0) && ((|ch_en0) === 1'b0) &&
            (dark_viol == 0) && (dark_blank_low > 10));
        wait_col_valid1(4000);
        chk("D76_t3h2_post_fault_reload_reenables",
            (col_valid === 1'b1) && (dark_done === 1'b1) && (dark_viol == 0));
        $display("T3H2_DARK_B samples=%0d blank_low=%0d viol=%0d done=%0d",
                 dark_samples, dark_blank_low, dark_viol, dark_done);
        wait_display(4000);
        chk("D77_t3h2_post_fault_conduction_nonzero",
            ((|ch_en) === 1'b1) && ((|ch_en0) === 1'b1));
        // the displayed bank must now come from table B, not from table A
        bad = 0;
        for (i = 0; i < 384; i = i + 1)
            if (dut.map_disp[i] !== lut_b[i & 255]) bad = bad + 1;
        chk("D78_t3h2_display_follows_new_table_after_switch", bad == 0);

        // ---- (g) stop outside any byte window (keeps the D42 census exact) --
        while (!((cyc > 660) && (cyc < 800))) @(posedge clk);
        stim_en = 1'b0;
        repeat (20) @(negedge clk);

        // ================= global safety monitors ===========================
        // reviewer request (RUN04): the per-window byte count must be measured
        // on REAL DCLK rising edges by the TB itself, AND the number of windows
        // that should have completed must match the number actually counted -
        // otherwise a census that silently drops windows could fake a PASS.
        //   opened384   : windows whose plan was 384 bytes (counted at the
        //                 LATCH-rise preparation)
        //   closed384   : ... that closed NORMALLY with exactly 384 edges
        //   dropped     : ... aborted by the final reset (must be <= 1)
        // so opened384 == closed384 + dropped must hold exactly.
        chk("D42_window_384_dclk_edges",
            (win_open_384 == win_closed_384 + win_dropped) &&
            (win_dropped <= 1) &&
            (win_closed_384 >= 8) &&
            (win_edges_seen == win_closed_384) &&
            (win_edges_min == 384) && (win_edges_max == 384));
        $display("T3_WINDOW_CENSUS opened384=%0d closed384=%0d counted=%0d min=%0d max=%0d dropped=%0d",
                 win_open_384, win_closed_384, win_edges_seen,
                 win_edges_min, win_edges_max, win_dropped);
        chk("D38_never_on_while_blank",       n_blank_viol == 0);
        chk("D39_never_on_without_valid_col", n_novalid_viol == 0);
        chk("D41_on_samples_not_vacuous",     n_on_samples > 3000);

        $display("COURSE_RTL_ASSERT_COUNT=%0d", assert_count);
        $display("COURSE_RTL_PASS_COUNT=%0d",   pass_count);
        $display("COURSE_RTL_FAIL_COUNT=%0d",   fail_count);
        if (fail_count == 0) begin
            $display("COURSE_RTL_RESULT=PASS");
            $display("COURSE_TCON_T3_SCENARIO_DONE=DATA_DRIVER_INTERNAL");
        end else begin
            $display("COURSE_RTL_RESULT=FAIL");
            $fatal(1, "COURSE_ASSERT_FAILED: %0d of %0d checks", fail_count, assert_count);
        end
        $finish;
    end
