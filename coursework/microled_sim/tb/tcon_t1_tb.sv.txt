`timescale 1ns / 1ps
// TCON T1 skeleton testbench.
//
// Scenarios (selected with +SCEN=<name>):
//   RESET_FAULT : reset state, "not ready => BLANK", fault -> BLANK
//   TIMEBASE    : slot lengths 976/977, latch/pwm_start/scan_stb offsets,
//                 40 ns pulse widths, 64 PWM ticks per window, COL_ID stepping
//   BOUNDARY    : one full frame = 1,000,000 CLK_REF cycles, 576 long slots,
//                 448 short slots, wrap 1023 -> 0
//
// Everything is observed from the DUT outputs only: the TB never drives
// blank / latch / pwm_start / scan_stb / col_id / pwm_tick. That is the point
// of T1 (V1.3 requires the TCON to generate its own timing).
//
// SCOPE: skeleton only. No 768-byte receive, no buffer, no D0/D1 payload.
module tcon_t1_tb;

    // ---------------- clock / DUT -----------------------------------------
    reg clk = 1'b0;
    always #5 clk = ~clk;                 // 100 MHz, 10 ns period

    reg        reset_n   = 1'b0;
    reg        cfg_valid = 1'b0;
    reg  [1:0] cfg_mode  = 2'd1;          // Hybrid
    reg        run_en    = 1'b0;
    reg        rx_ready  = 1'b0;
    reg        fault_n   = 1'b1;
    reg        vsync     = 1'b0;
    reg        col_sync  = 1'b0;
    reg  [7:0] video_d   = 8'h00;
    reg        de        = 1'b0;

    wire [9:0] col_id;
    wire       scan_stb, latch, pwm_start, pwm_tick, blank, dclk, load_active;
    wire [7:0] d0, d1, status;
    wire       frame_wrap, slot_long, slot_start;

    tcon_top dut (
        .clk_ref (clk),      .reset_n (reset_n),
        .pclk (clk),         .vsync (vsync),   .col_sync (col_sync),
        .video_d (video_d),  .de (de),
        .cfg_valid (cfg_valid), .cfg_mode (cfg_mode), .run_en (run_en),
        .rx_ready (rx_ready),   .fault_n (fault_n),
        .col_id (col_id),    .scan_stb (scan_stb),
        .latch (latch),      .pwm_start (pwm_start), .pwm_tick (pwm_tick),
        .blank (blank),      .dclk (dclk),           .load_active (load_active),
        .d0 (d0),            .d1 (d1),
        .status (status),    .frame_wrap (frame_wrap),
        .slot_long (slot_long), .slot_start (slot_start)
    );

    // ---------------- bookkeeping -----------------------------------------
    integer assert_count = 0, pass_count = 0, fail_count = 0;
    integer blk_viol = 0, latch_viol = 0, stb_viol = 0;
    string  scen;

    // Black-box safety monitors: a tick must never occur while blanked, and
    // latch / scan_stb must never leave the BLANK region.
    always @(negedge clk) begin
        if (reset_n) begin
            if (pwm_tick && blank)  blk_viol   = blk_viol   + 1;
            if (latch && !blank)    latch_viol = latch_viol + 1;
            if (scan_stb && !blank) stb_viol   = stb_viol   + 1;
        end
    end

    task automatic chk(input string nm, input logic cond);
        assert_count = assert_count + 1;
        if (cond) begin
            pass_count = pass_count + 1;
            $display("PASS %s", nm);
        end else begin
            fail_count = fail_count + 1;
            $display("FAIL %s  (t=%0t)", nm, $time);
            $display("COURSE_RTL_ASSERT_COUNT=%0d", assert_count);
            $display("COURSE_RTL_FAIL_COUNT=%0d", fail_count);
            $display("COURSE_RTL_RESULT=FAIL");
            $display("COURSE_FAILED_ASSERTION=%s", nm);
            $fatal(1, "COURSE_ASSERT_FAILED: %s", nm);
        end
    endtask

    task automatic step(input integer n);
        repeat (n) @(negedge clk);
    endtask

    // ---------------- scenario 1: reset / fault ---------------------------
    task automatic scen_reset_fault;
        $display("--- SCENARIO RESET_FAULT ---");
        reset_n = 1'b0; cfg_valid = 1'b0; run_en = 1'b0;
        rx_ready = 1'b0; fault_n = 1'b1; cfg_mode = 2'd1;
        step(4);
        chk("R1_reset_blank_high",    blank === 1'b1);
        chk("R1_reset_col_id_0",      col_id === 10'd0);
        chk("R1_reset_no_latch",      latch === 1'b0);
        chk("R1_reset_no_pwm_start",  pwm_start === 1'b0);
        chk("R1_reset_no_scan_stb",   scan_stb === 1'b0);
        chk("R1_reset_no_tick",       pwm_tick === 1'b0);
        chk("R1_reset_no_load",       load_active === 1'b0);

        reset_n = 1'b1;                       // released, still unconfigured
        step(40);
        chk("R2_unconfigured_blank_high", blank === 1'b1);
        chk("R2_unconfigured_no_stb",     scan_stb === 1'b0);

        cfg_valid = 1'b1; cfg_mode = 2'd2;    // illegal mode
        run_en = 1'b1; rx_ready = 1'b1;
        step(40);
        chk("R3_illegal_mode_blank_high", blank === 1'b1);

        cfg_mode = 2'd1;                      // legal -> unblank
        step(1200);
        chk("R4_ready_blank_low", blank === 1'b0);
        chk("R4_status_ready",    status[0] === 1'b1);

        fault_n = 1'b0;                       // fault -> BLANK immediately
        step(1);
        chk("R5_fault_blank_within_1_cycle", blank === 1'b1);
        chk("R5_fault_no_scan_stb",          scan_stb === 1'b0);
        step(400);
        chk("R5_fault_holds_blank",          blank === 1'b1);
        chk("R5_fault_sticky_status",        status[3] === 1'b1);

        fault_n = 1'b1;                       // recovery
        step(1200);
        chk("R6_recover_blank_low", blank === 1'b0);
        chk("R6_no_tick_while_blanked", blk_viol == 0);
        chk("R6_no_latch_outside_blank", latch_viol == 0);
    endtask

    task automatic reset_dut;
        reset_n = 1'b0; cfg_valid = 1'b0; run_en = 1'b0;
        rx_ready = 1'b0; fault_n = 1'b1; vsync = 1'b0; col_sync = 1'b0; de = 1'b0;
        step(20);
        reset_n = 1'b1;
        step(40);
    endtask

    // ---------------- scenario 2: time base and control pulses ------------
    integer i;
    realtime t_start, t_next, t_latch, t_lat_fall, t_pstart, t_stb;
    integer  len_cycles, tick_cnt, n_long, n_short;

    task automatic measure_slot(output integer len_c, output integer ticks,
                                output integer latch_off, output integer pstart_off,
                                output integer stb_off, output integer latch_pw);
        integer dummy_pw;
        @(posedge slot_start);
        t_start = $realtime;
        t_latch = -1; t_lat_fall = -1; t_pstart = -1; t_stb = -1;
        ticks = 0;
        fork
            begin : obs
                forever begin
                    @(negedge clk);
                    if (pwm_tick) ticks = ticks + 1;
                    if (latch && t_latch < 0)           t_latch    = $realtime;
                    if (!latch && t_latch >= 0 && t_lat_fall < 0) t_lat_fall = $realtime;
                    if (pwm_start && t_pstart < 0)      t_pstart   = $realtime;
                    if (scan_stb && t_stb < 0)          t_stb      = $realtime;
                end
            end
            begin : stopobs
                @(posedge slot_start);
                disable obs;
            end
        join
        t_next     = $realtime;
        // NOTE: SystemVerilog rounds a real-to-integer assignment (828.5 -> 829).
        // The samples sit half a cycle after the slot edge, so $rtoi (truncate)
        // is required to recover the true cycle index.
        len_c      = $rtoi((t_next  - t_start) / 10.0);
        latch_off  = $rtoi((t_latch - t_start) / 10.0);
        pstart_off = $rtoi((t_pstart - t_start) / 10.0);
        stb_off    = $rtoi((t_stb   - t_start) / 10.0);
        latch_pw   = $rtoi((t_lat_fall - t_latch) / 10.0);
        dummy_pw   = 0;
    endtask

    task automatic scen_timebase;
        integer len_c, ticks, l_off, p_off, s_off, l_pw;
        $display("--- SCENARIO TIMEBASE ---");
        reset_dut;
        reset_n = 1'b1; cfg_valid = 1'b1; cfg_mode = 2'd1;
        run_en = 1'b1; rx_ready = 1'b1; fault_n = 1'b1;
        step(1200);

        measure_slot(len_c, ticks, l_off, p_off, s_off, l_pw);
        $display("SLOT1 len=%0d ticks=%0d latch_off=%0d pstart_off=%0d stb_off=%0d latch_pw=%0d",
                 len_c, ticks, l_off, p_off, s_off, l_pw);
        chk("T1_slot_len_976_or_977",  (len_c == 976) || (len_c == 977));
        chk("T1_ticks_64_per_window",  ticks == 64);
        chk("T1_latch_offset_828",     l_off == 828);
        chk("T1_pwm_start_offset_840", p_off == 840);
        chk("T1_scan_stb_offset_880",  s_off == 880);
        chk("T1_latch_pulse_40ns",     l_pw == 4);
        chk("T1_no_tick_in_blank",     blk_viol == 0);

        measure_slot(len_c, ticks, l_off, p_off, s_off, l_pw);
        $display("SLOT2 len=%0d ticks=%0d latch_off=%0d", len_c, ticks, l_off);
        chk("T2_slot_len_976_or_977", (len_c == 976) || (len_c == 977));
        chk("T2_ticks_64",            ticks == 64);
        chk("T2_latch_offset_828",    l_off == 828);

        measure_slot(len_c, ticks, l_off, p_off, s_off, l_pw);
        $display("SLOT3 len=%0d ticks=%0d col_id=%0d", len_c, ticks, col_id);
        chk("T3_slot_len_976_or_977", (len_c == 976) || (len_c == 977));
        chk("T3_ticks_64",            ticks == 64);
        chk("T3_col_id_advancing",    col_id >= 10'd2);
        chk("T3_dclk_toggling",       dclk === 1'b0 || dclk === 1'b1);
        chk("T3_latch_outside_blank_viol0", latch_viol == 0);
    endtask

    // ---------------- scenario 3: full-frame boundary ---------------------
    task automatic scen_boundary;
        integer slots, longs, shorts, prev_len, start_col;
        realtime t_prev, t_now;
        $display("--- SCENARIO BOUNDARY ---");
        reset_dut;
        reset_n = 1'b1; cfg_valid = 1'b1; cfg_mode = 2'd1;
        run_en = 1'b1; rx_ready = 1'b1; fault_n = 1'b1;
        step(1200);

        // The slot-length sequence is periodic with period 1024, so ANY run of
        // 1024 consecutive slots sums to exactly one frame (1,000,000 cycles)
        // and contains exactly 576 long / 448 short slots. Counting a fixed
        // 1024 slots is therefore phase-invariant and needs no frame alignment
        // (an alignment loop would double the simulated time and timed out).
        @(posedge slot_start);
        t_prev = $realtime;
        #1;
        start_col = dut.col_id;
        slots = 0; longs = 0; shorts = 0;

        while (slots < 1024) begin
            @(posedge slot_start);
            t_now = $realtime;
            #1;
            prev_len = $rtoi((t_now - t_prev) / 10.0);
            t_prev   = t_now;
            if (prev_len == 977)      longs  = longs  + 1;
            else if (prev_len == 976) shorts = shorts + 1;
            slots = slots + 1;
        end

        $display("FRAME slots=%0d long=%0d short=%0d cycles=%0d start_col=%0d end_col=%0d",
                 slots, longs, shorts, longs * 977 + shorts * 976, start_col, dut.col_id);
        chk("B1_frame_cycles_1e6",  (longs * 977 + shorts * 976) == 1000000);
        chk("B2_long_slots_576",    longs == 576);
        chk("B3_short_slots_448",   shorts == 448);
        chk("B4_total_slots_1024",  slots == 1024);
        chk("B5_col_id_period_1024", dut.col_id == start_col);
        chk("B6_tick_viol_zero_over_frame", blk_viol == 0);
        chk("B7_latch_viol_zero_over_frame", latch_viol == 0);
        chk("B8_stb_viol_zero_over_frame",   stb_viol == 0);
    endtask

    // ---------------- main ------------------------------------------------
    initial begin
        if (!$value$plusargs("SCEN=%s", scen)) scen = "RESET_FAULT";
        $display("COURSE_TCON_T1_SCENARIO=%s", scen);
        // The full-frame scan runs 1,000,000 cycles; dumping it produces a
        // ~100 MB VCD and made the run hit the wall-clock timeout. Only the
        // short scenarios are dumped; the boundary scan is judged from counters.
        if (scen != "BOUNDARY") begin
            $dumpfile("tcon_t1.vcd");
            $dumpvars(0, tcon_t1_tb);
        end

        case (scen)
            "RESET_FAULT": scen_reset_fault();
            "TIMEBASE":    scen_timebase();
            "BOUNDARY":    scen_boundary();
            "ALL": begin
                scen_reset_fault();
                reset_dut;
                scen_timebase();
                reset_dut;
                // keep the waveform small: the 1,000,000-cycle frame scan needs
                // no waveform, only the counters checked in scen_boundary
                $dumpoff;
                scen_boundary();
                $dumpon;
            end
            default: begin
                $display("UNKNOWN_SCENARIO");
                $fatal(1, "unknown scenario");
            end
        endcase

        $display("COURSE_RTL_ASSERT_COUNT=%0d", assert_count);
        $display("COURSE_RTL_PASS_COUNT=%0d",   pass_count);
        $display("COURSE_RTL_FAIL_COUNT=%0d",   fail_count);
        $display("COURSE_RTL_RESULT=%s", (fail_count == 0) ? "PASS" : "FAIL");
        $display("COURSE_TCON_T1_SCENARIO_DONE=%s", scen);
        $finish;
    end

endmodule
