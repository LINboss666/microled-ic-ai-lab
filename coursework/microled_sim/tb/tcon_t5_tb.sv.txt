`timescale 1ns / 1ps
// TCON T5-B — five REAL chips: 1 x tcon_top + 2 x data_driver + 2 x scan_driver.
//
// This testbench is the one the T5-B task book asks for. Nothing here is a
// behaviour model: the DUTs are the same RTL that passed T3 / T3-H / T3-H2 /
// T4, wired through tb/t5_five_chip_shell.sv.
//
// WHAT IS PROVEN (all at the pin, none at a DUT counter)
//   * every load window carries exactly 384 bytes to D0 and to D1 on REAL DCLK
//     rising edges (data_pin_mon, no reference clock, half-open window);
//   * inside every legal display window, the column that the two Scan chips
//     actually select (1024-bit pooled one-hot) is the column whose 768 bytes
//     are sitting in the two Data display banks, and those bytes are the ones
//     the TB itself scheduled two slots earlier - byte for byte, 384+384 of
//     them, plus the LUT mapping and the PAM tier of every channel;
//   * the per-channel PWM conduction width equals the LUT effective width;
//   * safety: no conduction without a committed LUT, none after a LUT switch
//     until a full column was re-loaded, none while FAULT is latched, and no
//     automatic recovery from a latched fault.
//
// INDEPENDENCE (why this is not a self-fulfilling test)
//   The expected (frame, column, lane) of every display window comes from the
//   TB's OWN slot model (the 576/448 recurrence and a 2-slot pipeline depth
//   derived from the frozen phase map), never from disp_col / fwd_col /
//   slot_cycle. The DUT side is only ever READ: raw_disp / map_disp / ch_pam /
//   col_a are observations. `raw_disp` is not a synthesizable output port; the
//   course TB is allowed a read-only hierarchical path to it.
//
// PAYLOAD (per lane, local channel i, global row = 2*i + lane)
//   i = 0 -> col[7:0]
//   i = 1 -> {col[9:8], frame[5:0]}
//   i > 1 -> ((col & 255) + 13*(2*i+lane) + 7*frame + 0x5A) & 255
// so 10 bits of column and 6 bits of frame are recoverable from two bytes of
// EACH lane, and 0/256/512/768 are distinguishable (the T5-A `col*7+row*13`
// pattern repeated every 256 columns and is retired).
//
// TIMING (why the first capture edge really is byte 0)
//   The TCON time base restarts slot 0 on the rising edge of mode_en. This TB
//   raises run_en on the negedge where DCLK is HIGH, so that restart edge is a
//   1->0 DCLK toggle and DCLK is therefore LOW during slot cycles 0,2,4,... .
//   LATCH falls at the end of cycle 831 -> the Data chip opens its window on
//   the DCLK edge at cycle 833 with byte 0 on the bus, and 384 edges later the
//   LATCH rising edge at cycle 828 of the next slot commits it.
module tcon_t5_tb;

        .tcon_in_col (tcon_in_col), .tcon_disp_col (tcon_disp_col),

        .tcon_bytes_rcvd (tcon_bytes_rcvd), .tcon_bytes_sent (tcon_bytes_sent),

    );

    // ======================================================================
    wire [15:0] dp0_win, dp0_badlen, dp0_bytebad;
    wire        dp0_x;
    wire [31:0] dp0_edge;
    wire [9:0]  dp0_bytes, dp0_col;

    wire [15:0] dp1_win, dp1_badlen, dp1_bytebad;

    data_pin_mon #(.CHIP(0)) u_dpm0 (

        .n_win (dp0_win), .n_win_badlen (dp0_badlen), .n_byte_bad (dp0_bytebad),

        .n_edge (dp1_edge), .last_bytes (dp1_bytes), .last_col (dp1_col),
        .last_frame (dp1_frm), .any_x (dp1_x)
    );

    wire [15:0] spm_stb, spm_oob, spm_setbad, spm_holdbad;
    wire [9:0]  spm_col;
    wire        spm_blank;
    wire [31:0] spm_setup_ps, spm_hold_ps;
    scan_pin_mon u_spm (
        .reset_n (reset_n), .scan_stb (tcon_scan_stb), .col_id (tcon_col_id),
        .blank (tcon_blank), .n_stb (spm_stb), .n_outside_blank (spm_oob),
        .n_setup_bad (spm_setbad), .n_hold_bad (spm_holdbad),
        .last_col (spm_col), .last_blank (spm_blank),
        .setup_min_ps (spm_setup_ps), .hold_min_ps (spm_hold_ps)

    // ======================================================================
    // payload / LUT model (the ONLY source of expected values)
    // ======================================================================
    reg [8:0] lut_at [0:255];      // table A - the accepted gray LUT image
    reg [8:0] lut_bt [0:255];      // table B - derived, see build_lut_b
    integer   cur_tab;             // 0 = A active, 1 = B active (TB's own state)

    function automatic [7:0] pixf(input integer f, input integer c, input integer r);
        pixf = ((c & 255) + 13 * (r & 255) + 7 * (f & 255) + 8'h5A) & 255;
    endfunction

    function automatic [7:0] pbyte(input integer f, input integer c,
                                   input integer lane, input integer i);
        begin
            if (i == 0)      pbyte = c[7:0];
            else if (i == 1) pbyte = {c[9:8], f[5:0]};
            else             pbyte = pixf(f, c, 2 * i + lane);
        end
    endfunction

    function automatic [8:0] emap(input integer f, input integer c,
                                  input integer lane, input integer i,
                                  input integer tab);
        emap = tab ? lut_bt[pbyte(f, c, lane, i)] : lut_at[pbyte(f, c, lane, i)];
    endfunction

    // V1.3 段0218: 0 = off, 1..62 = that many ticks, 63 = the full 64 ticks
    function automatic integer weff(input integer code);
        if (code == 0)       weff = 0;
        else if (code == 63) weff = 64;
        else                 weff = code;
    endfunction

    // index of the single set bit of a 512-bit one-hot vector (binary reduction,
    // 9 iterations instead of 512 so a 1024-column frame stays affordable)
    function automatic integer ohidx(input [511:0] v);
        reg [511:0] t;
        integer     base, k;
        begin
            t = v; base = 0;
            for (k = 256; k >= 1; k = k / 2) begin
                if (|(t >> k)) begin base = base + k; t = t >> k; end
            end
            ohidx = base;
        end
    endfunction

    // ======================================================================
    // bookkeeping / assertion plumbing
    // ======================================================================
    integer assert_count = 0, pass_count = 0, fail_count = 0;

    task automatic chk(input string nm, input logic cond);
        begin
            assert_count = assert_count + 1;
            if (cond) begin
                pass_count = pass_count + 1;
                $display("PASS %s", nm);
            end else begin
                fail_count = fail_count + 1;
                $display("FAIL %s  (t=%0t)", nm, $time);
                $display("COURSE_FAILED_ASSERTION=%s", nm);
            end
        end
    endtask

    // ---- video driver state ---------------------------------------------
    localparam integer ST_IDLE = 0, ST_ALIGN = 1, ST_ARM = 2, ST_RUN = 3;
    integer drv_state;
    reg     req_start;
    reg     drv_on;
    integer start_col, start_frame, base_frame;
    integer vc, vcol, vframe, vlen, vfrac, slot_done;
    integer nbytes, bad_col, bad_nbytes, vsync_now;
    integer no_win_slots, saw_win;
    integer cvc;

    // ---- window / scoreboard counters ------------------------------------
    integer win_active, blank_d, wc, win_slot, win_ecol, win_efr;
    integer pwm_ch [0:7];
    integer pwm_act [0:15];
    integer pwm_exp [0:15];
    reg [383:0] pwm_cov0, pwm_cov1;

    integer disp_windows, nohot_bad, sel_bad, tag_bad0, tag_bad1, frm_bad;
    integer byte_bad, cont_n, ch_on_bad, blank_novalid;
    integer idle_blanklow, fault_blanklow, dark_viol, dark_samples;
    reg     must_dark;
    integer chk_en;
    integer pwm_bad, pwm_windows;

    reg [1023:0] vall;
    reg [511:0]  v0, v1;
    integer      gcol, i, j;
    integer      wi, wer, wem, wer1, wem1, wbad;
    reg          seen_c0, seen_c1, seen_c2, seen_c1022, seen_c1023, seen_nf0;
    reg [9:0]    tag0, tag1;
    reg [5:0]    frm0, frm1;
    integer      fd_ledger;

    // ======================================================================
    // video driver + slot model  (single writer of every stimulus pin)
    // ======================================================================
    task automatic emit(input integer k);
        begin
            de       = (k < nbytes);
            video_d  = (k < nbytes) ? pbyte(vframe, vcol, (k & 1), (k >> 1)) : 8'h00;
            col_sync = (k == 0);
            vsync    = (k == 0) && (vsync_now == 1);
        end
    endtask

    always @(negedge clk) begin
        if (!reset_n) begin
            drv_state = ST_IDLE;
            drv_on    = 1'b0;
            vc = 0; vcol = 0; vframe = 0; vlen = 976; vfrac = 576; slot_done = 0;
            nbytes = 768; vsync_now = 0; saw_win = 0; no_win_slots = 0;
            de = 1'b0; col_sync = 1'b0; vsync = 1'b0; video_d = 8'h00;
        end else begin
            case (drv_state)
            ST_IDLE: if (req_start) drv_state = ST_ALIGN;

            // wait for the negedge where DCLK is high: the next reference edge
            // then toggles DCLK 1->0 and restarts the time base with DCLK low
            ST_ALIGN: if (tcon_dclk === 1'b1) begin
                run_en    = 1'b1;
                rx_ready  = 1'b1;
                cfg_valid = 1'b1;
                cfg_mode  = 2'd1;
                drv_state = ST_ARM;
            end

            // exactly one negedge later slot cycle 0 has begun
            ST_ARM: begin
                vc = 0; vcol = start_col; vframe = start_frame;
                vlen = 976; vfrac = 576; slot_done = 0;
                nbytes = 768; vsync_now = (start_col == 0) ? 1 : 0;
                if (vcol == bad_col) nbytes = bad_nbytes;
                saw_win = 0; no_win_slots = 0;
                emit(0);
                vc = 1;
                drv_on    = 1'b1;
                drv_state = ST_RUN;
            end

            ST_RUN: begin
                cvc = vc;
                emit(cvc);
                if (!tcon_blank) saw_win = 1;
                vc = cvc + 1;
                if (vc >= vlen) begin
                    vc = 0;
                    slot_done = slot_done + 1;
                    if (!saw_win && (slot_done > 2)) no_win_slots = no_win_slots + 1;
                    saw_win = 0;
                    if (vcol == 1023) begin
                        vcol = 0; vframe = vframe + 1; vsync_now = 1;
                    end else begin
                        vcol = vcol + 1; vsync_now = 0;
                    end
                    if ((vfrac + 576) >= 1024) begin
                        vlen = 977; vfrac = vfrac + 576 - 1024;
                    end else begin
                        vlen = 976; vfrac = vfrac + 576;
                    end
                    nbytes = 768;
                    if (vcol == bad_col) nbytes = bad_nbytes;
                end
            end

            default: drv_state = ST_IDLE;
            endcase
        end
    end

    // ======================================================================
    // display-window monitor: one-hot / content / PWM  (read-only on the DUT)
    // ======================================================================
    task automatic pick_channels;
        integer base, jj;
        begin
            base = ((slot_done - 2) * 8) % 384;
            if (base < 0) base = 0;
            for (jj = 0; jj < 8; jj = jj + 1) pwm_ch[jj] = (base + jj) % 384;
        end
    endtask

    task automatic pwm_arm;
        integer jj, code;
        begin
            for (jj = 0; jj < 8; jj = jj + 1) begin
                pwm_act[jj]     = 0;
                pwm_act[8 + jj] = 0;
                code = emap(win_efr, win_ecol, 0, pwm_ch[jj], cur_tab);
                pwm_exp[jj]     = weff(code[5:0]);
                code = emap(win_efr, win_ecol, 1, pwm_ch[jj], cur_tab);
                pwm_exp[8 + jj] = weff(code[5:0]);
            end
        end
    endtask

    task automatic pwm_judge;
        integer jj;
        begin
            pwm_windows = pwm_windows + 1;
            for (jj = 0; jj < 16; jj = jj + 1) begin
                if (pwm_act[jj] != pwm_exp[jj]) pwm_bad = pwm_bad + 1;
            end
            for (jj = 0; jj < 8; jj = jj + 1) begin
                pwm_cov0[pwm_ch[jj]] = 1'b1;
                pwm_cov1[pwm_ch[jj]] = 1'b1;
            end
        end
    endtask

    task automatic window_body;
        begin
            disp_windows = disp_windows + 1;
            v0 = s0_col_a; v1 = s1_col_a;
            vall = {v1, v0};
            if (vall === 1024'd0) begin
                nohot_bad = nohot_bad + 1;
                gcol = -1;
            end else if ((vall & (vall - 1'b1)) !== 1024'd0) begin
                nohot_bad = nohot_bad + 1;
                gcol = -1;
            end else if (v0 !== 512'd0) begin
                gcol = ohidx(v0) * 2;
            end else begin
                gcol = ohidx(v1) * 2 + 1;
            end
            if (gcol !== win_ecol) sel_bad = sel_bad + 1;

            tag0 = {dut.u_data0.raw_disp[1][7:6], dut.u_data0.raw_disp[0]};
            frm0 =  dut.u_data0.raw_disp[1][5:0];
            tag1 = {dut.u_data1.raw_disp[1][7:6], dut.u_data1.raw_disp[0]};
            frm1 =  dut.u_data1.raw_disp[1][5:0];
            if (tag0 !== win_ecol) tag_bad0 = tag_bad0 + 1;
            if (tag1 !== win_ecol) tag_bad1 = tag_bad1 + 1;
            if ((frm0 !== win_efr[5:0]) || (frm1 !== win_efr[5:0])) frm_bad = frm_bad + 1;

            wbad = 0;
            for (wi = 0; wi < 384; wi = wi + 1) begin
                wer = pbyte(win_efr, win_ecol, 0, wi);
                if (dut.u_data0.raw_disp[wi] !== wer[7:0])                wbad = wbad + 1;
                wem = emap(win_efr, win_ecol, 0, wi, cur_tab);
                if (dut.u_data0.map_disp[wi] !== wem)                     wbad = wbad + 1;
                if (dut.d0_ch_pam[wi*3 +: 3] !== wem[8:6])                wbad = wbad + 1;
                wer1 = pbyte(win_efr, win_ecol, 1, wi);
                if (dut.u_data1.raw_disp[wi] !== wer1[7:0])               wbad = wbad + 1;
                wem1 = emap(win_efr, win_ecol, 1, wi, cur_tab);
                if (dut.u_data1.map_disp[wi] !== wem1)                    wbad = wbad + 1;
                if (dut.d1_ch_pam[wi*3 +: 3] !== wem1[8:6])               wbad = wbad + 1;
            end
            byte_bad = byte_bad + wbad;
            cont_n   = cont_n + 768;

            if (wbad == 0) begin
                if ((win_efr == base_frame)     && (win_ecol == 0))    seen_c0    = 1'b1;
                if ((win_efr == base_frame)     && (win_ecol == 1))    seen_c1    = 1'b1;
                if ((win_efr == base_frame)     && (win_ecol == 2))    seen_c2    = 1'b1;
                if ((win_efr == base_frame)     && (win_ecol == 1022)) seen_c1022 = 1'b1;
                if ((win_efr == base_frame)     && (win_ecol == 1023)) seen_c1023 = 1'b1;
                if ((win_efr == base_frame + 1) && (win_ecol == 0))    seen_nf0   = 1'b1;
            end

            if (((|d0_ch_en) !== 1'b1) && ((|d1_ch_en) !== 1'b1)) ch_on_bad = ch_on_bad + 1;
            if ((d0_col_valid !== 1'b1) || (d1_col_valid !== 1'b1)) blank_novalid = blank_novalid + 1;

            if (fd_ledger != 0)
                $fdisplay(fd_ledger, "%0d,%0d,%0d,%0d,%0d,%0d,%0d,%0d,%0d,%0d",
                          slot_done, win_ecol, win_efr, gcol, tag0, frm0, tag1, frm1,
                          wbad, cur_tab);
        end
    endtask

    always @(negedge clk) begin
        if (!reset_n) begin
            win_active = 0; blank_d = 0; wc = 0; win_slot = 0;
            win_ecol = 0; win_efr = 0;
            pwm_cov0 = 384'd0; pwm_cov1 = 384'd0;
        end else begin
            if (!tcon_blank && (blank_d == 1) && !win_active) begin
                win_active = 1;
                wc         = 0;
                win_slot   = slot_done;
                win_ecol   = (slot_done >= 2) ? ((slot_done - 2) % 1024) : 0;
                win_efr    = (slot_done >= 2) ? (((slot_done - 2) / 1024) + base_frame)
                                              : base_frame;
                pick_channels;
                pwm_arm;
            end
            if (win_active && tcon_pwm_tick) begin
                for (j = 0; j < 8; j = j + 1) begin
                    if (d0_ch_en[pwm_ch[j]] === 1'b1) pwm_act[j]     = pwm_act[j] + 1;
                    if (d1_ch_en[pwm_ch[j]] === 1'b1) pwm_act[8 + j] = pwm_act[8 + j] + 1;
                end
            end
            if (win_active && (wc == 300) && (chk_en != 0)) window_body;
            if (tcon_blank && (blank_d == 0) && win_active) begin
                win_active = 0;
                pwm_judge;
            end
            if (win_active) wc = wc + 1;
            blank_d = tcon_blank;
        end
    end

    // ---- DCLK period, measured on the real pin (Phase B only) -------------
    realtime t_dc;
    realtime dc_min, dc_max;
    reg      dc_en, dc_first;

    always @(posedge tcon_dclk) begin
        if (!dc_en) begin
            dc_first = 1'b0;
        end else if (!dc_first) begin
            dc_first = 1'b1;
            t_dc = $realtime;
        end else begin
            if (($realtime - t_dc) < dc_min) dc_min = $realtime - t_dc;
            if (($realtime - t_dc) > dc_max) dc_max = $realtime - t_dc;
            t_dc = $realtime;
        end
    end

    // ======================================================================
    // global safety / accounting monitors
    // ======================================================================
    always @(negedge clk) begin
        if (reset_n) begin
            if (!tcon_blank && (tcon_fsm_state !== 2'd2)) idle_blanklow = idle_blanklow + 1;
            if ((tcon_fsm_state === 2'd3) && !tcon_blank) fault_blanklow = fault_blanklow + 1;
            if (must_dark) begin
                dark_samples = dark_samples + 1;
                if (((|d0_ch_en) !== 1'b0) || ((|d1_ch_en) !== 1'b0)) dark_viol = dark_viol + 1;
            end
        end
    end

    // ======================================================================
    // stimulus helpers
    // ======================================================================
    task automatic run_slots(input integer n);
        integer target;
        begin
            target = slot_done + n;
            while (slot_done < target) @(posedge clk);
        end
    endtask

    task automatic wait_dclk(input integer n);
        integer k;
        begin
            for (k = 0; k < n; k = k + 1) @(posedge tcon_dclk);
        end
    endtask

    task automatic wait_lut(input integer maxn);
        integer n;
        begin
            n = 0;
            while (!((d0_lut_valid === 1'b1) && (d1_lut_valid === 1'b1)) && (n < maxn)) begin
                @(posedge tcon_dclk); n = n + 1;
            end
        end
    endtask

    // 256 entries into the INACTIVE bank of both chips, then one COMMIT pulse
    task automatic lut_program(input integer which, input integer do_commit);
        integer k;
        begin
            for (k = 0; k < 256; k = k + 1) begin
                @(negedge tcon_dclk);
                cfg0_we = 1'b1; cfg0_addr = k[7:0];
                cfg0_data = which ? lut_bt[k] : lut_at[k];
                cfg1_we = 1'b1; cfg1_addr = k[7:0];
                cfg1_data = which ? lut_bt[k] : lut_at[k];
            end
            @(negedge tcon_dclk);
            cfg0_we = 1'b0; cfg1_we = 1'b0;
            if (do_commit != 0) begin
                @(negedge tcon_dclk);
                cfg0_cmt = 1'b1; cfg1_cmt = 1'b1;
                @(posedge tcon_dclk);
                @(negedge tcon_dclk);
                cfg0_cmt = 1'b0; cfg1_cmt = 1'b0;
                wait_lut(60);
            end
        end
    endtask

    task automatic power_on_reset;
        begin
            reset_n   = 1'b0;
            cfg_valid = 1'b0;
            run_en    = 1'b0;
            rx_ready  = 1'b0;
            fault_n   = 1'b1;
            cfg_mode  = 2'd1;
            cfg_hybrid = 1'b1;
            cfg0_we = 1'b0; cfg1_we = 1'b0; cfg0_cmt = 1'b0; cfg1_cmt = 1'b0;
            req_start = 1'b0;
            must_dark = 1'b0;
            repeat (100) @(posedge clk);
            @(negedge clk); reset_n = 1'b1;
            repeat (6) @(negedge clk);
        end
    endtask

    task automatic begin_frame(input integer c0, input integer f0);
        begin
            start_col   = c0;
            start_frame = f0;
            base_frame  = f0;
            seen_c0 = 1'b0; seen_c1 = 1'b0; seen_c2 = 1'b0;
            seen_c1022 = 1'b0; seen_c1023 = 1'b0; seen_nf0 = 1'b0;
            req_start   = 1'b1;
            @(posedge clk);
            while (drv_state != ST_RUN) @(posedge clk);
            req_start = 1'b0;
        end
    endtask

    task automatic dump_on;
        begin $dumpon; end
    endtask
    task automatic dump_off;
        begin $dumpoff; end
    endtask

    // ======================================================================
    // scenario
    // ======================================================================
    integer ph_base, nwin_base, nobase, cbase, tbase, sbase, ebase, switch_slot;

    initial begin
        $display("COURSE_T5_SCENARIO=FIVE_REAL_RTL_1TCON_2DATA_2SCAN");
        $display("COURSE_T5_CONTRACT=V13_T5B_FROZEN");

        // ---- defaults ----
        assert_count = 0; pass_count = 0; fail_count = 0;
        reset_n = 1'b0; vsync = 1'b0; col_sync = 1'b0; de = 1'b0; video_d = 8'h00;
        cfg_valid = 1'b0; run_en = 1'b0; rx_ready = 1'b0; fault_n = 1'b1;
        cfg_mode = 2'd1; cfg_hybrid = 1'b1;
        cfg0_we = 1'b0; cfg1_we = 1'b0; cfg0_cmt = 1'b0; cfg1_cmt = 1'b0;
        cfg0_addr = 8'd0; cfg1_addr = 8'd0; cfg0_data = 9'd0; cfg1_data = 9'd0;
        req_start = 1'b0; drv_state = ST_IDLE; drv_on = 1'b0;
        vc = 0; vcol = 0; vframe = 0; vlen = 976; vfrac = 576; slot_done = 0;
        nbytes = 768; bad_col = -1; bad_nbytes = 768; vsync_now = 0;
        base_frame = 0; start_col = 0; start_frame = 0;
        cur_tab = 0; chk_en = 1; must_dark = 1'b0;
        disp_windows = 0; nohot_bad = 0; sel_bad = 0; tag_bad0 = 0; tag_bad1 = 0;
        frm_bad = 0; byte_bad = 0; cont_n = 0; ch_on_bad = 0; blank_novalid = 0;
        idle_blanklow = 0; fault_blanklow = 0; dark_viol = 0; dark_samples = 0;
        pwm_bad = 0; pwm_windows = 0;
        dc_en = 1'b0; dc_first = 1'b0; t_dc = 0.0;
        dc_min = 1.0e9; dc_max = 0.0;
        win_active = 0; blank_d = 0; wc = 0; win_slot = 0; win_ecol = 0; win_efr = 0;
        pwm_cov0 = 384'd0; pwm_cov1 = 384'd0;
        for (i = 0; i < 8; i = i + 1) pwm_ch[i] = 0;
        for (i = 0; i < 16; i = i + 1) pwm_act[i] = 0;
        $readmemh("t5_lut_a.hex", lut_at);
        for (i = 0; i < 256; i = i + 1) begin
            if (i == 0)        lut_bt[i] = 9'h000;
            else if (i == 255) lut_bt[i] = 9'h1FF;
            else               lut_bt[i] = {3'd7 - lut_at[i][8:6], lut_at[i][5:0]};
        end
        fd_ledger = $fopen("t5_column_ledger.csv", "w");
        if (fd_ledger != 0)
            $fdisplay(fd_ledger, "slot,exp_col,exp_frame,scan_col,tag0,frame0,tag1,frame1,bad_bytes,lut_tab");

        $dumpfile("t5.vcd");
        $dumpvars(1, tcon_t5_tb);
        $dumpoff;

        // ==================================================================
        // Phase A — configuration, LUT preload, idle safety
        // ==================================================================
        $display("--- Phase A: reset, LUT preload, idle safety ---");
        power_on_reset;
        dump_on;
        chk("D01_idle_after_reset", (tcon_fsm_state === 2'd1));
        chk("D02_no_column_before_video", (d0_col_valid === 1'b0) && (d1_col_valid === 1'b0));
        chk("D03_lut_not_committed_yet", (d0_lut_valid === 1'b0) && (d1_lut_valid === 1'b0));

        lut_program(0, 1);
        chk("D04_both_luts_committed", (d0_lut_valid === 1'b1) && (d1_lut_valid === 1'b1));
        chk("D05_bank_flipped_once", (d0_lut_bank === 1'b1) && (d1_lut_bank === 1'b1));
        chk("D06_commit_not_refused", (d0_lut_cmt_bad === 1'b0) && (d1_lut_cmt_bad === 1'b0));

        must_dark = 1'b1;
        ph_base  = dark_samples;
        wait_dclk(300);
        chk("D07_idle_channels_off", (dark_viol == 0) && ((dark_samples - ph_base) >= 300));
        must_dark = 1'b0;
        chk("D08_idle_never_unblanks", (idle_blanklow == 0));
        dump_off;

        // ==================================================================
        // Phase B — one full frame plus the wrap (1027 slots)
        // ==================================================================
        $display("--- Phase B: full frame 1027 slots ---");
        must_dark = 1'b1;                       // slots 0..1 are precharge
        nwin_base = disp_windows;
        dc_en = 1'b1; dc_first = 1'b0; dc_min = 1.0e9; dc_max = 0.0;
        dump_on;
        begin_frame(0, 0);
        run_slots(2);
        chk("D09_precharge_no_window", ((disp_windows - nwin_base) == 0) && (dark_viol == 0));
        must_dark = 1'b0;
        run_slots(1);
        chk("D10_fsm_active", (tcon_fsm_state === 2'd2));
        dc_en = 1'b0;
        dump_off;
        dump_on;
        run_slots(1024);
        dump_off;

        $display("PHB slots=%0d windows=%0d sel_bad=%0d tag_bad=%0d/%0d frm_bad=%0d byte_bad=%0d cont=%0d onehot_bad=%0d pwm_bad=%0d/%0d no_win=%0d",
                 slot_done, disp_windows, sel_bad, tag_bad0, tag_bad1, frm_bad,
                 byte_bad, cont_n, nohot_bad, pwm_bad, pwm_windows, no_win_slots);

        dc_en = 1'b0;
        chk("D11_no_rx_error_in_frame", (tcon_err_flags === 6'b000000));
        chk("D12_dclk_period_20ns",
            ((dc_min > 19.99) && (dc_min < 20.01)) &&
            ((dc_max > 19.99) && (dc_max < 20.01)));
        chk("D13_both_ports_every_window_384",
            (dp0_badlen == 0) && (dp1_badlen == 0) &&
            (dp0_bytes == 384) && (dp1_bytes == 384));
        chk("D14_window_count_equal", (dp0_win == dp1_win) && (dp0_win >= 1020));
        chk("D15_payload_bytes_exact", (dp0_bytebad == 0) && (dp1_bytebad == 0));
        chk("D16_no_x_on_data_bus", (dp0_x == 0) && (dp1_x == 0));
        chk("D17_display_windows_expected", (disp_windows >= 1020));
        chk("D18_scan_selection_matches", (sel_bad == 0));
        chk("D19_pooled_onehot_every_window", (nohot_bad == 0));
        chk("D20_data0_tag_matches_scan", (tag_bad0 == 0));
        chk("D21_data1_tag_matches_scan", (tag_bad1 == 0));
        chk("D22_frame_tag_matches", (frm_bad == 0));
        chk("D23_bank_content_768_bytes", (byte_bad == 0) && (cont_n >= 768 * 1020));
        chk("D24_pwm_width_matches_lut", (pwm_bad == 0) && (pwm_windows >= 1000));
        chk("D25_pwm_coverage_all_channels", (&pwm_cov0) && (&pwm_cov1));
        chk("D26_channel_on_in_every_window", (ch_on_bad == 0));
        chk("D27_display_implies_both_valid", (blank_novalid == 0));
        chk("D28_no_dark_window_in_healthy_frame", (no_win_slots == 0));
        chk("D29_strobe_count_equals_slots",
            (spm_stb >= 16'd1020) && (spm_oob == 0));
        chk("D30_scan_setup_hold_ok",
            (spm_setbad == 0) && (spm_holdbad == 0) &&
            (spm_setup_ps >= 32'd10000) && (spm_hold_ps >= 32'd5000));
        chk("D31_seen_columns_0_1_2_1022_1023_nextframe0",
            seen_c0 && seen_c1 && seen_c2 && seen_c1022 && seen_c1023 && seen_nf0);

        // ==================================================================
        // Phase C — LUT A -> B switch while running
        // ==================================================================
        $display("--- Phase C: LUT A->B switch ---");
        lut_program(1, 0);                      // table B into the INACTIVE bank
        @(posedge tcon_blank);                  // start of a BLANK interval
        repeat (10) @(negedge clk);             // 100 ns in, far ahead of cycle 828
        @(negedge tcon_dclk);
        cfg0_cmt = 1'b1; cfg1_cmt = 1'b1;
        @(posedge tcon_dclk);
        @(negedge tcon_dclk);
        cfg0_cmt = 1'b0; cfg1_cmt = 1'b0;
        // In steady state cmt_ok can only be true on the LATCH rising edge, so
        // the switch MUST land there - that is exactly the T3-H2 same-edge case.
        i = 0;
        while (i < 400) begin
            @(posedge tcon_dclk); #1;
            if (d0_lut_bank === 1'b0) i = 400; else i = i + 1;
        end
        switch_slot = slot_done;
        nwin_base = disp_windows; nobase = no_win_slots;
        $display("PHC switch_slot=%0d bank=%b/%b colv=%b/%b", switch_slot,
                 d0_lut_bank, d1_lut_bank, d0_col_valid, d1_col_valid);
        chk("D32_switch_flips_both_banks_on_latch_edge",
            (d0_lut_bank === 1'b0) && (d1_lut_bank === 1'b0));
        chk("D33_switch_invalidates_display",
            (d0_col_valid === 1'b0) && (d1_col_valid === 1'b0));
        cur_tab = 1;                            // from now on B is the expected image
        run_slots(4);
        chk("D34_one_dark_window_after_switch", (no_win_slots - nobase) == 1);
        chk("D35_display_resumes_with_new_table",
            ((disp_windows - nwin_base) == 2) && (byte_bad == 0));
        chk("D36_no_stale_column_after_switch", (sel_bad == 0) && (tag_bad0 == 0));

        // ==================================================================
        // Phase D — FAULT with a pending COMMIT (both chips watched)
        // ==================================================================
        $display("--- Phase D: fault + pending commit ---");
        while (!((vc >= 829) && (vc <= 831))) @(negedge clk);
        fault_n = 1'b0;                          // LATCH is still high: it drops now
                                                  // -> the Data window is left open
        repeat (4) @(posedge clk);
        chk("D37_fault_latched", (tcon_fsm_state === 2'd3));
        lut_program(0, 0);                       // table A into the now-inactive bank
        @(negedge tcon_dclk);
        cfg0_cmt = 1'b1; cfg1_cmt = 1'b1;
        @(posedge tcon_dclk);
        @(negedge tcon_dclk);
        cfg0_cmt = 1'b0; cfg1_cmt = 1'b0;
        wait_dclk(6);
        chk("D38_pending_commit_armed",
            (dut.u_data0.cmt_pend === 1'b1) && (dut.u_data1.cmt_pend === 1'b1));

        ph_base = 0; nobase = 0; cbase = 0;
        sbase = spm_stb;
        for (i = 0; i < 400; i = i + 1) begin
            @(posedge tcon_dclk); #1;
            if (d0_lut_bank !== 1'b0) ph_base = ph_base + 1;
            if (d1_lut_bank !== 1'b0) ph_base = ph_base + 1000;
            if (dut.u_data0.cmt_pend !== 1'b1) nobase = nobase + 1;
            if (dut.u_data1.cmt_pend !== 1'b1) nobase = nobase + 1000;
            if (((|d0_ch_en) !== 1'b0) || ((|d1_ch_en) !== 1'b0)) cbase = cbase + 1;
        end
        $display("PHD samples=400 bankflip=%0d pend_bad=%0d on_viol=%0d", ph_base, nobase, cbase);
        chk("D39_fault_blocks_bank_flip", (ph_base == 0));
        chk("D40_pending_commit_persists", (nobase == 0));
        chk("D41_fault_keeps_all_channels_off", (cbase == 0));

        fault_n = 1'b1;                          // release: TCON is still latched
        run_slots(10);
        chk("D42_latched_fault_blocks_commit",
            (d0_lut_bank === 1'b0) && (d1_lut_bank === 1'b0) &&
            (dut.u_data0.cmt_pend === 1'b1));
        chk("D43_latched_fault_stays_dark", (fault_blanklow == 0) && (tcon_fsm_state === 2'd3));
        chk("D44_no_strobe_no_latch_in_fault", (spm_stb == sbase));

        // ==================================================================
        // Phase E — explicit recovery: reset + reconfigure + legal frame
        // ==================================================================
        $display("--- Phase E: explicit recovery ---");
        power_on_reset;
        chk("D45_reset_clears_lut", (d0_lut_valid === 1'b0) && (d1_lut_valid === 1'b0));
        lut_program(0, 1);
        chk("D46_recovered_lut_committed",
            (d0_lut_valid === 1'b1) && (d1_lut_valid === 1'b1));
        cur_tab = 0;
        must_dark = 1'b1;
        nwin_base = disp_windows;
        begin_frame(0, 0);
        run_slots(2);
        chk("D47_precharge_no_early_window", ((disp_windows - nwin_base) == 0) && (dark_viol == 0));
        must_dark = 1'b0;
        run_slots(3);
        chk("D48_recovery_returns_to_active", (tcon_fsm_state === 2'd2));
        chk("D49_recovery_displays_column0", ((disp_windows - nwin_base) >= 1) && (sel_bad == 0));
        chk("D50_recovery_no_rx_error", (tcon_err_flags === 6'b000000));

        // ==================================================================
        // Phase F — TCON RX poisoning (underrun column)
        // ==================================================================
        $display("--- Phase F: RX poisoning ---");
        bad_col = (vcol + 3) % 1024;
        bad_nbytes = 767;
        run_slots(6);
        $display("PHF vcol=%0d bad_col=%0d err=%b fsm=%0d", vcol, bad_col, tcon_err_flags, tcon_fsm_state);
        chk("D51_underrun_error_bit", (tcon_err_flags[0] === 1'b1));
        chk("D52_poison_latches_fault", (tcon_fsm_state === 2'd3));
        chk("D53_poison_never_unblanks", (fault_blanklow == 0));
        ebase = spm_stb;
        fault_n = 1'b0; repeat (4) @(posedge clk); fault_n = 1'b1;
        run_slots(6);
        chk("D54_faultn_cycle_does_not_recover",
            (tcon_fsm_state === 2'd3) && (fault_blanklow == 0) && (spm_stb == ebase));

        // ==================================================================
        // Phase G — recovery is possible only through an explicit reset
        // ==================================================================
        $display("--- Phase G: second explicit recovery ---");
        bad_col = -1; bad_nbytes = 768;
        power_on_reset;
        lut_program(0, 1);
        cur_tab = 0;
        nwin_base = disp_windows;
        begin_frame(0, 0);
        run_slots(5);
        chk("D55_reset_recovers_after_poison",
            (tcon_fsm_state === 2'd2) && ((disp_windows - nwin_base) >= 2));
        chk("D56_final_content_still_exact", (byte_bad == 0) && (sel_bad == 0) && (nohot_bad == 0));

        // ==================================================================
        // census
        // ==================================================================
        $display("T5_COUNTS slots=%0d windows=%0d compared_bytes=%0d byte_bad=%0d sel_bad=%0d onehot_bad=%0d tag_bad=%0d/%0d frm_bad=%0d pwm_bad=%0d pwm_windows=%0d dark_samples=%0d dark_viol=%0d idle_blanklow=%0d fault_blanklow=%0d",
                 slot_done, disp_windows, cont_n, byte_bad, sel_bad, nohot_bad,
                 tag_bad0, tag_bad1, frm_bad, pwm_bad, pwm_windows,
                 dark_samples, dark_viol, idle_blanklow, fault_blanklow);
        $display("T5_ROWS load_windows=%0d/%0d badlen=%0d/%0d byte_bad=%0d/%0d strobes=%0d setup_min_ps=%0d hold_min_ps=%0d",
                 dp0_win, dp1_win, dp0_badlen, dp1_badlen, dp0_bytebad, dp1_bytebad,
                 spm_stb, spm_setup_ps, spm_hold_ps);
        if (fd_ledger != 0) $fclose(fd_ledger);
        $display("COURSE_RTL_ASSERT_COUNT=%0d", assert_count);
        $display("COURSE_RTL_PASS_COUNT=%0d",   pass_count);
        $display("COURSE_RTL_FAIL_COUNT=%0d",   fail_count);
        $display("COURSE_RTL_RESULT=%s", (fail_count == 0) ? "PASS" : "FAIL");
        $finish;
    end

    // ======================================================================
    // bounded run: never hang a single authorized ModelSim run
    // ======================================================================
    initial begin
        #30000000;                              // 30 ms absolute guard
        $display("COURSE_T5_WALL_GUARD=trip");
        $display("COURSE_RTL_ASSERT_COUNT=%0d", assert_count);
        $display("COURSE_RTL_FAIL_COUNT=1");
        $display("COURSE_RTL_RESULT=FAIL");
        $finish;
    end

endmodule
