`timescale 1ns / 1ps
// tcon_t4_tb — two-chip Scan Driver digital verification (T4).
//
// WHAT IS TESTED
//   rtl/scan_driver.v: two instances (CHIP_ID=0/1), each with a REAL 512-bit
//   one-hot output bus. Address latching on the true posedge scan_stb, chip
//   parity decode (col_id[0]), local index col_id[9:1], BLANK/FAULT/RESET
//   asynchronous hardware-priority shut-off, and the cross-chip interlock
//   across all 1024 global columns.
//
// WHAT IS NOT TESTED
//   512 high-side MOS devices, 8.68 ohm output resistance, discharge of
//   unselected columns, FAULT_n open-drain reporting, SPI, UVLO/OTP, PVT,
//   CDC/skew, real STA. A digital PASS says NOTHING about those.
//
// INDEPENDENCE OF THE CHECK
//   Three independent expectations are compared on every checked window:
//     1. rtl/scan_exp.v  — clean-room behavioural model fed the same pins
//     2. a per-address constant computed here from the loop variable:
//        (512'd1 << (addr/2)) on the chip selected by addr[0]
//     3. a global popcount rule: exactly ONE of the 1024 bits is high during
//        a display window, zero during BLANK/FAULT/RESET
//   None of them reads a DUT internal net. The DUT's diagnostic outputs are
//   themselves cross-checked against its own 512-bit pins (D39), not used as
//   the oracle.
//
// NON-VACUITY
//   expected_strobes / observed_strobes / checked_strobes / zero_or_multi_hot
//   / coverage_by_chip are printed and asserted, so a silently skipped sweep
//   cannot show up as PASS.
module tcon_t4_tb;

    // ---------------- reference clock: MEASUREMENT GRID ONLY --------------
    // This clock is NOT a Scan Driver pin and the DUTs must not depend on it.
    // It only gives the TB a 5 ns sampling grid (D38 proves the latch is not
    // clock-sampled).
    reg clk = 1'b0;
    always #5 clk = ~clk;

    // ---------------- pins ------------------------------------------------
    reg         reset_n  = 1'b0;
    reg         blank    = 1'b1;
    reg         fault_n  = 1'b1;
    reg         scan_stb = 1'b0;
    reg  [9:0]  col_id   = 10'd0;

    wire [511:0] c0, c1;          // DUT 512-bit outputs
    wire [511:0] e0, e1;          // independent pin-level oracle

    // diagnostics (observation only; never the oracle)
    wire        v0, v1;
    wire [9:0]  sc0, sc1;
    wire [8:0]  sn0, sn1;
    wire        er0, er1;

    scan_driver #(.CHIP_ID(0)) dut0 (
        .reset_n (reset_n), .blank (blank), .fault_n (fault_n),
        .scan_stb (scan_stb), .col_id (col_id), .col_a (c0),
        .sel_valid (v0), .sel_col (sc0), .sel_chan (sn0),
        .err_stb_outside_blank (er0)
    );

    scan_driver #(.CHIP_ID(1)) dut1 (
        .reset_n (reset_n), .blank (blank), .fault_n (fault_n),
        .scan_stb (scan_stb), .col_id (col_id), .col_a (c1),
        .sel_valid (v1), .sel_col (sc1), .sel_chan (sn1),
        .err_stb_outside_blank (er1)
    );

    scan_exp #(.CHIP_ID(0)) exp0 (
        .reset_n (reset_n), .blank (blank), .fault_n (fault_n),
        .scan_stb (scan_stb), .col_id (col_id), .exp_out (e0)
    );

    scan_exp #(.CHIP_ID(1)) exp1 (
        .reset_n (reset_n), .blank (blank), .fault_n (fault_n),
        .scan_stb (scan_stb), .col_id (col_id), .exp_out (e1)
    );

    // ---------------- accepted pin-level monitor (reused unmodified) ------
    wire [15:0] pm_stb, pm_outside, pm_setup_bad, pm_hold_bad;
    wire [9:0]  pm_last_col;
    wire        pm_last_blank;
    wire [31:0] pm_setup_min_ps, pm_hold_min_ps;

    scan_pin_mon pm (
        .reset_n (reset_n), .scan_stb (scan_stb), .col_id (col_id),
        .blank (blank),
        .n_stb (pm_stb), .n_outside_blank (pm_outside),
        .n_setup_bad (pm_setup_bad), .n_hold_bad (pm_hold_bad),
        .last_col (pm_last_col), .last_blank (pm_last_blank),
        .setup_min_ps (pm_setup_min_ps), .hold_min_ps (pm_hold_min_ps)
    );

    // ---------------- helpers --------------------------------------------
    function integer popc(input [1023:0] v);
        integer i, s;
        begin
            s = 0;
            for (i = 0; i < 1024; i = i + 1) if (v[i]) s = s + 1;
            popc = s;
        end
    endfunction

    integer assert_count = 0, pass_count = 0, fail_count = 0;

    task automatic chk(input string nm, input logic cond);
        begin
            assert_count = assert_count + 1;
            if (cond) begin
                pass_count = pass_count + 1;
                $display("PASS %s", nm);
            end else begin
                fail_count = fail_count + 1;
                $display("FAIL %s  (t=%0t blank=%b fault_n=%b col_id=%0d c0=%h c1=%h)",
                         nm, $time, blank, fault_n, col_id, c0[63:0], c1[63:0]);
            end
        end
    endtask

    // ---------------- stimulus bookkeeping (call sites) -------------------
    integer n_planned_legal   = 0;
    integer n_planned_illegal = 0;
    integer n_addr_checked    = 0;
    integer cov0 = 0, cov1 = 0;

    task automatic pulse_legal;              // blank MUST be 1 here
        begin
            scan_stb = 1'b1; #25;
            scan_stb = 1'b0; #10;
            n_planned_legal = n_planned_legal + 1;
        end
    endtask

    task automatic strobe_addr(input [9:0] a);
        begin
            blank = 1'b1; #5;
            col_id = a; #25;
            pulse_legal;
        end
    endtask

    // ---------------- pin-side monitors ----------------------------------
    reg  [1023:0] gs;
    reg           was_on = 1'b0;
    integer n_cmp = 0, n_cmp_bad = 0, n_blank_on = 0, n_fault_on = 0;
    integer n_onehot_samples = 0, n_diag_sel_bad = 0, n_episodes = 0;

    // Sampled 0.1 ns AFTER the 5 ns grid so that a stimulus event happening at
    // the same simulation time is already fully settled (all delta cycles of
    // that time step are done). Without this offset a pin change that lands
    // exactly on a negedge would be sampled mid-delta and could report a
    // phantom "output high while BLANK" violation.
    always @(negedge clk) begin
        #0.1;
        if (reset_n) begin
            gs = {c1, c0};
            n_cmp = n_cmp + 1;
            if ({c1, c0} !== {e1, e0}) n_cmp_bad = n_cmp_bad + 1;
            if (blank    && (|gs)) n_blank_on = n_blank_on + 1;
            if (!fault_n && (|gs)) n_fault_on = n_fault_on + 1;
            if (popc(gs) == 1) n_onehot_samples = n_onehot_samples + 1;
            if (!was_on && (|gs)) n_episodes = n_episodes + 1;
            was_on <= (|gs);
            if (v0 !== (|c0)) n_diag_sel_bad = n_diag_sel_bad + 1;
            if (v1 !== (|c1)) n_diag_sel_bad = n_diag_sel_bad + 1;
        end else begin
            was_on <= 1'b0;
        end
    end

    // ---------------- sweep counters -------------------------------------
    integer sw_mismatch = 0, sw_nonone = 0, sw_nontarget = 0;
    integer sw_idx_bad = 0, sw_blank_kill_bad = 0;
    integer b_nonone = 0;
    integer zero_or_multi_hot = 0;

    integer addr, half;
    integer ep_before;

    // ---------------- VCD: windowed, never a full-frame dump -------------
    initial begin
        $dumpfile("t4.vcd");
        $dumpvars(0, tcon_t4_tb);
        $dumpoff;
    end

    initial begin
        $display("COURSE_TCON_T4_SCENARIO=TWO_SCAN_DRIVERS_512WAY");
        $display("COURSE_TCON_T4_CONTRACT=V13_SCAN_DIGITAL");
        #1;

        // ================= P0: reset =====================================
        $display("--- P0 reset ---");
        reset_n = 1'b0; blank = 1'b1; fault_n = 1'b1;
        #200;
        chk("D01_reset_chip0_all_off", c0 === 512'd0);
        chk("D02_reset_chip1_all_off", c1 === 512'd0);
        chk("D03_reset_diags_cleared",
            (v0 === 1'b0) && (v1 === 1'b0) && (er0 === 1'b0) && (er1 === 1'b0));
        reset_n = 1'b1; #50;
        blank = 1'b0; #30;
        chk("D04_reset_release_no_strobe_all_off", (|{c1, c0}) === 1'b0);
        blank = 1'b1; #5;

        // ================= P1: boundary global columns ===================
        $display("--- P1 boundary columns 0/1/2/3/1022/1023 ---");
        strobe_addr(10'd0); blank = 1'b0; #30;
        chk("D05_global_col0_chip0_chan0", (c0 === (512'd1 << 9'd0)) && (c1 === 512'd0));
        if (popc({c1, c0}) != 1) b_nonone = b_nonone + 1;
        blank = 1'b1; #5;

        strobe_addr(10'd1); blank = 1'b0; #30;
        chk("D06_global_col1_chip1_chan0", (c1 === (512'd1 << 9'd0)) && (c0 === 512'd0));
        if (popc({c1, c0}) != 1) b_nonone = b_nonone + 1;
        blank = 1'b1; #5;

        strobe_addr(10'd2); blank = 1'b0; #30;
        chk("D07_global_col2_chip0_chan1", (c0 === (512'd1 << 9'd1)) && (c1 === 512'd0));
        if (popc({c1, c0}) != 1) b_nonone = b_nonone + 1;
        blank = 1'b1; #5;

        strobe_addr(10'd3); blank = 1'b0; #30;
        chk("D08_global_col3_chip1_chan1", (c1 === (512'd1 << 9'd1)) && (c0 === 512'd0));
        if (popc({c1, c0}) != 1) b_nonone = b_nonone + 1;
        chk("D11_boundary_col2_col3_no_residue",
            (c0 === 512'd0) && (popc({c1, c0}) == 1));
        blank = 1'b1; #5;

        strobe_addr(10'd1022); blank = 1'b0; #30;
        chk("D09_global_col1022_chip0_chan511", (c0 === (512'd1 << 9'd511)) && (c1 === 512'd0));
        if (popc({c1, c0}) != 1) b_nonone = b_nonone + 1;
        blank = 1'b1; #5;

        strobe_addr(10'd1023); blank = 1'b0; #30;
        chk("D10_global_col1023_chip1_chan511", (c1 === (512'd1 << 9'd511)) && (c0 === 512'd0));
        if (popc({c1, c0}) != 1) b_nonone = b_nonone + 1;
        blank = 1'b1; #5;

        // ================= P2: all 1024 global columns ===================
        $display("--- P2 sweep 0..1023, every address compared bit-exactly ---");
        for (addr = 0; addr < 1024; addr = addr + 1) begin
            if (addr == 0)   $dumpon;
            if (addr == 4)   $dumpoff;

            blank = 1'b1; #5;
            col_id = addr[9:0]; #25;
            pulse_legal;

            blank = 1'b0; #30;

            half = addr / 2;
            // expectation #2: constant computed here, no model involved
            if (addr[0] == 1'b0) begin
                if (c0 !== (512'd1 << half)) sw_mismatch = sw_mismatch + 1;
                if (c1 !== 512'd0)           sw_nontarget = sw_nontarget + 1;
                if (c0[half] !== 1'b1)       sw_idx_bad = sw_idx_bad + 1;
                cov0 = cov0 + 1;
            end else begin
                if (c1 !== (512'd1 << half)) sw_mismatch = sw_mismatch + 1;
                if (c0 !== 512'd0)           sw_nontarget = sw_nontarget + 1;
                if (c1[half] !== 1'b1)       sw_idx_bad = sw_idx_bad + 1;
                cov1 = cov1 + 1;
            end
            if (popc({c1, c0}) != 1) sw_nonone = sw_nonone + 1;
            n_addr_checked = n_addr_checked + 1;

            // BLANK must kill the array in the same delta
            blank = 1'b1; #1;
            if ((|{c1, c0}) !== 1'b0) sw_blank_kill_bad = sw_blank_kill_bad + 1;
        end

        chk("D12_sweep_1024_addresses_checked", n_addr_checked == 1024);
        chk("D13_sweep_expectation_zero_mismatch", sw_mismatch == 0);
        chk("D14_sweep_local_index_exact",        sw_idx_bad == 0);
        chk("D15_sweep_per_chip_512_512",         (cov0 == 512) && (cov1 == 512));
        chk("D16_sweep_global_always_onehot",     sw_nonone == 0);
        chk("D17_sweep_non_target_always_zero",   sw_nontarget == 0);
        chk("D18_sweep_blank_kill_immediate",     sw_blank_kill_bad == 0);

        // ================= P3: latch semantics ===========================
        $display("--- P3 latch stability / no strobe, no change ---");
        strobe_addr(10'd2); blank = 1'b0; #30;
        ep_before = n_episodes;

        col_id = 10'd40; #30;
        chk("D20_col_change_without_strobe_keeps_column",
            (c0 === (512'd1 << 9'd1)) && (c1 === 512'd0));
        chk("D21_no_extra_episode_on_col_change", n_episodes == ep_before);

        strobe_addr(10'd40); blank = 1'b0; #30;
        chk("D22_strobe_latches_new_column",
            (c0 === (512'd1 << 9'd20)) && (c1 === 512'd0));

        col_id = 10'd100; #20; col_id = 10'd200; #20; col_id = 10'd300; #20;
        chk("D23_latch_holds_across_many_col_changes",
            (c0 === (512'd1 << 9'd20)) && (c1 === 512'd0));

        strobe_addr(10'd3); blank = 1'b0; #30;
        chk("D24_other_chip_strobe_clears_this_chip",
            (c1 === (512'd1 << 9'd1)) && (c0 === 512'd0));

        // ================= P4: BLANK / FAULT asynchronous gate ===========
        $display("--- P4 BLANK / FAULT immediate shut-off ---");
        $dumpon;
        blank = 1'b1; #1;
        chk("D25_blank_high_off_within_1ns", (|{c1, c0}) === 1'b0);
        #50;
        chk("D26_blank_never_on", n_blank_on == 0);

        blank = 1'b0; #30;
        chk("D27_blank_release_restores_column",
            (c1 === (512'd1 << 9'd1)) && (c0 === 512'd0));

        fault_n = 1'b0; #1;
        chk("D28_fault_low_off_within_1ns", (|{c1, c0}) === 1'b0);
        #50;
        chk("D29_fault_never_on", n_fault_on == 0);

        fault_n = 1'b1; #1;
        chk("D30_fault_release_restores_column",
            (c1 === (512'd1 << 9'd1)) && (c0 === 512'd0));

        // ================= P5: illegal strobe outside BLANK ==============
        $display("--- P5 strobe outside BLANK (frozen safe policy) ---");
        col_id = 10'd8; #25;
        scan_stb = 1'b1; #1;
        chk("D32_stb_outside_blank_global_off", (|{c1, c0}) === 1'b0);
        #24; scan_stb = 1'b0; #10;
        n_planned_illegal = n_planned_illegal + 1;
        chk("D31_stb_outside_blank_flags_both", (er0 === 1'b1) && (er1 === 1'b1));

        strobe_addr(10'd16); blank = 1'b0; #30;
        chk("D33_legal_strobe_recovers_after_violation",
            (c0 === (512'd1 << 9'd8)) && (c1 === 512'd0));
        chk("D34_err_flag_sticky", (er0 === 1'b1) && (er1 === 1'b1));
        $dumpoff;

        // ================= P6: pin timing / diag / counters ==============
        $display("--- P6 pin timing, diagnostics, census ---");
        // RUN02 (review-mandated): print the REAL planned/observed pair at the
        // exact instant D19 is evaluated. D19 runs BEFORE the extra D38 strobe,
        // so the planned count here is 1035, not the 1036 seen at P7.
        $display("COURSE_T4_STROBES_AT_D19 expected=%0d observed=%0d",
                 n_planned_legal + n_planned_illegal, pm_stb);
        chk("D19_strobe_count_expected_eq_observed",
            pm_stb == (n_planned_legal + n_planned_illegal));
        chk("D35_setup_min_ge_10ns",  pm_setup_min_ps >= 32'd10000);
        chk("D36_hold_min_ge_5ns",    pm_hold_min_ps  >= 32'd5000);
        chk("D37_exactly_one_deliberate_violation", pm_outside == 16'd1);

        // the latch must NOT be sampled by the 100 MHz reference clock:
        // assert the strobe 6 ns after a reference posedge, then demand the
        // latched address 2 ns later - a clock-sampled design is still stale.
        blank = 1'b1; #5;
        col_id = 10'd30; #25;
        @(posedge clk); #6;
        scan_stb = 1'b1; #2;
        chk("D38_response_not_clock_sampled", (sc0 === 10'd30) && (sn0 === 9'd15));
        #23; scan_stb = 1'b0; #10;
        n_planned_legal = n_planned_legal + 1;
        blank = 1'b0; #30;
        chk("D39_diag_sel_valid_matches_output",
            (n_diag_sel_bad == 0) && (v0 === 1'b1) && (v1 === 1'b0));

        // ================= P7: asynchronous reset ========================
        // RUN02 (review-mandated): snapshot the census and the pin-monitor
        // minima BEFORE the reset, because scan_pin_mon clears n_stb and puts
        // setup/hold minima back to the 1e9 sentinel on negedge reset_n. The
        // post-reset prints at the very end are kept only for auditability and
        // must never be cited as the P6 measurement.
        zero_or_multi_hot = sw_nonone + b_nonone;
        $display("COURSE_T4_STROBES_PRE_RESET expected=%0d observed=%0d checked=%0d zero_or_multi_hot=%0d cov_chip0=%0d cov_chip1=%0d",
                 n_planned_legal + n_planned_illegal, pm_stb, n_addr_checked,
                 zero_or_multi_hot, cov0, cov1);
        $display("COURSE_T4_TIMING_PRE_RESET setup_min_ps=%0d hold_min_ps=%0d setup_bad=%0d hold_bad=%0d",
                 pm_setup_min_ps, pm_hold_min_ps, pm_setup_bad, pm_hold_bad);

        $display("--- P7 async reset mid-display ---");
        reset_n = 1'b0; #1;
        chk("D40_async_reset_off_within_1ns", (|{c1, c0}) === 1'b0);
        reset_n = 1'b1; #30;
        chk("D41_after_reset_needs_new_strobe", (|{c1, c0}) === 1'b0);
        chk("D42_monitor_samples_not_vacuous", n_cmp >= 9500);

        // ================= census =======================================
        // POST-RESET snapshot: n_stb / setup_min / hold_min are already cleared
        // by P7, so these numbers are sentinels by construction. Labelled
        // POST_RESET on purpose; the authoritative values are the *_PRE_RESET
        // lines above.
        $display("COURSE_T4_STROBES_POST_RESET expected=%0d observed=%0d checked=%0d zero_or_multi_hot=%0d cov_chip0=%0d cov_chip1=%0d",
                 n_planned_legal + n_planned_illegal, pm_stb, n_addr_checked,
                 zero_or_multi_hot, cov0, cov1);
        $display("COURSE_T4_MONITOR samples=%0d oracle_mismatch=%0d onehot_samples=%0d diag_sel_bad=%0d blank_on=%0d fault_on=%0d",
                 n_cmp, n_cmp_bad, n_onehot_samples, n_diag_sel_bad, n_blank_on, n_fault_on);
        $display("COURSE_T4_TIMING_POST_RESET setup_min_ps=%0d hold_min_ps=%0d setup_bad=%0d hold_bad=%0d (cleared by P7 reset - do not cite)",
                 pm_setup_min_ps, pm_hold_min_ps, pm_setup_bad, pm_hold_bad);

        $display("COURSE_RTL_ASSERT_COUNT=%0d", assert_count);
        $display("COURSE_RTL_PASS_COUNT=%0d",   pass_count);
        $display("COURSE_RTL_FAIL_COUNT=%0d",   fail_count);
        if (fail_count == 0) begin
            $display("COURSE_RTL_RESULT=PASS");
            $display("COURSE_TCON_T4_SCENARIO_DONE=TWO_SCAN_DRIVERS_512WAY");
        end else begin
            $display("COURSE_RTL_RESULT=FAIL");
            $fatal(1, "COURSE_ASSERT_FAILED: %0d of %0d checks", fail_count, assert_count);
        end
        $finish;
    end

endmodule
