`timescale 1ns / 1ps
// TCON T6-A **L1** testbench — the rev2 SPI/register suite (57 assertions,
// unchanged) PLUS the L1 safety phase E01..E20 (P0-D same-edge clear).
//
// PREPARED ONLY. This round is zero-ModelSim: the file is compiled with vlog
// and never run. Elaboration, the 57-assertion regression and E01..E20 belong
// to the L1 budget, which has NOT been granted.
//
// The TB drives the four real pins only (SPI_SCK, SPI_MOSI, SPI_CS_n, SPI_MISO)
// and observes the register outputs. It never reaches into the CDC internals to
// "make" a test pass.
//
// rev2 changes forced by the T6-A RUN01 review (2026-11-19):
//   * a LEGAL CS_n high gap (cs_gap, >= the documented 100 ns) between
//     transactions; rev1 produced a zero-length CS pulse that no synchroniser
//     can see;
//   * the write/read counters count what the DUT really did (captured write
//     frames / latched read snapshots), not what the TB asked for;
//   * new assertions: read first byte, read-is-not-write, 47/49-edge rejection,
//     MISO high-Z, snapshot-not-torn, CDC loss/duplicate, EN=0 safety,
//     fault-blocks-unblank, INT_EN does not mask safety.
module tcon_t6a_l1_tb;

    // ---- 100 MHz core clock ---------------------------------------------
    reg clk = 1'b0;
    reg clk_en = 1'b1;              // Phase S only; 1 everywhere else
    always #5 if (clk_en) clk = ~clk;

    // ---- SPI pins --------------------------------------------------------
    reg  spi_sck  = 1'b0;
    reg  spi_mosi = 1'b0;
    reg  spi_cs_n = 1'b1;
    wire spi_miso;

    // ---- chip-side inputs -------------------------------------------------
    reg        reset_n = 1'b0;
    reg  [3:0] status_in = 4'b0100;      // {LUT_VALID,SAFE,ACTIVE,READY}
    reg  [10:0] hw_fault = 11'd0;
    // V1.3 §12: FAULT_n is an open-drain wired-AND ACTIVE-LOW pin with an
    // external pull-up. It is driven here EXPLICITLY: an unconnected input
    // floats to Z and `fault_n ? 0 : 11'h080` would read that as "fault".
    reg         fault_n  = 1'b1;
    reg        test_en = 1'b0;
    reg  [9:0] col_index_in = 10'd0;

    // ---- observability ----------------------------------------------------
    wire        en, blank_r, hw_blank, test_mode, sw_reset_pulse, fps_sel;
    wire [15:0] wdog;
    wire [10:0] fault_status, int_status, int_en;
    wire        irq, err_reg;
    wire [5:0]  bitcnt_o;
    wire        frame_full_o;
    wire [47:0] sh_o;
    wire [31:0] rd_snapshot_o;
    wire [15:0] wr_frames_o, wr_cap_cnt_o, wr_rej_cnt_o;
    wire        mb_lost_o, mb_badlen_o;
    wire        wr_cap_o, wr_ok_o, rd_cap_o;
    wire        disp_armed_o, hw_fault_caught_o;

    tcon_spi_top dut (
        .clk (clk), .reset_n (reset_n),
        .spi_sck (spi_sck), .spi_mosi (spi_mosi), .spi_miso (spi_miso),
        .spi_cs_n (spi_cs_n),
        .status_in (status_in), .hw_fault (hw_fault), .fault_n (fault_n),
        .test_en (test_en),
        .col_index_in (col_index_in),
        .en (en), .blank_r (blank_r), .hw_blank (hw_blank),
        .test_mode (test_mode),
        .sw_reset_pulse (sw_reset_pulse), .fps_sel (fps_sel), .wdog (wdog),
        .fault_status (fault_status), .int_status (int_status), .int_en (int_en),
        .irq (irq), .err_reg (err_reg),
        .bitcnt_o (bitcnt_o), .frame_full_o (frame_full_o),
        .sh_o (sh_o), .rd_snapshot_o (rd_snapshot_o),
        .wr_frames_o (wr_frames_o), .mb_lost_o (mb_lost_o),
        .mb_badlen_o (mb_badlen_o), .wr_cap_cnt_o (wr_cap_cnt_o),
        .wr_rej_cnt_o (wr_rej_cnt_o), .wr_cap_o (wr_cap_o),
        .wr_ok_o (wr_ok_o), .rd_cap_o (rd_cap_o),
        .disp_armed_o (disp_armed_o), .hw_fault_caught_o (hw_fault_caught_o)
    );

    // ---------------------------------------------------------------------
    // assertion plumbing
    // ---------------------------------------------------------------------
    integer assert_count = 0, pass_count = 0, fail_count = 0;

    task automatic chk(input string nm, input logic cond);
        begin
            assert_count = assert_count + 1;
            if (cond) begin
                pass_count = pass_count + 1;
                $display("PASS %s", nm);
            end else begin
                fail_count = fail_count + 1;
                $display("FAIL %s  (t=%0t)", nm, $time);
                $display("COURSE_FAILED_ASSERTION=%s", nm);
            end
        end
    endtask

    // ---------------------------------------------------------------------
    // SPI driver: mode 0, MSB first, 16-bit command + 32-bit data
    // ---------------------------------------------------------------------
    real    t_half;                     // half SCLK period (ns)
    real    cs_gap;                     // CS_n high time between transactions
    integer n_wr_ok, n_rd_ok, n_commit, n_sw_pulse, n_frames, n_cap;
    reg     ff_d = 1'b0;
    reg [7:0]  rd_first_byte;
    reg        rd_first_bit;
    reg        rd_first_def;
    reg        last_sample = 1'b0;
    integer    rd_bit_index;

    // real counts: what the DUT actually did
    always @(posedge clk) if (reset_n && wr_ok_o)   n_wr_ok = n_wr_ok + 1;
    always @(posedge clk) if (reset_n && rd_cap_o)  n_rd_ok = n_rd_ok + 1;
    always @(posedge clk) if (reset_n && wr_cap_o)  n_cap    = n_cap + 1;
    always @(posedge clk) if (reset_n && sw_reset_pulse) n_sw_pulse = n_sw_pulse + 1;
    // one count per transaction that really reached 48 bits
    always @(posedge clk) begin
        ff_d <= frame_full_o;
        if (reset_n && frame_full_o && !ff_d) n_frames = n_frames + 1;
    end

    task automatic spi_begin;
        begin
            rd_first_bit = 1'b0;
            rd_first_byte = 8'h00;
            rd_first_def = 1'b0;
            rd_bit_index = 0;
            spi_cs_n = 1'b0;
            #(t_half);
        end
    endtask

    // one SCLK period; MOSI is set while SCK is low (mode 0) and MISO is
    // sampled T/4 after the rising edge - before the slave updates it again.
    task automatic spi_bit(input b, input sample);
        begin
            spi_mosi = b;
            #(t_half); spi_sck = 1'b1;
            if (sample) begin
                #(t_half * 0.25);
                last_sample = spi_miso;
                if (rd_bit_index == 0) begin
                    rd_first_bit = spi_miso;          // four-state on purpose
                    rd_first_def = (spi_miso === 1'b0) || (spi_miso === 1'b1);
                end
                if (rd_bit_index < 8)
                    rd_first_byte = {rd_first_byte[6:0], (spi_miso === 1'b1)};
                rd_bit_index = rd_bit_index + 1;
                #(t_half * 0.75);
            end else begin
                #(t_half);
            end
            spi_sck = 1'b0;
        end
    endtask

    task automatic spi_end;
        begin
            #(t_half);
            spi_cs_n = 1'b1;
            #(cs_gap);                                 // legal inter-transaction gap
        end
    endtask

    task automatic spi_xfer(input [15:0] cmd, input [31:0] wd,
                            output [31:0] rdout, input integer nbits_total);
        integer i, nb;
        begin
            rdout = 32'd0;
            nb = 0;
            spi_begin;
            for (i = 15; i >= 0; i = i - 1) begin
                if (nb < nbits_total) begin
                    spi_bit(cmd[i], 1'b0);
                    nb = nb + 1;
                end
            end
            for (i = 31; i >= 0; i = i - 1) begin
                if (nb < nbits_total) begin
                    spi_bit(wd[i], 1'b1);
                    if (last_sample === 1'b1) rdout[i] = 1'b1;
                    else                      rdout[i] = 1'b0;
                    nb = nb + 1;
                end
            end
            spi_end;
            n_commit = n_commit + 1;
        end
    endtask

    // arbitrary length (for the 47 / 49 edge cases)
    task automatic spi_len(input [15:0] cmd, input [32:0] dta, input integer n);
        integer k;
        begin
            spi_begin;
            for (k = 0; k < n; k = k + 1) begin
                if (k < 16) spi_bit(cmd[15 - k], (k >= 16));
                else        spi_bit(dta[32 - (k - 16)], (k >= 16));
            end
            spi_end;
        end
    endtask

    task automatic wr(input [14:0] a, input [31:0] d);
        reg [31:0] dummy;
        begin
            spi_xfer({1'b0, a}, d, dummy, 48);
        end
    endtask

    task automatic rd(input [14:0] a, output [31:0] d);
        begin
            spi_xfer({1'b1, a}, 32'd0, d, 48);
        end
    endtask

    // ---------------------------------------------------------------------
    // scenario
    // ---------------------------------------------------------------------
    reg [31:0] v, v2, v3;
    reg        v_sel;
    integer i, err_base, nf_base, cap_base;
    real t_phase;


    // =====================================================================
    // Phase S — L1 safety (P0-D)
    //   Every pulse below is 4 ns wide and starts 3 ns after a core rising
    //   edge, so it both starts and ends strictly between two edges: no
    //   posedge clk can ever sample it. Only the asynchronous set catches it
    //   (P0-B), and the clear must remove status[7] and the capture on the
    //   SAME edge (P0-D) - the old registered clr_fault left exactly one
    //   edge with status[7]=0 but caught=1, which `split_cnt` counts.
    // =====================================================================
    reg        mon_en = 1'b0;
    reg        st7_d  = 1'b0;
    integer    split_cnt = 0, resurrect_cnt = 0, pulse_edges = 0, perm_samples = 0, k;

    always @(posedge clk) begin
        if (!fault_n) pulse_edges = pulse_edges + 1;      // edges INSIDE the pulse
    end

    always @(posedge clk) #1 begin                        // post-edge values
        if (reset_n && mon_en) begin
            if ((fault_status[7] === 1'b0) && (hw_fault_caught_o === 1'b1))
                split_cnt = split_cnt + 1;                // the resurrection window
            if ((st7_d === 1'b0) && (fault_status[7] === 1'b1))
                resurrect_cnt = resurrect_cnt + 1;        // 0 -> 1 with no pulse
            st7_d = fault_status[7];
        end
    end

    task automatic fault_pulse_4ns;
        begin
            pulse_edges = 0;
            @(posedge clk);
            #3; fault_n = 1'b0;                            // 3 ns after the edge
            #4; fault_n = 1'b1;                            // gone before the next
        end
    endtask

    initial begin
        $display("COURSE_T6A_SCENARIO=TCON_SPI_MODE0_48BIT_REGFILE");
        $display("COURSE_T6A_CONTRACT=V13_SPI_REGISTER_SUBSET");

        assert_count = 0; pass_count = 0; fail_count = 0;
        n_wr_ok = 0; n_rd_ok = 0; n_commit = 0; n_sw_pulse = 0; n_frames = 0; n_cap = 0;
        t_half = 50.0;                                    // 10 MHz
        cs_gap = 200.0;                                   // >= 100 ns documented

        reset_n = 1'b0; spi_cs_n = 1'b1; spi_sck = 1'b0; spi_mosi = 1'b0;
        status_in = 4'b0100; hw_fault = 11'd0; test_en = 1'b0; col_index_in = 10'd0;
        repeat (20) @(posedge clk);
        reset_n = 1'b1;
        repeat (10) @(posedge clk);

        // ================= A. reset values ================================
        $display("--- A. reset values ---");
        rd(15'h00, v);  chk("D01_chip_id_0x10", v == 32'h0000_0010);
        rd(15'h04, v);  chk("D02_ctrl_reset_blank", v == 32'h0000_0002);
        rd(15'h18, v);  chk("D03_status_live_4", v == 32'h0000_0004);
        rd(15'h08, v);  chk("D04_mode_reset_0", v == 32'h0000_0000);
        rd(15'h1C, v);  chk("D05_watchdog_reset_2", v == 32'h0000_0002);
        rd(15'h20, v);  chk("D06_video_cfg_1023_767", v == 32'h000B_FFFF);
        col_index_in = 10'd777;
        rd(15'h24, v);  chk("D07_col_index_live", v == 32'd777);
        col_index_in = 10'd0;

        // ================= B. legal writes and readback ====================
        $display("--- B. write / readback ---");
        wr(15'h08, 32'h0000_0001); rd(15'h08, v);
        chk("D08_mode_write_1", v == 32'h0000_0001);
        wr(15'h1C, 32'h0000_1234); rd(15'h1C, v);
        chk("D09_watchdog_write", v == 32'h0000_1234);
        wr(15'h14, 32'h0000_05A5); rd(15'h14, v);
        chk("D10_int_en_write", v == 32'h0000_05A5);
        wr(15'h04, 32'h0000_0003); rd(15'h04, v);         // EN=1, BLANK=1
        chk("D11_en_set_when_safe", (v == 32'h0000_0003) && (en === 1'b1));
        wr(15'h14, 32'h0000_0000);                        // clean up the mask

        err_base = err_reg;
        wr(15'h08, 32'h0000_0000); rd(15'h08, v);         // MODE while EN=1
        chk("D12_config_while_en_rejected",
            (v == 32'h0000_0001) && (err_reg === 1'b1) && (fault_status[8] === 1'b1));
        wr(15'h0C, 32'h0000_0100); rd(15'h0C, v);         // W1C ERR_REG
        chk("D13_err_reg_w1c", (v[8] === 1'b0) && (err_reg === 1'b0));

        wr(15'h04, 32'h0000_0001); rd(15'h04, v);         // EN=1, BLANK=0 while ready=0
        chk("D14_unblank_needs_ready", (err_reg === 1'b1) && (blank_r === 1'b1));
        wr(15'h0C, 32'h0000_0100);
        status_in[0] = 1'b1;                              // READY
        wr(15'h04, 32'h0000_0001); rd(15'h04, v);         // EN=1, BLANK=0
        chk("D15_unblank_with_ready", (v == 32'h0000_0001) && (blank_r === 1'b0));

        nf_base = n_sw_pulse;
        wr(15'h04, 32'h0000_0006);                        // EN=0 BLANK=1 SW_RESET=1
        rd(15'h04, v);
        chk("D16_sw_reset_pulse_self_clears",
            (n_sw_pulse - nf_base == 1) && (v[2] === 1'b0) && (v == 32'h0000_0002));

        wr(15'h04, 32'h0000_000A); rd(15'h04, v);         // TEST without TEST_EN
        chk("D17_test_needs_test_en", (err_reg === 1'b1) && (v[3] === 1'b0));
        wr(15'h0C, 32'h0000_0100);
        test_en = 1'b1;
        wr(15'h04, 32'h0000_000A); rd(15'h04, v);
        chk("D18_test_with_test_en", (v[3] === 1'b1) && (test_mode === 1'b1));

        // ================= C. illegal / edge cases =========================
        $display("--- C. illegal accesses and framing ---");
        wr(15'h06, 32'hDEAD_BEEF); rd(15'h06, v);
        chk("D19_misaligned_rejected", (err_reg === 1'b1) && (v == 32'd0));
        wr(15'h0C, 32'h0000_0100);
        wr(15'h60, 32'hCAFE_F00D); rd(15'h60, v);
        chk("D20_unimplemented_rejected", (err_reg === 1'b1) && (v == 32'd0));
        wr(15'h0C, 32'h0000_0100);
        wr(15'h00, 32'hFFFF_FFFF); rd(15'h00, v);
        chk("D21_ro_write_rejected", (err_reg === 1'b1) && (v == 32'h0000_0010));
        wr(15'h0C, 32'h0000_0100);
        wr(15'h08, 32'hFFFF_FFFE); rd(15'h08, v);         // only bit0 may latch
        chk("D22_reserved_bits_write_0", (v == 32'h0000_0000));

        // incomplete frame: only 30 SCK then CS_n rises -> discarded
        err_base = err_reg;
        spi_xfer({1'b0, 15'h1C}, 32'h0000_BEEF, v, 30);
        rd(15'h1C, v2);
        chk("D23_short_frame_discarded",
            (v2 == 32'h0000_1234) && (err_reg === err_base));

        // exactly 20 SCK (command only) -> no write, no error
        spi_xfer({1'b0, 15'h1C}, 32'h0000_9999, v, 20);
        rd(15'h1C, v2);
        chk("D24_command_only_no_commit", (v2 == 32'h0000_1234) && (err_reg === err_base));

        // back-to-back: two writes separated by a LEGAL (>=100 ns) CS_n high gap
        spi_begin;
        v3 = {1'b0, 15'h1C};
        for (i = 15; i >= 0; i = i - 1) spi_bit(v3[i], 1'b0);
        v3 = 32'h0000_0777;
        for (i = 31; i >= 0; i = i - 1) spi_bit(v3[i], 1'b0);
        spi_end;                                          // CS high for cs_gap
        spi_xfer({1'b0, 15'h08}, 32'h0000_0001, v, 48);   // second transaction
        rd(15'h1C, v2); rd(15'h08, v3);
        chk("D25_back_to_back_transactions",
            (v2 == 32'h0000_0777) && (v3 == 32'h0000_0001));

        // ================= D. fault latching and W1C =======================
        $display("--- D. fault / interrupt ---");
        hw_fault[0] = 1'b1;                               // UVLO
        repeat (4) @(posedge clk);
        rd(15'h0C, v);
        chk("D26_fault_latched", (v[0] === 1'b1));
        wr(15'h0C, 32'h0000_0001); rd(15'h0C, v);         // W1C while the fault persists
        chk("D27_live_fault_cannot_clear", (v[0] === 1'b1));
        hw_fault[0] = 1'b0;
        repeat (4) @(posedge clk);
        wr(15'h0C, 32'h0000_0001); rd(15'h0C, v);
        chk("D28_w1c_after_fault_gone", (v[0] === 1'b0));

        rd(15'h10, v);
        chk("D29_int_status_latched", (v[0] === 1'b1));
        wr(15'h10, 32'h0000_0001); rd(15'h10, v);
        chk("D30_int_w1c", (v[0] === 1'b0));

        hw_fault[2] = 1'b1; repeat (4) @(posedge clk);     // OT
        wr(15'h14, 32'h0000_0000); rd(15'h14, v);
        chk("D31_int_en_zero_masks_irq", (irq === 1'b0) && (int_status[2] === 1'b1));
        wr(15'h14, 32'h0000_0004);
        chk("D32_int_en_unmasks_irq", (irq === 1'b1));
        wr(15'h14, 32'h0000_0000);
        wr(15'h0C, 32'h0000_0004);
        hw_fault[2] = 1'b0; repeat (4) @(posedge clk);
        wr(15'h0C, 32'h0000_0004);
        rd(15'h0C, v);
        chk("D33_fault_register_clean", (v[10:0] === 11'd0));

        status_in[1] = 1'b1;                              // ACTIVE
        rd(15'h18, v);
        chk("D34_status_is_live_not_latched", (v == 32'h0000_0007));
        status_in[1] = 1'b0;

        // ================= E. clock rate / phase independence ==============
        $display("--- E. SCLK rate and phase ---");
        t_half = 125.0;                                   // 4 MHz
        wr(15'h1C, 32'h0000_0ABC); rd(15'h1C, v);
        chk("D35_slow_sclk_4mhz", v == 32'h0000_0ABC);
        t_half = 50.0;

        // start the transaction on several core-clock phases
        for (i = 0; i < 4; i = i + 1) begin
            @(posedge clk);
            #(i * 2.5);
            wr(15'h1C, 32'h0000_0100 + i); rd(15'h1C, v);
            if (i == 0) chk("D36_phase0_readback", v == 32'h0000_0100);
            if (i == 1) chk("D37_phase1_readback", v == 32'h0000_0101);
            if (i == 2) chk("D38_phase2_readback", v == 32'h0000_0102);
            if (i == 3) chk("D39_phase3_readback", v == 32'h0000_0103);
        end

        // write must not drive MISO: during a write the data phase returns 0
        spi_xfer({1'b0, 15'h1C}, 32'h0000_0555, v, 48);
        chk("D40_write_returns_zeros", v == 32'd0);

        // 48-bit framing was really seen
        chk("D41_frame_full_seen", (n_frames > 30) && (frame_full_o === 1'b1 || bitcnt_o == 6'd48));

        // non-vacuity floors (real DUT events, not TB bookkeeping)
        chk("D42_enough_real_writes", n_wr_ok >= 20);
        chk("D43_enough_real_reads", n_rd_ok >= 30);

        // ================= F. RUN01 复审新增的非空洞检查 ===================
        $display("--- F. SPI frame / CDC / safety ---");

        // F1 read first byte: 32 data bits, the first one must be a defined
        // level (not X/Z from a mis-timed tri-state) and the whole word exact.
        wr(15'h1C, 32'h0000_A55A);
        rd(15'h1C, v2);
        chk("D44_new_read_first_byte_defined",
            (rd_first_def === 1'b1) && (rd_first_byte === 8'h00) &&
            (v2 == 32'h0000_A55A) && (rd_first_bit === 1'b0));

        // F2 a READ at a W1C address must never clear anything
        hw_fault[1] = 1'b1; repeat (4) @(posedge clk); hw_fault[1] = 1'b0;
        repeat (4) @(posedge clk);
        rd(15'h0C, v);                                    // latched bit 1
        v3 = v;
        rd(15'h0C, v2);                                   // read again, data all-ones
        chk("D45_new_read_is_not_write",
            (v3[1] === 1'b1) && (v2[1] === 1'b1) && (fault_status[1] === 1'b1));
        wr(15'h0C, 32'h0000_0002); rd(15'h0C, v);         // clean up with a real W1C
        chk("D46_new_w1c_still_works_after_read", (v[1] === 1'b0));

        // F3 47 edges: rejected, no commit, register untouched
        cap_base = wr_cap_cnt_o;
        spi_len({1'b0, 15'h1C}, 33'h0_0000_BEEF, 47);
        rd(15'h1C, v2);
        chk("D47_new_47_edges_rejected",
            (v2 == 32'h0000_A55A) && (wr_cap_cnt_o == cap_base) && (mb_badlen_o === 1'b1));

        // F4 49 edges: rejected, no commit, register untouched
        cap_base = wr_cap_cnt_o;
        spi_len({1'b0, 15'h1C}, 33'h1_0000_BEEF, 49);
        rd(15'h1C, v2);
        chk("D48_new_49_edges_rejected",
            (v2 == 32'h0000_A55A) && (wr_cap_cnt_o == cap_base));

        // F5 MISO is high-Z while deselected and driven while selected
        chk("D49_new_miso_hiz_when_deselected", (spi_cs_n === 1'b1) && (spi_miso === 1'bz));
        spi_begin;
        #(2.0);
        v_sel = (spi_miso !== 1'bz);
        spi_end;
        chk("D50_new_miso_driven_when_selected", (v_sel === 1'b1));

        // F6 read snapshot must not tear when a live register changes mid-read
        col_index_in = 10'd321;
        repeat (2) @(posedge clk);
        spi_begin;
        v3 = {1'b1, 15'h24};
        for (i = 15; i >= 0; i = i - 1) spi_bit(v3[i], 1'b0);
        v2 = 32'd0;
        for (i = 31; i >= 0; i = i - 1) begin
            if (i == 24) col_index_in = 10'd999;           // change mid data phase
            spi_bit(1'b0, 1'b1);
            if (last_sample === 1'b1) v2[i] = 1'b1;
        end
        spi_end;
        chk("D51_new_read_snapshot_not_torn", v2 == 32'd321);
        col_index_in = 10'd0;

        // F7 CDC: no frame lost, none delivered twice
        chk("D52_new_cdc_no_loss_no_duplicate",
            (mb_lost_o === 1'b0) && (wr_frames_o == wr_cap_cnt_o) && (n_cap == wr_cap_cnt_o));

        // F8 EN=0 must force BLANK even when the write asks for BLANK=0
        wr(15'h04, 32'h0000_0003);                        // EN=1, BLANK=1
        wr(15'h0C, 32'h0000_0100);                        // clear any err
        wr(15'h04, 32'h0000_0001); rd(15'h04, v);         // EN=1, BLANK=0
        chk("D53_new_display_enabled", (v == 32'h0000_0001) && (hw_blank === 1'b0));
        wr(15'h04, 32'h0000_0000); rd(15'h04, v);         // EN=0 + BLANK=0 request
        chk("D54_new_en_zero_forces_blank",
            (v == 32'h0000_0002) && (en === 1'b0) && (blank_r === 1'b1) && (hw_blank === 1'b1));

        // F9 a live hardware fault blocks unblanking, whatever INT_EN says
        hw_fault[3] = 1'b1; repeat (4) @(posedge clk);
        wr(15'h14, 32'h0000_0000);                        // INT_EN = 0
        err_base = err_reg;
        wr(15'h04, 32'h0000_0001); rd(15'h04, v);         // try EN=1, BLANK=0
        chk("D55_new_fault_blocks_unblank",
            (en === 1'b0) && (hw_blank === 1'b1) && (err_reg === 1'b1) && (v[0] === 1'b0));
        chk("D56_new_int_en_zero_keeps_safety", (irq === 1'b0) && (hw_blank === 1'b1));
        hw_fault[3] = 1'b0; repeat (4) @(posedge clk);
        wr(15'h0C, 32'h0000_0100);

        // F10 the documented minimum CS_n high time is what the TB actually used
        chk("D57_new_cs_high_gap_legal", cs_gap >= 100.0);


        $display("--- Phase S: L1 safety (capture / read / clear / re-arm) ---");
        clk_en = 1'b1;
        fault_n = 1'b1;
        hw_fault = 11'd0;
        status_in = 4'b0101;                  // SAFE=1 READY=1
        reset_n = 1'b0; #100; reset_n = 1'b1; #100;
        mon_en = 1'b1;
        split_cnt = 0; resurrect_cnt = 0; perm_samples = 0;

        // (1) a lit panel first
        wr(15'h04, 32'h0000_0001);            // EN=1 BLANK=0
        repeat (4) @(negedge clk);
        chk("E01_armed_before_the_pulse",
            (disp_armed_o === 1'b1) && (hw_blank === 1'b0));

        // (2) a 4 ns pulse that no core edge can see
        fault_pulse_4ns;
        chk("E02_pulse_really_escaped_the_clock", (pulse_edges == 0));
        @(posedge clk); #1;
        chk("E03_capture_caught_it",        (hw_fault_caught_o === 1'b1));
        chk("E04_status_reports_comm_to",   (fault_status[7]    === 1'b1));
        chk("E05_int_reports_it_too",       (int_status[7]      === 1'b1));
        chk("E06_panel_dark",               (hw_blank           === 1'b1));
        repeat (3) @(posedge clk);
        chk("E07_no_auto_restore_after_the_source_is_gone",
            (fault_status[7] === 1'b1) && (hw_blank === 1'b1));

        // (3) ONE legal W1C must clear status[7] and the capture on the SAME edge
        st7_d = fault_status[7];
        split_cnt = 0; resurrect_cnt = 0;
        wr(15'h0C, 32'h0000_0080);            // W1C covering COMM_TO
        repeat (4) @(posedge clk);
        chk("E08_status_and_capture_zero_together", (fault_status[7] === 1'b0) &&
                                                    (hw_fault_caught_o === 1'b0));
        chk("E09_no_edge_with_status_cleared_but_capture_set", (split_cnt == 0));
        chk("E10_bit7_did_not_resurrect", (resurrect_cnt == 0));
        chk("E11_still_dark_without_a_rearm", (hw_blank === 1'b1));

        // (4) explicit re-arm, and it must really permit conduction
        perm_samples = 0;
        wr(15'h04, 32'h0000_0001);            // EN=1 BLANK=0 again
        repeat (4) @(negedge clk);
        for (k = 0; k < 3; k = k + 1) begin
            if ((disp_armed_o === 1'b1) && (hw_blank === 1'b0)) perm_samples = perm_samples + 1;
            @(negedge clk);
        end
        chk("E12_rearm_accepted", (disp_armed_o === 1'b1) && (hw_blank === 1'b0));
        chk("E13_three_nonzero_permit_samples", (perm_samples == 3));

        // (5) W1C = 0 clears nothing
        fault_pulse_4ns;
        @(posedge clk); @(posedge clk);
        wr(15'h0C, 32'h0000_0000);
        repeat (3) @(posedge clk);
        chk("E14_w1c_zero_keeps_the_capture", (fault_status[7] === 1'b1) &&
                                              (hw_fault_caught_o === 1'b1));

        // (6) a partial mask keeps bit7
        wr(15'h0C, 32'h0000_0001);
        repeat (3) @(posedge clk);
        chk("E15_partial_mask_keeps_bit7", (fault_status[7] === 1'b1));

        // (7) a still-asserted source cannot be cleared
        hw_fault = 11'h080;
        wr(15'h0C, 32'h0000_0080);
        repeat (3) @(posedge clk);
        chk("E16_live_source_cannot_be_cleared", (fault_status[7] === 1'b1));
        hw_fault = 11'd0;
        repeat (2) @(posedge clk);
        wr(15'h0C, 32'h0000_0080);
        repeat (3) @(posedge clk);
        chk("E17_clear_works_once_the_source_is_gone", (fault_status[7] === 1'b0));

        // (8) a read transaction must never clear
        fault_pulse_4ns;
        @(posedge clk); @(posedge clk);
        rd(15'h0C, v);
        repeat (3) @(posedge clk);
        chk("E18_read_cannot_clear", (fault_status[7] === 1'b1));

        // (9) dead clock: the capture must hold and the clear must be impossible
        clk_en = 1'b0;
        #200;
        chk("E19_dark_with_a_dead_clock", (hw_blank === 1'b1) &&
                                          (fault_status[7] === 1'b1));
        clk_en = 1'b1;
        #50;
        chk("E20_capture_survived_the_dead_clock", (hw_fault_caught_o === 1'b1));

        $display("T6A_L1_SAFETY split_edges=%0d resurrections=%0d permit_samples=%0d",
                 split_cnt, resurrect_cnt, perm_samples);

        $display("T6A_COUNTS writes=%0d reads=%0d commits=%0d sw_pulses=%0d frames=%0d captured=%0d rejected=%0d",
                 n_wr_ok, n_rd_ok, n_commit, n_sw_pulse, n_frames, wr_cap_cnt_o, wr_rej_cnt_o);
        $display("T6A_CDC frames_sent=%0d frames_captured=%0d lost=%0d badlen=%0d",
                 wr_frames_o, wr_cap_cnt_o, mb_lost_o, mb_badlen_o);
        $display("COURSE_RTL_ASSERT_COUNT=%0d", assert_count);
        $display("COURSE_RTL_PASS_COUNT=%0d",   pass_count);
        $display("COURSE_RTL_FAIL_COUNT=%0d",   fail_count);
        $display("COURSE_RTL_RESULT=%s", (fail_count == 0) ? "PASS" : "FAIL");
        $finish;
    end

    initial begin
        #20000000;                                        // 20 ms guard
        $display("COURSE_T6A_WALL_GUARD=trip");
        $display("COURSE_RTL_ASSERT_COUNT=%0d", assert_count);
        $display("COURSE_RTL_FAIL_COUNT=1");
        $display("COURSE_RTL_RESULT=FAIL");
        $finish;
    end
