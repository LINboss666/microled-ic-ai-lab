`timescale 1ns / 1ps
// t5_five_chip_shell — 1 TCON + 2 Data + 2 Scan, real RTL, T5-A wiring draft.
//
// STATUS: T5-A (zero-ModelSim). This file is COMPILED but never simulated in
// this round. It exists to prove the five real modules elaborate together with
// the V1.3 pin set, and to freeze the wiring table before any T5 budget is
// spent.
//
// ============================ PIN HONESTY ==============================
// Data's `din_valid` is NOT a V1.3 chip pin. V1.3 段0548 says the Data driver
// itself delimits a transfer by LATCH and counts 384 DCLK edges ("无需额外
// VALID线 ... 接收384字节，之后停写"). TCON's `load_active` is an RTL-internal
// observation output (it does not appear anywhere in V1.3), and it is
// DELIBERATELY NOT CONNECTED to `din_valid` here: doing so would invent a
// physical pin.
//
// What is connected instead: NOTHING. Both Data instances run with
// `USE_DIN_VALID=0` (the V1.3 self-delimiting receiver added by T3-H2) and
// their `din_valid` port is tied to 1'b0, exactly as the final chip wrapper
// does. The 384-edge demarcation is therefore performed INSIDE the chip and no
// shell logic can fabricate a byte-valid signal. The earlier `dv_stub` was
// withdrawn because it produced 385 valid edges (reviewer T5-A finding P0).
//
// STATUS: this file compiles only against the T3-H2 RTL (parameter
// USE_DIN_VALID). It is a wiring draft for T5 and is NOT evidence of anything
// until T3-H2 is independently reviewed and T5 is separately authorized.
// =======================================================================
module t5_five_chip_shell #(
    parameter integer N_COL = 1024
) (
    // ---- TCON video + config -------------------------------------------
    input  wire        clk_ref,
    input  wire        reset_n,
    input  wire        pclk,
    input  wire        vsync,
    input  wire        col_sync,
    input  wire [7:0]  video_d,
    input  wire        de,
    input  wire        cfg_valid,
    input  wire [1:0]  cfg_mode,
    input  wire        run_en,
    input  wire        rx_ready,
    input  wire        fault_n,

    // ---- per-Data configuration (in the chip these come from SPI) ------
    input  wire        cfg_hybrid,
    input  wire        cfg0_lut_we,
    input  wire [7:0]  cfg0_lut_addr,
    input  wire [8:0]  cfg0_lut_data,
    input  wire        cfg0_lut_commit,
    input  wire        cfg1_lut_we,
    input  wire [7:0]  cfg1_lut_addr,
    input  wire [8:0]  cfg1_lut_data,
    input  wire        cfg1_lut_commit,

    // ---- observation ----------------------------------------------------
    output wire [383:0]  d0_ch_en,
    output wire [383:0]  d1_ch_en,
    output wire [1151:0] d0_ch_pam,
    output wire [1151:0] d1_ch_pam,
    output wire          d0_col_valid,
    output wire          d1_col_valid,
    output wire          d0_lut_valid,
    output wire          d1_lut_valid,
    output wire [511:0]  s0_col_a,
    output wire [511:0]  s1_col_a,
    output wire [9:0]    tcon_col_id,
    output wire          tcon_scan_stb
);

    // ---- TCON pins carried to the four drivers --------------------------
    wire        dclk, latch, blank, pwm_start, pwm_tick;
    wire [7:0]  d0, d1;
    wire [9:0]  col_id_w;
    wire        scan_stb_w;
    wire        load_active_obs;    // observation only - NEVER wired to din_valid

    tcon_top #(.N_COL(N_COL)) u_tcon (
        .clk_ref (clk_ref), .reset_n (reset_n),
        .pclk (pclk), .vsync (vsync), .col_sync (col_sync),
        .video_d (video_d), .de (de),
        .cfg_valid (cfg_valid), .cfg_mode (cfg_mode), .run_en (run_en),
        .rx_ready (rx_ready), .fault_n (fault_n),
        .col_id (col_id_w), .scan_stb (scan_stb_w),
        .latch (latch), .pwm_start (pwm_start), .pwm_tick (pwm_tick),
        .blank (blank), .dclk (dclk), .load_active (load_active_obs),
        .d0 (d0), .d1 (d1),
        .status (), .frame_wrap (), .slot_long (), .slot_start (),
        .err_flags (), .err_now (), .bank_valid (), .bytes_rcvd (),
        .bytes_sent (), .load_blank (), .load_cnt (),
        .fsm_state (), .frame_id (), .in_col (),
        .disp_valid (), .disp_poison (), .disp_col (), .disp_frame (),
        .fwd_valid (), .fwd_poison (), .fwd_col (), .fwd_frame (),
        .load_done ()
    );

    assign tcon_col_id   = col_id_w;
    assign tcon_scan_stb = scan_stb_w;

    // ---- NO byte-window stub any more (reviewer T5-A finding P0) ----------
    // The previous `dv_stub` produced 385 valid edges (arming edge + 384 more)
    // and was therefore never a proof of the 384-byte protocol. It is gone:
    // both Data instances now run with USE_DIN_VALID=0 and their `din_valid`
    // port tied to 1'b0, so the 384-edge delimiting is done INSIDE the chip and
    // nothing in this shell can fabricate it.
    //
    // Consequence: this shell is only exercisable after the T3-H2 patch is
    // accepted; until then `vlog` still compiles it but the parameter does not
    // exist in the accepted RTL. Do not cite this file as T5 evidence.

    // ---- Data #0 (even rows 0,2,..766) ----------------------------------
    data_driver #(.NCH(384), .PWM_MAX(63), .CNTW(8), .USE_DIN_VALID(0)) u_data0 (
        .reset_n (reset_n), .dclk (dclk), .din (d0), .din_valid (1'b0),
        .latch (latch), .blank (blank), .fault_n (fault_n),
        .pwm_start (pwm_start), .pwm_tick (pwm_tick),
        .cfg_hybrid (cfg_hybrid),
        .cfg_lut_we (cfg0_lut_we), .cfg_lut_addr (cfg0_lut_addr),
        .cfg_lut_data (cfg0_lut_data), .cfg_lut_commit (cfg0_lut_commit),
        .ch_en (d0_ch_en), .ch_pam (d0_ch_pam), .tick_cnt (),
        .col_valid (d0_col_valid), .bytes_rcvd (), .lut_valid (d0_lut_valid),
        .lut_bank (), .lut_cmt_bad (), .err_flags ()
    );

    // ---- Data #1 (odd rows 1,3,..767) -----------------------------------
    data_driver #(.NCH(384), .PWM_MAX(63), .CNTW(8), .USE_DIN_VALID(0)) u_data1 (
        .reset_n (reset_n), .dclk (dclk), .din (d1), .din_valid (1'b0),
        .latch (latch), .blank (blank), .fault_n (fault_n),
        .pwm_start (pwm_start), .pwm_tick (pwm_tick),
        .cfg_hybrid (cfg_hybrid),
        .cfg_lut_we (cfg1_lut_we), .cfg_lut_addr (cfg1_lut_addr),
        .cfg_lut_data (cfg1_lut_data), .cfg_lut_commit (cfg1_lut_commit),
        .ch_en (d1_ch_en), .ch_pam (d1_ch_pam), .tick_cnt (),
        .col_valid (d1_col_valid), .bytes_rcvd (), .lut_valid (d1_lut_valid),
        .lut_bank (), .lut_cmt_bad (), .err_flags ()
    );

    // ---- Scan #0 (even global columns 0,2,..1022) -----------------------
    scan_driver #(.CHIP_ID(0)) u_scan0 (
        .reset_n (reset_n), .blank (blank), .fault_n (fault_n),
        .scan_stb (scan_stb_w), .col_id (col_id_w), .col_a (s0_col_a),
        .sel_valid (), .sel_col (), .sel_chan (), .err_stb_outside_blank ()
    );

    // ---- Scan #1 (odd global columns 1,3,..1023) ------------------------
    scan_driver #(.CHIP_ID(1)) u_scan1 (
        .reset_n (reset_n), .blank (blank), .fault_n (fault_n),
        .scan_stb (scan_stb_w), .col_id (col_id_w), .col_a (s1_col_a),
        .sel_valid (), .sel_col (), .sel_chan (), .err_stb_outside_blank ()
    );

endmodule
