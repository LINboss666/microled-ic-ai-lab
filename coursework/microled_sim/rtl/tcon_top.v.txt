`timescale 1ns / 1ps
// TCON top level — T2B: V1.3 column-start input, frame+column tags, latched safe
// state. See t2b/T2B_V13_FROZEN_INTERFACE.md, t2b/FAULT_RECOVERY_FSM.md.
//
// WHAT IS IMPLEMENTED
//   * column time base (1024 slots, 576x977 + 448x976 = 1,000,000 cycles)
//   * BLANK / LATCH / PWM_START / PWM_TICK / SCAN_STB generated here
//   * 768-byte column reception with a COL_SYNC + DE + byte0 start edge (V1.3)
//   * two 768 B column banks, ping-pong by column parity
//   * 384 bytes per Data port per load window on a CONTINUOUS 50 MHz DCLK,
//     even/odd row split (D0 = row 2k, D1 = row 2k+1)
//   * a latch-safe FSM (RESET / IDLE / ACTIVE / FAULT) that withholds the
//     display permit after any data, load or protocol error until reset_n
//   * frame+column tags for the four pipeline stages:
//        rx(s) = s, fwd(s) = s, disp(s) = s-1, lit(s) = s-2
//     COL_ID is driven from the tracked Data display-bank label, never from a
//     hand-computed "s-1".
//
// WHAT IS NOT IMPLEMENTED (do not read the module name as a chip)
//   SPI/register file, interrupts, CDC, 384/512-channel transistor-level array,
//   grey-scale physical engine, level shifters, layout, STA, PVT/SOA, whole chip.
//   Data and Scan behaviour models live in the testbench only: they are course
//   verification stand-ins, not a 384/512-channel electrical implementation.
module tcon_top #(
    parameter integer N_COL = 1024
) (
    input  wire        clk_ref,
    input  wire        reset_n,

    // ---- video front-end (V1.3: COL_SYNC marks the first byte of a column) --
    input  wire        pclk,         // = clk_ref in this model
    input  wire        vsync,
    input  wire        col_sync,
    input  wire [7:0]  video_d,
    input  wire        de,

    // ---- configuration / readiness -----------------------------------------
    input  wire        cfg_valid,
    input  wire [1:0]  cfg_mode,     // 0 = Pure 8-bit PWM, 1 = Hybrid 3+6
    input  wire        run_en,
    input  wire        rx_ready,     // MODE-level front-end ready (not per column)
    input  wire        fault_n,      // open-drain wired-AND, active low

    // ---- to dual Scan -------------------------------------------------------
    output reg  [9:0]  col_id,
    output wire        scan_stb,

    // ---- to dual Data -------------------------------------------------------
    output wire        latch,
    output wire        pwm_start,
    output wire        pwm_tick,
    output wire        blank,
    output wire        dclk,         // continuous 50 MHz
    output wire        load_active,
    output wire [7:0]  d0,
    output wire [7:0]  d1,

    // ---- observability ------------------------------------------------------
    output wire [7:0]  status,
    output wire        frame_wrap,
    output wire        slot_long,
    output wire        slot_start,
    output wire [5:0]  err_flags,
    output wire [5:0]  err_now,
    output wire [1:0]  bank_valid,
    output wire [9:0]  bytes_rcvd,
    output wire [8:0]  bytes_sent,
    output wire        load_blank,
    output wire [9:0]  load_cnt,

    // ---- T2B labels / FSM observation (simulation observation only) --------
    output wire [1:0]  fsm_state,
    output wire [5:0]  frame_id,
    output wire [9:0]  in_col,
    output reg         disp_valid,
    output reg         disp_poison,
    output reg  [9:0]  disp_col,
    output reg  [5:0]  disp_frame,
    output wire        fwd_valid,
    output wire        fwd_poison,
    output wire [9:0]  fwd_col,
    output wire [5:0]  fwd_frame,
    output wire        load_done
);

    // ---------------------------------------------------------------------
    // timing constants (single source of truth)
    // ---------------------------------------------------------------------
    localparam integer WIN_START    = 768;
    localparam integer LATCH_AT     = 828;
    localparam integer PWM_START_AT = 840;
    localparam integer SCAN_STB_AT  = 880;
    localparam integer PULSE_LEN    = 4;     // 4 x 10 ns = 40 ns
    localparam integer TICK_DIV     = 12;    // 120 ns Hybrid @100 Hz
    localparam integer LOAD_START   = LATCH_AT + PULSE_LEN;   // 832
    localparam integer LOAD_LEN     = 768;
    localparam integer NB_PORT      = 384;
    localparam integer DEADLINE     = LOAD_START - 2;         // 830

    // ---------------------------------------------------------------------
    // FSM
    // ---------------------------------------------------------------------
    localparam [1:0] ST_RESET  = 2'd0;
    localparam [1:0] ST_IDLE   = 2'd1;   // no frame seen yet: timing skeleton only
    localparam [1:0] ST_ACTIVE = 2'd2;   // V1.3 mode: tags + display permit
    localparam [1:0] ST_FAULT  = 2'd3;   // latched safe state (Q3=B)

    reg [1:0] state;
    assign fsm_state = state;

    wire act = (state == ST_ACTIVE);
    wire flt = (state == ST_FAULT);

    // ---------------------------------------------------------------------
    // readiness / safety
    // ---------------------------------------------------------------------
    wire mode_ok = (cfg_mode == 2'd0) || (cfg_mode == 2'd1);
    wire mode_en = reset_n && cfg_valid && mode_ok && run_en;
    wire safe    = mode_en && rx_ready && fault_n;

    // ---------------------------------------------------------------------
    // time base
    // ---------------------------------------------------------------------
    wire [10:0] slot_cycle;
    wire [10:0] slot_len;
    wire [9:0]  tb_col;
    wire        tb_start, tb_last, tb_wrap;

    tcon_timebase #(.N_COL(N_COL)) u_tb (
        .clk        (clk_ref),
        .reset_n    (reset_n),
        .enable     (mode_en),
        .slot_cycle (slot_cycle),
        .slot_len   (slot_len),
        .col_index  (tb_col),
        .slot_start (tb_start),
        .slot_last  (tb_last),
        .frame_wrap (tb_wrap),
        .slot_long  (slot_long)
    );

    // ---------------------------------------------------------------------
    // slot phase sequence
    // ---------------------------------------------------------------------
    wire window, blank_seq, latch_seq, pstart_seq, stb_seq, tick_base, colid_stable;

    tcon_slot_seq #(
        .WIN_START    (WIN_START),
        .LATCH_AT     (LATCH_AT),
        .PWM_START_AT (PWM_START_AT),
        .SCAN_STB_AT  (SCAN_STB_AT),
        .PULSE_LEN    (PULSE_LEN),
        .TICK_DIV     (TICK_DIV)
    ) u_seq (
        .slot_cycle    (slot_cycle),
        .slot_len      (slot_len),
        .window        (window),
        .blank         (blank_seq),
        .latch         (latch_seq),
        .pwm_start     (pstart_seq),
        .scan_stb      (stb_seq),
        .tick_base     (tick_base),
        .colid_stable  (colid_stable)
    );

    // ---------------------------------------------------------------------
    // video receive -> column bank
    // ---------------------------------------------------------------------
    wire       wr_en, wr_bank, col_valid, col_poison, col_start_ok;
    wire [9:0] wr_idx;
    wire [7:0] wr_data;
    wire       rd_bank;
    wire [8:0] rd_idx;
    wire [7:0] bk_d0, bk_d1;

    tcon_video_rx #(
        .RX_LEN      (768),
        .DEADLINE    (DEADLINE)
    ) u_rx (
        .clk        (clk_ref),
        .reset_n    (reset_n),
        .safe       (safe),
        .active     (act),
        .slot_cycle (slot_cycle),
        .col_index  (tb_col),
        .vsync      (vsync),
        .col_sync   (col_sync),
        .video_d    (video_d),
        .de         (de),
        .wr_en      (wr_en),
        .wr_idx     (wr_idx),
        .wr_data    (wr_data),
        .wr_bank    (wr_bank),
        .col_valid  (col_valid),
        .col_poison (col_poison),
        .in_col     (in_col),
        .frame_id   (frame_id),
        .err_flags  (err_flags),
        .err_now    (err_now),
        .bytes_rcvd (bytes_rcvd)
    );

    tcon_col_bank u_bank (
        .clk     (clk_ref),
        .wr_en   (wr_en),
        .wr_bank (wr_bank),
        .wr_idx  (wr_idx),
        .wr_data (wr_data),
        .rd_bank (rd_bank),
        .rd_idx  (rd_idx),
        .rd_d0   (bk_d0),
        .rd_d1   (bk_d1)
    );

    // ---------------------------------------------------------------------
    // Data transmit
    // ---------------------------------------------------------------------
    tcon_data_tx #(
        .LOAD_START (LOAD_START),
        .LOAD_LEN   (LOAD_LEN),
        .NB_PORT    (NB_PORT)
    ) u_tx (
        .clk         (clk_ref),
        .reset_n     (reset_n),
        .safe        (safe),
        .fault       (flt),
        .slot_cycle  (slot_cycle),
        .col_index   (tb_col),
        .col_valid   (col_valid),
        .col_poison  (col_poison),
        .frame_in    (frame_id),
        .rd_bank     (rd_bank),
        .rd_idx      (rd_idx),
        .bk_d0       (bk_d0),
        .bk_d1       (bk_d1),
        .load_active (load_active),
        .dclk        (dclk),
        .d0          (d0),
        .d1          (d1),
        .fwd_col     (fwd_col),
        .fwd_frame   (fwd_frame),
        .fwd_poison  (fwd_poison),
        .fwd_valid   (fwd_valid),
        .load_done   (load_done),
        .bytes_sent  (bytes_sent),
        .load_blank  (load_blank),
        .load_cnt_o  (load_cnt)
    );

    // ---------------------------------------------------------------------
    // forward-label bookkeeping: per bank, the last window that completed
    // ---------------------------------------------------------------------
    reg [1:0]  fw_valid_r;      // per bank
    reg [1:0]  fw_poison_r;     // per bank
    reg [9:0]  fw_col0, fw_col1;
    reg [5:0]  fw_frame0, fw_frame1;
    reg [1:0]  last_bank;       // bank whose window completed most recently

    always @(posedge clk_ref or negedge reset_n) begin
        if (!reset_n) begin
            fw_valid_r  <= 2'b00;
            fw_poison_r <= 2'b00;
            fw_col0     <= 10'd0;
            fw_col1     <= 10'd0;
            fw_frame0   <= 6'd0;
            fw_frame1   <= 6'd0;
            last_bank   <= 2'd0;
        end else if (load_done) begin
            last_bank <= {1'b0, fwd_col[0]};
            if (fwd_col[0]) begin
                fw_valid_r[1]  <= fwd_valid;
                fw_poison_r[1] <= fwd_poison;
                fw_col1        <= fwd_col;
                fw_frame1      <= fwd_frame;
            end else begin
                fw_valid_r[0]  <= fwd_valid;
                fw_poison_r[0] <= fwd_poison;
                fw_col0        <= fwd_col;
                fw_frame0      <= fwd_frame;
            end
        end
    end

    wire        fw_sel_valid  = last_bank[0] ? fw_valid_r[1]  : fw_valid_r[0];
    wire        fw_sel_poison = last_bank[0] ? fw_poison_r[1] : fw_poison_r[0];
    wire [9:0]  fw_sel_col    = last_bank[0] ? fw_col1        : fw_col0;
    wire [5:0]  fw_sel_frame  = last_bank[0] ? fw_frame1      : fw_frame0;
    wire        fw_ok         = fw_sel_valid && !fw_sel_poison;

    // ---------------------------------------------------------------------
    // display-bank label = what the Data driver holds after LATCH
    // ---------------------------------------------------------------------
    wire latch_evt = (slot_cycle == (LATCH_AT - 1));   // LATCH pulse starts next cycle

    always @(posedge clk_ref or negedge reset_n) begin
        if (!reset_n) begin
            disp_col    <= 10'd0;
            disp_frame  <= 6'd0;
            disp_valid  <= 1'b0;
            disp_poison <= 1'b0;
        end else if (flt) begin
            // frozen: no display permit can be manufactured while latched
            disp_valid  <= 1'b0;
        end else if (act && latch_evt) begin
            disp_col    <= fw_sel_col;
            disp_frame  <= fw_sel_frame;
            disp_valid  <= fw_sel_valid;
            disp_poison <= fw_sel_poison;
        end
    end

    wire disp_ok = disp_valid && !disp_poison;

    // ---------------------------------------------------------------------
    // BLANK / strobes: no display permit without a valid display label
    // ---------------------------------------------------------------------
    assign blank     = blank_seq || !safe || flt || (act && !disp_ok);
    assign latch     = latch_seq  && safe && !flt && (!act || fw_ok);
    assign pwm_start = pstart_seq && safe && !flt && (!act || disp_ok);
    assign scan_stb  = stb_seq    && safe && !flt && (!act || disp_ok);
    assign pwm_tick  = tick_base  && window && safe && !flt && (!act || disp_ok);

    // ---------------------------------------------------------------------
    // COL_ID: tracked display-bank label in ACTIVE, local column in IDLE
    // ---------------------------------------------------------------------
    always @(posedge clk_ref or negedge reset_n) begin
        if (!reset_n)        col_id <= 10'd0;
        else if (flt)        ;                            // frozen
        else if (!act)       col_id <= tb_col;            // skeleton (no video)
        else if (latch_evt)  col_id <= fw_sel_col;        // what LATCH just made current
    end

    // ---------------------------------------------------------------------
    // FSM transitions
    // ---------------------------------------------------------------------
    wire frame_start = vsync && col_sync && (slot_cycle == 11'd0) && safe;
    wire err_any     = (|err_now);
    wire fault_evt   = act && (err_any || !fault_n);

    always @(posedge clk_ref or negedge reset_n) begin
        if (!reset_n)          state <= ST_RESET;
        else case (state)
            ST_RESET:          state <= ST_IDLE;
            ST_IDLE:  if (frame_start) state <= ST_ACTIVE;
            ST_ACTIVE: if (fault_evt)  state <= ST_FAULT;
            default:           state <= ST_FAULT;         // latched: reset only
        endcase
    end

    // ---------------------------------------------------------------------
    // status / stickies
    // ---------------------------------------------------------------------
    reg fault_seen, vsync_seen, wrap_seen;

    always @(posedge clk_ref or negedge reset_n) begin
        if (!reset_n) begin
            fault_seen <= 1'b0;
            vsync_seen <= 1'b0;
            wrap_seen  <= 1'b0;
        end else begin
            if (!fault_n)  fault_seen <= 1'b1;
            if (vsync)     vsync_seen <= 1'b1;
            if (tb_wrap)   wrap_seen  <= 1'b1;
        end
    end

    assign status     = { flt, slot_long, wrap_seen, vsync_seen,
                          fault_seen, load_active, blank, safe };
    assign frame_wrap = tb_wrap;
    assign slot_start = tb_start;
    assign bank_valid = { col_valid && tb_col[0], col_valid && !tb_col[0] };

endmodule
