`timescale 1ns / 1ps
// TCON top level — T2A / T2AR: real data path.
//
// WHAT IS IMPLEMENTED
//   * column time base (1024 slots / 576x977 + 448x976 = 1,000,000 cycles)
//   * BLANK / LATCH / PWM_START / PWM_TICK / SCAN_STB / COL_ID generated here
//   * 768-byte column reception with per-column classification
//   * two 768 B column banks, ping-pong by column parity
//   * 384 bytes per Data port per load window on a CONTINUOUS 50 MHz DCLK,
//     even/odd row split (D0 = row 2k, D1 = row 2k+1)
//
// T2AR CHANGES (independent review 2026-11-14)
//   1. `mode_en` (time base) and `safe` (output gating) are separate, so a fault
//      never shifts the frame time base.
//   2. `slot_bad`: if a fault is present when a new column boundary is taken, the
//      whole slot is disarmed (no LATCH / PWM_START / SCAN_STB). COL_ID keeps
//      its old value for that slot, and because no strobe is emitted, a scan
//      strobe can never carry a stale column number. The next clean boundary
//      re-synchronises COL_ID. This is the recovery policy for review P0-C:
//      we do NOT reset the time base to hide the problem.
//   3. DCLK is continuous again (T2A had gated it to the load window).
//
// WHAT IS NOT IMPLEMENTED (do not read the module name as a chip)
//   SPI/register file, interrupts, CDC, 384/512-channel transistor-level array,
//   grey-scale physical engine (belongs to the Data driver), level shifters,
//   layout, STA, PVT/SOA, whole chip.
//
// NOTE ON PCLK: PCLK is the same 100 MHz as CLK_REF in this course model. No
// claim of a verified clock-domain crossing is made anywhere in this design.
module tcon_top #(
    parameter integer N_COL = 1024
) (
    input  wire        clk_ref,
    input  wire        reset_n,

    // ---- video front-end ----------------------------------------------------
    input  wire        pclk,         // = clk_ref in this model
    input  wire        vsync,
    input  wire        col_sync,
    input  wire [7:0]  video_d,
    input  wire        de,

    // ---- configuration / readiness -----------------------------------------
    input  wire        cfg_valid,
    input  wire [1:0]  cfg_mode,     // 0 = Pure 8-bit PWM, 1 = Hybrid 3+6
    input  wire        run_en,
    input  wire        rx_ready,     // MODE-level front-end ready (not per column)
    input  wire        fault_n,      // open-drain wired-AND, active low

    // ---- to dual Scan -------------------------------------------------------
    output reg  [9:0]  col_id,
    output wire        scan_stb,

    // ---- to dual Data -------------------------------------------------------
    output wire        latch,
    output wire        pwm_start,
    output wire        pwm_tick,
    output wire        blank,
    output wire        dclk,         // continuous 50 MHz
    output wire        load_active,
    output wire [7:0]  d0,
    output wire [7:0]  d1,

    // ---- observability ------------------------------------------------------
    output wire [7:0]  status,
    output wire        frame_wrap,
    output wire        slot_long,
    output wire        slot_start,
    output wire [5:0]  err_flags,
    output wire [5:0]  err_now,
    output wire [1:0]  bank_valid,
    output wire [9:0]  bytes_rcvd,
    output wire [8:0]  bytes_sent,
    output wire        load_blank,
    output wire [9:0]  load_cnt       // observability alias (T1 TB reads it)
);

    // ---------------------------------------------------------------------
    // timing constants (single source of truth)
    // ---------------------------------------------------------------------
    localparam integer WIN_START    = 768;
    localparam integer LATCH_AT     = 828;
    localparam integer PWM_START_AT = 840;
    localparam integer SCAN_STB_AT  = 880;
    localparam integer PULSE_LEN    = 4;     // 4 x 10 ns = 40 ns
    localparam integer TICK_DIV     = 12;    // 120 ns Hybrid @100 Hz
    localparam integer LOAD_START   = LATCH_AT + PULSE_LEN;   // 832
    localparam integer LOAD_LEN     = 768;   // 384 bytes/port at 50 MHz
    localparam integer NB_PORT      = 384;
    localparam integer ACCEPT_LAST  = LOAD_START - 2;         // 830

    // ---------------------------------------------------------------------
    // readiness / safety
    // ---------------------------------------------------------------------
    wire mode_ok = (cfg_mode == 2'd0) || (cfg_mode == 2'd1);
    wire mode_en = reset_n && cfg_valid && mode_ok && run_en;
    wire safe    = mode_en && rx_ready && fault_n;

    // ---------------------------------------------------------------------
    // time base
    // ---------------------------------------------------------------------
    wire [10:0] slot_cycle;
    wire [10:0] slot_len;
    wire [9:0]  tb_col;
    wire        tb_start, tb_last, tb_wrap;

    tcon_timebase #(.N_COL(N_COL)) u_tb (
        .clk        (clk_ref),
        .reset_n    (reset_n),
        .enable     (mode_en),
        .slot_cycle (slot_cycle),
        .slot_len   (slot_len),
        .col_index  (tb_col),
        .slot_start (tb_start),
        .slot_last  (tb_last),
        .frame_wrap (tb_wrap),
        .slot_long  (slot_long)
    );

    // ---------------------------------------------------------------------
    // slot phase sequence
    // ---------------------------------------------------------------------
    wire window, blank_seq, latch_seq, pstart_seq, stb_seq, tick_base, colid_stable;

    tcon_slot_seq #(
        .WIN_START    (WIN_START),
        .LATCH_AT     (LATCH_AT),
        .PWM_START_AT (PWM_START_AT),
        .SCAN_STB_AT  (SCAN_STB_AT),
        .PULSE_LEN    (PULSE_LEN),
        .TICK_DIV     (TICK_DIV)
    ) u_seq (
        .slot_cycle    (slot_cycle),
        .slot_len      (slot_len),
        .window        (window),
        .blank         (blank_seq),
        .latch         (latch_seq),
        .pwm_start     (pstart_seq),
        .scan_stb      (stb_seq),
        .tick_base     (tick_base),
        .colid_stable  (colid_stable)
    );

    // ---------------------------------------------------------------------
    // slot disarm: a column boundary taken while unsafe disables the whole slot
    // so that no strobe can ever carry the stale COL_ID (review P0-C).
    // ---------------------------------------------------------------------
    reg slot_bad;
    always @(posedge clk_ref or negedge reset_n) begin
        if (!reset_n)    slot_bad <= 1'b0;
        else if (tb_start) slot_bad <= !safe;
    end

    assign blank     = blank_seq || !safe;
    assign latch     = latch_seq  && safe && !slot_bad;
    assign pwm_start = pstart_seq && safe && !slot_bad;
    assign scan_stb  = stb_seq    && safe && !slot_bad;
    assign pwm_tick  = tick_base  && window && safe && !slot_bad;

    // ---------------------------------------------------------------------
    // video receive -> column bank
    // ---------------------------------------------------------------------
    wire       wr_en, wr_bank, col_valid;
    wire [9:0] wr_idx;
    wire [7:0] wr_data;
    wire       rd_bank;
    wire [8:0] rd_idx;
    wire [7:0] bk_d0, bk_d1;

    tcon_video_rx #(
        .RX_LEN      (768),
        .ACCEPT_LAST (ACCEPT_LAST),
        .LOAD_START  (LOAD_START)
    ) u_rx (
        .clk        (clk_ref),
        .reset_n    (reset_n),
        .safe       (safe),
        .slot_last  (tb_last),
        .slot_cycle (slot_cycle),
        .col_index  (tb_col),
        .vsync      (vsync),
        .col_sync   (col_sync),
        .video_d    (video_d),
        .de         (de),
        .wr_en      (wr_en),
        .wr_idx     (wr_idx),
        .wr_data    (wr_data),
        .wr_bank    (wr_bank),
        .bank_valid (bank_valid),
        .col_valid  (col_valid),
        .err_flags  (err_flags),
        .err_now    (err_now),
        .bytes_rcvd (bytes_rcvd),
        .col_ok     ()
    );

    tcon_col_bank u_bank (
        .clk     (clk_ref),
        .wr_en   (wr_en),
        .wr_bank (wr_bank),
        .wr_idx  (wr_idx),
        .wr_data (wr_data),
        .rd_bank (rd_bank),
        .rd_idx  (rd_idx),
        .rd_d0   (bk_d0),
        .rd_d1   (bk_d1)
    );

    // ---------------------------------------------------------------------
    // Data transmit (continuous DCLK + D0/D1 + load window)
    // ---------------------------------------------------------------------
    tcon_data_tx #(
        .LOAD_START (LOAD_START),
        .LOAD_LEN   (LOAD_LEN),
        .NB_PORT    (NB_PORT)
    ) u_tx (
        .clk         (clk_ref),
        .reset_n     (reset_n),
        .safe        (safe),
        .slot_cycle  (slot_cycle),
        .col_index   (tb_col),
        .col_valid   (col_valid),
        .rd_bank     (rd_bank),
        .rd_idx      (rd_idx),
        .bk_d0       (bk_d0),
        .bk_d1       (bk_d1),
        .load_active (load_active),
        .dclk        (dclk),
        .d0          (d0),
        .d1          (d1),
        .bytes_sent  (bytes_sent),
        .load_blank  (load_blank),
        .load_cnt_o  (load_cnt)
    );

    // ---------------------------------------------------------------------
    // COL_ID: advances only on a clean (safe) column boundary
    // ---------------------------------------------------------------------
    always @(posedge clk_ref or negedge reset_n)
        if (!reset_n) col_id <= 10'd0;
        else if (safe && tb_start) col_id <= tb_col;

    // ---------------------------------------------------------------------
    // status / stickies
    // ---------------------------------------------------------------------
    reg fault_seen, vsync_seen, wrap_seen;

    always @(posedge clk_ref or negedge reset_n) begin
        if (!reset_n) begin
            fault_seen <= 1'b0;
            vsync_seen <= 1'b0;
            wrap_seen  <= 1'b0;
        end else begin
            if (!fault_n)  fault_seen <= 1'b1;
            if (vsync)     vsync_seen <= 1'b1;
            if (tb_wrap)   wrap_seen  <= 1'b1;
        end
    end

    assign status     = { slot_bad, slot_long, wrap_seen, vsync_seen,
                          fault_seen, load_active, blank, safe };
    assign frame_wrap = tb_wrap;
    assign slot_start = tb_start;

endmodule
