`timescale 1ns / 1ps
// TCON top-level skeleton (T1).
//
// WHAT THIS IS
//   * V1.3-shaped functional port list for the TCON, reduced to what T1 owns.
//   * The TCON now GENERATES its own column time base, BLANK, LATCH,
//     PWM_START, PWM_TICK, SCAN_STB and COL_ID. Nothing is driven by the TB.
//   * Safety: whenever video/config/Data readiness is not satisfied, or a fault
//     is present, BLANK is forced high combinationally (0 cycles, far below the
//     V1.3 <=50 ns target) and no control pulse can leave the block.
//
// WHAT THIS IS NOT (do not read the module name as a chip)
//   NOT_IMPLEMENTED in T1: 768-byte column reception, 2x768 B ping-pong buffer,
//   even/odd split, real D0/D1 payload, SPI registers, interrupts, level
//   shifters, Data/Scan transistor level, array, layout, PVT/SOA, STA.
//   D0/D1 are tied to 8'h00 in T1 on purpose.
module tcon_top (
    input  wire        clk_ref,      // 100 MHz system time base
    input  wire        reset_n,      // active low, asynchronous assert

    // ---- video front-end (T1: observed only, not yet consumed) --------------
    input  wire        pclk,
    input  wire        vsync,
    input  wire        col_sync,
    input  wire [7:0]  video_d,
    input  wire        de,

    // ---- configuration / readiness -----------------------------------------
    input  wire        cfg_valid,    // configuration has been loaded
    input  wire [1:0]  cfg_mode,     // 0 = Pure 8-bit PWM, 1 = Hybrid 3+6
    input  wire        run_en,       // register-block enable
    input  wire        rx_ready,     // T2 will drive: column data available
    input  wire        fault_n,      // open-drain wired-AND, active low

    // ---- to dual Scan -------------------------------------------------------
    output reg  [9:0]  col_id,
    output wire        scan_stb,

    // ---- to dual Data -------------------------------------------------------
    output wire        latch,
    output wire        pwm_start,
    output wire        pwm_tick,
    output wire        blank,
    output wire        dclk,         // 50 MHz continuous
    output wire        load_active,  // 768-cycle load window (C interpretation)
    output wire [7:0]  d0,
    output wire [7:0]  d1,

    // ---- observability ------------------------------------------------------
    output wire [7:0]  status,
    output wire        frame_wrap,
    output wire        slot_long,
    output wire        slot_start
);

    // ---------------------------------------------------------------------
    // timing constants shared by the slot sequencer and the load window
    // ---------------------------------------------------------------------
    localparam integer WIN_START    = 768;
    localparam integer LATCH_AT     = 828;
    localparam integer PWM_START_AT = 840;
    localparam integer SCAN_STB_AT  = 880;
    localparam integer PULSE_LEN    = 4;     // 4 x 10 ns = 40 ns
    localparam integer TICK_DIV     = 12;    // 120 ns Hybrid @100 Hz
    localparam integer LOAD_START   = LATCH_AT + PULSE_LEN;  // 832
    localparam integer LOAD_LEN     = 768;   // 384 bytes/port at 50 MHz

    // ---------------------------------------------------------------------
    // readiness / safety (declared first: used by the time-base enable)
    // ---------------------------------------------------------------------
    wire mode_ok = (cfg_mode == 2'd0) || (cfg_mode == 2'd1);
    wire safe    = reset_n && cfg_valid && mode_ok && run_en && rx_ready && fault_n;

    // ---------------------------------------------------------------------
    // time base
    // ---------------------------------------------------------------------
    wire [10:0] slot_cycle;
    wire [10:0] slot_len;
    wire [9:0]  tb_col;
    wire        tb_start, tb_last, tb_wrap;

    tcon_timebase u_tb (
        .clk        (clk_ref),
        .reset_n    (reset_n),
        .enable     (safe),
        .slot_cycle (slot_cycle),
        .slot_len   (slot_len),
        .col_index  (tb_col),
        .slot_start (tb_start),
        .slot_last  (tb_last),
        .frame_wrap (tb_wrap),
        .slot_long  (slot_long)
    );

    // ---------------------------------------------------------------------
    // slot phase sequence
    // ---------------------------------------------------------------------
    wire window, blank_seq, latch_seq, pstart_seq, stb_seq, tick_base, colid_stable;

    tcon_slot_seq #(
        .WIN_START    (WIN_START),
        .LATCH_AT     (LATCH_AT),
        .PWM_START_AT (PWM_START_AT),
        .SCAN_STB_AT  (SCAN_STB_AT),
        .PULSE_LEN    (PULSE_LEN),
        .TICK_DIV     (TICK_DIV)
    ) u_seq (
        .slot_cycle    (slot_cycle),
        .slot_len      (slot_len),
        .window        (window),
        .blank         (blank_seq),
        .latch         (latch_seq),
        .pwm_start     (pstart_seq),
        .scan_stb      (stb_seq),
        .tick_base     (tick_base),
        .colid_stable  (colid_stable)
    );

    // ---------------------------------------------------------------------
    // readiness / safety
    // ---------------------------------------------------------------------
    reg  fault_seen, vsync_seen, wrap_seen;

    // BLANK must be high whenever the pipeline is not safe. Combinational, so a
    // fault reaches BLANK in 0 cycles (V1.3 target: <= 50 ns).
    assign blank = blank_seq || !safe;

    assign latch     = latch_seq  && safe;
    assign pwm_start = pstart_seq && safe;
    assign scan_stb  = stb_seq    && safe;
    assign pwm_tick  = tick_base  && window && safe;

    // 50 MHz continuous DCLK derived from the 100 MHz reference (toggle /1).
    reg dclk_r;
    always @(posedge clk_ref or negedge reset_n)
        if (!reset_n) dclk_r <= 1'b0;
        else          dclk_r <= ~dclk_r;
    assign dclk = dclk_r;

    // COL_ID only advances on a completed, safe slot.
    always @(posedge clk_ref or negedge reset_n)
        if (!reset_n) col_id <= 10'd0;
        else if (safe && tb_start) col_id <= tb_col;

    // ---------------------------------------------------------------------
    // 768-cycle load window, phase-locked to the slot (C interpretation).
    //
    //   LATCH      : cycles [828, 832)   -> 4 cycles = 40 ns
    //   load       : cycles [832, 1600)  -> 768 cycles = 7.68 us  (start = 832)
    //   next LATCH : cycle 1804          -> 204-cycle gap, 200 net after the
    //                                       4-cycle last-data guard
    //
    // Half-open, cross-slot form relative to the NEXT slot start (976):
    //   [832-976, 1600-976) = [-144, 624)
    //
    // T1R: the previous version detected the FALLING edge of LATCH through a
    // register, so the window opened one cycle late (833 instead of 832) and a
    // fault-induced false falling edge could start a phantom transfer. The
    // window is now derived from slot_cycle only, and gated combinationally by
    // `safe`, so a fault closes it in the same cycle and clears the counter.
    // ---------------------------------------------------------------------
    reg        load_run;
    reg [9:0]  load_cnt;

    // load_start_pt is asserted during cycle 831, i.e. the edge that enters 832
    wire load_start_pt = (slot_cycle == (LOAD_START - 1));

    always @(posedge clk_ref or negedge reset_n) begin
        if (!reset_n) begin
            load_run <= 1'b0;
            load_cnt <= 10'd0;
        end else if (!safe) begin
            load_run <= 1'b0;                 // no phantom load on a fault
            load_cnt <= 10'd0;
        end else if (load_start_pt && !load_run) begin
            load_run <= 1'b1;                 // high during [832, 1600)
            load_cnt <= 10'd0;
        end else if (load_run) begin
            if (load_cnt == (LOAD_LEN - 1)) begin
                load_run <= 1'b0;
                load_cnt <= 10'd0;
            end else begin
                load_cnt <= load_cnt + 10'd1;
            end
        end
    end

    // Combinated safety gate: a fault de-asserts load_active in 0 cycles.
    assign load_active = load_run && safe;

    // ---------------------------------------------------------------------
    // status / stickies
    // ---------------------------------------------------------------------
    always @(posedge clk_ref or negedge reset_n) begin
        if (!reset_n) begin
            fault_seen <= 1'b0;
            vsync_seen <= 1'b0;
            wrap_seen  <= 1'b0;
        end else begin
            if (!fault_n)  fault_seen <= 1'b1;
            if (vsync)     vsync_seen <= 1'b1;
            if (tb_wrap)   wrap_seen  <= 1'b1;
        end
    end

    assign status = { 1'b0, slot_long, wrap_seen, vsync_seen,
                      fault_seen, load_run, blank, safe };
    assign frame_wrap = tb_wrap;
    assign slot_start = tb_start;

    // T1: no payload yet. T2 replaces these with the ping-pong buffer outputs.
    assign d0 = 8'h00;
    assign d1 = 8'h00;

endmodule
