`timescale 1ns / 1ps
// tcon_spi_regs — TCON 32-bit register file + SPI->core CDC (T6-A).
//
// V1.3 §13 (段0569-0570, 0704):
//   * all registers are 32 bit, 4-byte aligned, reserved bits read 0 / write 0;
//   * unimplemented address: read 0, write ignored and ERR_REG set;
//   * timing / mode / current / LUT configuration may only be written when
//     EN = 0 and SAFE = 1; a write that violates this sets ERR_REG and is NOT
//     committed;
//   * BLANK assertion and EN clearing take effect at any time;
//   * protection (RESET / hardware FAULT / BLANK) outranks SPI and the
//     interrupt mask; INT_EN only masks the notification.
//
// This module owns the 100 MHz side of the SPI interface. It never samples a
// multi-bit bus without a synchronised single-bit qualifier:
//   write : the frame bus `sh` is frozen once SCLK stops; the qualifier is the
//           2-FF synchronised CS_n RISING edge, which is by construction at
//           least half a SCLK period after the last bit was captured;
//   read  : the frozen address `rd_addr_hold` is sampled only after the 2-FF
//           synchronised `rd_req` level is seen; the 32-bit answer goes back as
//           a stable register that the SPI block latches one falling edge later.
module tcon_spi_regs (
    input  wire        clk,          // 100 MHz core clock
    input  wire        reset_n,

    // ---- SPI-side qualifiers (SCLK domain, see spi_frame) ----------------
    input  wire [47:0] sh,
    input  wire        frame_full,
    input  wire        cs_n,
    input  wire [14:0] rd_addr_hold,
    input  wire        rd_req,
    output reg  [31:0] rd_snapshot,

    // ---- real hardware status / faults (NOT constants) --------------------
    input  wire [3:0]  status_in,    // {LUT_VALID, SAFE, ACTIVE, READY}
    input  wire [10:0] hw_fault,     // UVLO,OV,OT,CLOCK_LOST,FRAME_TO,VID_TIMING,
                                     // UF_OF,COMM_TO,(ERR_REG internal),LUT_ERR,SYNC_ERR
    input  wire        test_en,      // chip TEST_EN pin (not a new SPI pin)
    input  wire [9:0]  col_index_in, // live TCON display column

    // ---- register outputs to the rest of the chip -------------------------
    output reg         en,
    output reg         blank_r,
    output reg         test_mode,
    output wire        sw_reset_pulse,
    output reg         fps_sel,
    output reg  [15:0] wdog,
    output reg  [10:0] fault_status,
    output reg  [10:0] int_status,
    output reg  [10:0] int_en,
    output wire        irq,
    output reg         err_reg,

    // observation
    output wire [10:0] fault_vis_o
);

    // ---------------------------------------------------------------------
    // register addresses (addr[5:2]; °0x00..0x24 of the TCON subset)
    // ---------------------------------------------------------------------
    localparam [3:0] A_CHIPID = 4'h0, A_CTRL  = 4'h1, A_MODE  = 4'h2,
                     A_FAULT  = 4'h3, A_INT   = 4'h4, A_INTEN = 4'h5,
                     A_STATUS = 4'h6, A_WDOG  = 4'h7, A_VIDCFG= 4'h8,
                     A_COLIDX = 4'h9;

    localparam [7:0] CHIP_ID_TCON = 8'h10;   // V1.3 §13 reset column 10/21/22/31/32h

    // ---------------------------------------------------------------------
    // CS_n synchroniser + rising-edge detect (the write commit event)
    // ---------------------------------------------------------------------
    reg cs_m, cs_s1, cs_s2;
    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            cs_m <= 1'b1; cs_s1 <= 1'b1; cs_s2 <= 1'b1;
        end else begin
            cs_m  <= cs_n;
            cs_s1 <= cs_m;
            cs_s2 <= cs_s1;
        end
    end
    wire cs_rise = cs_s1 && !cs_s2;

    // ---------------------------------------------------------------------
    // rd_req synchroniser (read qualifier)
    // ---------------------------------------------------------------------
    reg rq_s1, rq_s2;
    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin rq_s1 <= 1'b0; rq_s2 <= 1'b0; end
        else          begin rq_s1 <= rd_req; rq_s2 <= rq_s1; end
    end

    // ---------------------------------------------------------------------
    // register read mux (combinational, used for the snapshot)
    // ---------------------------------------------------------------------
    wire [3:0] ra = rd_addr_hold[5:2];
    wire       ra_ok = (rd_addr_hold[14:6] == 9'd0) && (rd_addr_hold[1:0] == 2'd0);

    reg [31:0] rval;
    always @(*) begin
        rval = 32'h0000_0000;
        if (ra_ok) begin
            case (ra)
            A_CHIPID: rval = {24'd0, CHIP_ID_TCON};
            A_CTRL  : rval = {28'd0, test_mode, 1'b0, blank_r, en};  // SW_RESET reads 0
            A_MODE  : rval = {31'd0, fps_sel};
            A_FAULT : rval = {21'd0, fault_status};
            A_INT   : rval = {21'd0, int_status};
            A_INTEN : rval = {21'd0, int_en};
            A_STATUS: rval = {28'd0, status_in};
            A_WDOG  : rval = {16'd0, wdog};
            A_VIDCFG: rval = 32'h000B_FFFF;                          // 1023 / 767
            A_COLIDX: rval = {22'd0, col_index_in};
            default : rval = 32'h0000_0000;
            endcase
        end
    end

    // ---------------------------------------------------------------------
    // state + write commit
    // ---------------------------------------------------------------------
    reg        sw_pulse;
    reg [10:0] fs_d;                 // previous visible fault state (for INT latch)
    reg [10:0] n_fault, n_int;

    wire safe_in  = status_in[2];
    wire ready_in = status_in[0];
    wire cfg_ok   = (!en) && safe_in;

    assign sw_reset_pulse = sw_pulse;
    assign irq            = |(int_status & int_en);
    assign fault_vis_o    = fault_status | hw_fault;

    integer k;
    reg [14:0] wa;
    reg [31:0] wd;
    reg        err_set;
    reg [10:0] w1c_f, w1c_i;

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            en           <= 1'b0;
            blank_r      <= 1'b1;               // reset value 0x00000002
            test_mode    <= 1'b0;
            sw_pulse     <= 1'b0;
            fps_sel      <= 1'b0;
            wdog         <= 16'd2;
            fault_status <= 11'd0;
            int_status   <= 11'd0;
            int_en       <= 11'd0;
            err_reg      <= 1'b0;
            rd_snapshot  <= 32'd0;
            fs_d         <= 11'd0;
        end else begin
            err_set = 1'b0;
            sw_pulse <= 1'b0;

            // ---- 1. write commit on the CS_n rising edge -----------------
            if (cs_rise && frame_full) begin
                wa = sh[46:32];
                wd = sh[31:0];
                if (wa[1:0] != 2'b00) begin
                    err_set = 1'b1;                       // not 4-byte aligned
                end else if (wa[14:6] != 9'd0) begin
                    err_set = 1'b1;                       // not implemented
                end else begin
                    case (wa[5:2])
                    A_CHIPID: err_set = 1'b1;             // RO
                    A_CTRL: begin
                        if (wd[3] && !test_en)                       err_set = 1'b1;
                        else if (wd[0] && !safe_in)                  err_set = 1'b1;
                        else if (!wd[1] && !(en && safe_in && ready_in)) err_set = 1'b1;
                        else begin
                            en        <= wd[0];
                            blank_r   <= wd[1];
                            test_mode <= wd[3];
                            if (wd[2]) sw_pulse <= 1'b1;         // W1P, self clearing
                        end
                    end
                    A_MODE: begin
                        if (cfg_ok) fps_sel <= wd[0];
                        else        err_set = 1'b1;
                    end
                    A_WDOG: begin
                        if (!cfg_ok)                      err_set = 1'b1;
                        else if ((wd[15:0] == 16'd0) && !test_en) err_set = 1'b1;
                        else                              wdog <= wd[15:0];
                    end
                    A_FAULT: ;                            // handled below (W1C)
                    A_INT:   ;                            // handled below (W1C)
                    A_INTEN: int_en <= wd[10:0];
                    A_STATUS, A_VIDCFG, A_COLIDX: err_set = 1'b1;   // RO
                    default: err_set = 1'b1;
                    endcase
                end
            end

            // ---- 2. W1C masks -------------------------------------------
            w1c_f = 11'd0;
            w1c_i = 11'd0;
            if (cs_rise && frame_full && (wa[1:0] == 2'b00) && (wa[14:6] == 9'd0)) begin
                if (wa[5:2] == A_FAULT) w1c_f = sh[10:0];
                if (wa[5:2] == A_INT)   w1c_i = sh[10:0];
            end

            // ---- 3. next visible fault / interrupt state -----------------
            n_fault = fault_status | hw_fault | (err_set ? 11'h100 : 11'd0);
            if (w1c_f != 11'd0) n_fault = n_fault & ~(w1c_f & ~hw_fault);
            if (err_set)        n_fault[8] = 1'b1;

            n_int = int_status | (n_fault & ~fs_d);
            if (w1c_i != 11'd0) n_int = n_int & ~w1c_i;

            fault_status <= n_fault;
            int_status   <= n_int;
            fs_d         <= n_fault;
            if (err_set) err_reg <= 1'b1;
            else if (w1c_f[8]) err_reg <= 1'b0;

            // ---- 4. read snapshot (only after the qualifier is seen) -----
            if (rq_s2) rd_snapshot <= rval;
        end
    end

endmodule
