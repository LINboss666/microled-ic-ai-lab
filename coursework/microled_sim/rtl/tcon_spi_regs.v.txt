`timescale 1ns / 1ps
// tcon_spi_regs — TCON 32-bit register file + SPI->core CDC (T6-A rev2).
//
// V1.3 §13 (段0569-0570, 0704):
//   * all registers are 32 bit, 4-byte aligned; reserved bits read 0 / write 0;
//   * unimplemented or misaligned address: read 0, write ignored, ERR_REG set;
//   * timing / mode / current / LUT configuration may only be written when
//     EN = 0 and SAFE = 1; a violation sets ERR_REG and is NOT committed;
//   * BLANK assertion and EN clearing take effect at any time;
//   * protection (RESET / hardware FAULT / BLANK) outranks SPI and the interrupt
//     mask; INT_EN only masks the notification.
//
// ===========================================================================
// WHAT CHANGED IN rev2 (T6-A RUN01 复审, 2026-11-19)
//   * a READ transaction can no longer reach the write path at all: writes only
//     arrive through the mailbox epoch, and only frames with command bit15==0
//     ever flip that epoch. The W1C paths therefore cannot be triggered by a
//     read either - not by a qualifier, by construction.
//   * the 32-bit read answer is latched exactly ONCE per read request (edge of
//     the synchronised rd_req) instead of being refreshed every core clock, so
//     a live register (STATUS / COL_INDEX) cannot tear in the middle of the
//     32 data bits.
//   * CTRL: clearing EN forces BLANK; unblanking requires SAFE && READY && no
//     hardware fault, and is decided on the NEW en value, not the old one.
// ===========================================================================
module tcon_spi_regs (
    input  wire        clk,          // 100 MHz core clock
    input  wire        reset_n,

    // ---- write mailbox (cs_n domain, see spi_frame) -----------------------
    input  wire [47:0] wr_frame,     // stable 48-bit frame of the last write
    input  wire        wr_tog,       // flips once per complete write frame
    output wire        wr_ack_tog,   // this domain's consumed epoch

    // ---- read request (SCLK domain) ---------------------------------------
    input  wire [14:0] rd_addr_hold, // frozen from the 16th rising edge
    input  wire        rd_req,       // level for the 32 data bits
    output reg  [31:0] rd_snapshot,

    // ---- real hardware status / faults (NOT constants) --------------------
    input  wire [3:0]  status_in,    // {LUT_VALID, SAFE, ACTIVE, READY}
    input  wire [10:0] hw_fault,     // UVLO,OV,OT,CLOCK_LOST,FRAME_TO,VID_TIMING,
                                     // UF_OF,COMM_TO,(ERR_REG internal),LUT_ERR,SYNC_ERR
    input  wire        test_en,      // chip TEST_EN pin (not a new SPI pin)
    input  wire [9:0]  col_index_in, // live TCON display column

    // ---- register outputs to the rest of the chip -------------------------
    output reg         en,
    output reg         blank_r,      // software BLANK request
    output wire        hw_blank,     // what the panel really sees
    output reg         test_mode,
    output wire        sw_reset_pulse,
    output reg         fps_sel,
    output reg  [15:0] wdog,
    output reg  [10:0] fault_status,
    output reg  [10:0] int_status,
    output reg  [10:0] int_en,
    output wire        irq,
    output reg         err_reg,

    // ---- observation ------------------------------------------------------
    output reg  [15:0] wr_cap_cnt,   // write frames captured from the mailbox
    output wire        wr_cap_o,     // 1-cycle pulse per captured write frame
    output reg         wr_ok_o,      // 1-cycle pulse per ACCEPTED write
    output reg  [15:0] wr_rej_cnt,   // rejected writes
    output wire        rd_cap_o,     // 1-cycle pulse per latched read snapshot
    output wire [10:0] fault_vis_o,

    // ---- safety observation (no functional effect) -------------------------
    output wire        disp_armed_o,      // host re-enable permit
    output wire        hw_fault_caught_o  // asynchronous fault capture latch
);

    // ---------------------------------------------------------------------
    // register addresses (addr[5:2]); 0x00..0x24 of the TCON subset
    // ---------------------------------------------------------------------
    localparam [3:0] A_CHIPID = 4'h0, A_CTRL  = 4'h1, A_MODE  = 4'h2,
                     A_FAULT  = 4'h3, A_INT   = 4'h4, A_INTEN = 4'h5,
                     A_STATUS = 4'h6, A_WDOG  = 4'h7, A_VIDCFG= 4'h8,
                     A_COLIDX = 4'h9;

    localparam [7:0] CHIP_ID_TCON = 8'h10;   // V1.3 §13 reset column 10/21/22/31/32h

    // ---------------------------------------------------------------------
    // write mailbox epoch: 2 FF synchroniser + consumed-epoch tracking
    //   wr_cap is a single-cycle pulse per write frame - never per SCLK edge.
    // ---------------------------------------------------------------------
    reg tg_s1, tg_s2, tg_seen;
    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            tg_s1 <= 1'b0; tg_s2 <= 1'b0; tg_seen <= 1'b0;
        end else begin
            tg_s1 <= wr_tog;
            tg_s2 <= tg_s1;
            if (tg_s2 != tg_seen) tg_seen <= tg_s2;
        end
    end
    assign wr_ack_tog = tg_seen;
    assign wr_cap_o   = (tg_s2 != tg_seen);

    // ---------------------------------------------------------------------
    // read request: 2 FF + falling-edge-free single capture
    // ---------------------------------------------------------------------
    reg rq_s1, rq_s2, rq_s3;
    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin rq_s1 <= 1'b0; rq_s2 <= 1'b0; rq_s3 <= 1'b0; end
        else          begin rq_s1 <= rd_req; rq_s2 <= rq_s1; rq_s3 <= rq_s2; end
    end
    assign rd_cap_o = rq_s2 && !rq_s3;

    // ---------------------------------------------------------------------
    // register read mux (combinational; rd_addr_hold is frozen while in use)
    // ---------------------------------------------------------------------
    wire [3:0] ra = rd_addr_hold[5:2];
    wire       ra_ok = (rd_addr_hold[14:6] == 9'd0) && (rd_addr_hold[1:0] == 2'd0);

    reg [31:0] rval;
    always @(*) begin
        rval = 32'h0000_0000;
        if (ra_ok) begin
            case (ra)
            A_CHIPID: rval = {24'd0, CHIP_ID_TCON};
            A_CTRL  : rval = {28'd0, test_mode, 1'b0, blank_r, en};  // SW_RESET reads 0
            A_MODE  : rval = {31'd0, fps_sel};
            A_FAULT : rval = {21'd0, fault_status};
            A_INT   : rval = {21'd0, int_status};
            A_INTEN : rval = {21'd0, int_en};
            A_STATUS: rval = {28'd0, status_in};
            A_WDOG  : rval = {16'd0, wdog};
            A_VIDCFG: rval = 32'h000B_FFFF;                          // 1023 / 767
            A_COLIDX: rval = {22'd0, col_index_in};
            default : rval = 32'h0000_0000;
            endcase
        end
    end

    // ---------------------------------------------------------------------
    // state + write commit
    // ---------------------------------------------------------------------
    reg        sw_pulse;
    reg        disp_armed;      // host permit: set only by an explicit unblank
    reg        hw_fault_caught; // asynchronous fault capture
    reg [10:0] fs_d;
    reg [10:0] n_fault, n_int;

    wire safe_in  = status_in[2];
    wire ready_in = status_in[0];
    wire cfg_ok   = (!en) && safe_in;
    wire any_hw_fault = |hw_fault;
    // every latched fault EXCEPT ERR_REG (bit8, a register-access error that
    // must not blank the panel) is a display fault and survives its source
    localparam [10:0] DISP_FAULT_MASK = 11'h7FF & ~11'h100;
    wire disp_fault = |(fault_status & DISP_FAULT_MASK);

    // the frame fields, stable for the whole CS_n transaction
    wire [14:0] wa_w = wr_frame[46:32];
    wire [31:0] wd_w = wr_frame[31:0];

    // ---------------------------------------------------------------------
    // P0-D: the clear request is COMBINATIONAL, not a registered pulse.
    //   rev3 registered `clr_fault` inside the write block, so the capture
    //   latch - a different always block - sampled its OLD value on that
    //   edge and only cleared on the NEXT one, while `n_fault` re-ORed the
    //   still-set capture back into bit7 on that very next edge.  The fault
    //   could then never be cleared and the host could never re-enable.
    //   Driving the latch from this request removes the register entirely:
    //   `fault_status[7]` and `hw_fault_caught` fall on the SAME edge, from
    //   the SAME accepted write.
    // ---------------------------------------------------------------------
    wire clear_caught_req = wr_cap_o &&
                            (wa_w[1:0]  == 2'b00) &&
                            (wa_w[14:6] == 9'd0)  &&
                            (wa_w[5:2]  == A_FAULT) &&
                            wr_frame[7]           &&   // W1C covers COMM_TO
                            !any_hw_fault;             // a live source wins

    // ---------------------------------------------------------------------
    // P0-B: asynchronous fault capture
    //   hw_fault is a LEVEL from the analog comparators / the wired-AND
    //   FAULT_n line. Sampling it only on the core clock loses any pulse
    //   shorter than one period (a 4 ns glitch between two 100 MHz edges never
    //   reaches the FSM). The SET path is therefore the assertion edge itself;
    //   the CLEAR is synchronous, so a dead clock can never clear it.
    // ---------------------------------------------------------------------
    always @(posedge clk or posedge any_hw_fault or negedge reset_n) begin
        if (!reset_n)          hw_fault_caught <= 1'b0;
        else if (any_hw_fault) hw_fault_caught <= 1'b1;
        else if (clear_caught_req) hw_fault_caught <= 1'b0;
    end

    assign sw_reset_pulse = sw_pulse;
    assign irq            = |(int_status & int_en);
    assign fault_vis_o    = fault_status | hw_fault |
                            (hw_fault_caught ? 11'h080 : 11'd0);
    // no enable, software blank, or any hardware fault => the panel is blank.
    // Software cannot talk its way out of this: it is a pure OR at the output.
    // six independent locks, any one of them blanks the panel:
    //   software BLANK, EN cleared, a LIVE fault, the asynchronously caught
    //   fault, the host permit latch and the sticky fault history. The last
    //   three stop the panel from coming back on its own.
    assign hw_blank       = blank_r | (!en) | any_hw_fault | hw_fault_caught |
                            (!disp_armed) | disp_fault;
    assign disp_armed_o      = disp_armed;
    assign hw_fault_caught_o = hw_fault_caught;

    integer k;
    reg [14:0] wa;
    reg [31:0] wd;
    reg        err_set;
    reg        accept;
    reg [10:0] w1c_f, w1c_i;

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            en           <= 1'b0;
            blank_r      <= 1'b1;               // reset value 0x00000002
            test_mode    <= 1'b0;
            sw_pulse     <= 1'b0;
            fps_sel      <= 1'b0;
            wdog         <= 16'd2;
            fault_status <= 11'd0;
            int_status   <= 11'd0;
            int_en       <= 11'd0;
            err_reg      <= 1'b0;
            rd_snapshot  <= 32'd0;
            fs_d         <= 11'd0;
            wr_cap_cnt   <= 16'd0;
            wr_rej_cnt   <= 16'd0;
            wr_ok_o      <= 1'b0;
            disp_armed   <= 1'b0;
        end else begin
            err_set  = 1'b0;
            accept   = 1'b0;
            sw_pulse <= 1'b0;
            wr_ok_o  <= 1'b0;

            // ---- 1. write commit: one pulse per captured write frame -------
            if (wr_cap_o) begin
                wa = wa_w;
                wd = wd_w;
                wr_cap_cnt <= wr_cap_cnt + 16'd1;
                if (wa[1:0] != 2'b00) begin
                    err_set = 1'b1;                       // not 4-byte aligned
                end else if (wa[14:6] != 9'd0) begin
                    err_set = 1'b1;                       // not implemented
                end else begin
                    case (wa[5:2])
                    A_CHIPID: err_set = 1'b1;             // RO
                    A_CTRL: begin
                        if (wd[3] && !test_en)                          err_set = 1'b1;
                        else if (wd[0] && (!safe_in || any_hw_fault))    err_set = 1'b1;
                        else if (wd[0] && !wd[1] &&
                                 !(safe_in && ready_in && !any_hw_fault &&
                                   !hw_fault_caught && !disp_fault)) err_set = 1'b1;
                        else begin
                            en        <= wd[0];
                            // EN=0 forces BLANK, whatever bit1 asks for
                            blank_r   <= wd[0] ? wd[1] : 1'b1;
                            test_mode <= wd[3];
                            // re-arm ONLY on an explicit unblank that passed
                            // every check; EN=0 or BLANK=1 clears the permit
                            disp_armed <= wd[0] && !wd[1];
                            if (wd[2]) sw_pulse <= 1'b1;     // W1P, self clearing
                            accept = 1'b1;
                        end
                    end
                    A_MODE: begin
                        if (cfg_ok) begin fps_sel <= wd[0]; accept = 1'b1; end
                        else              err_set = 1'b1;
                    end
                    A_WDOG: begin
                        if (!cfg_ok)                            err_set = 1'b1;
                        else if ((wd[15:0] == 16'd0) && !test_en) err_set = 1'b1;
                        else begin wdog <= wd[15:0]; accept = 1'b1; end
                    end
                    A_FAULT: accept = 1'b1;                  // W1C below
                    A_INT:   accept = 1'b1;                  // W1C below
                    A_INTEN: begin int_en <= wd[10:0]; accept = 1'b1; end
                    A_STATUS, A_VIDCFG, A_COLIDX: err_set = 1'b1;   // RO
                    default: err_set = 1'b1;
                    endcase
                end
                if (err_set) wr_rej_cnt <= wr_rej_cnt + 16'd1;
                if (accept)  wr_ok_o    <= 1'b1;
            end

            // ---- 2. W1C masks (reachable only from a captured WRITE frame) -
            w1c_f = 11'd0;
            w1c_i = 11'd0;
            if (wr_cap_o && (wa[1:0] == 2'b00) && (wa[14:6] == 9'd0)) begin
                if (wa[5:2] == A_FAULT) w1c_f = wd[10:0];
                if (wa[5:2] == A_INT)   w1c_i = wd[10:0];
            end

            // ---- 3. next visible fault / interrupt state -------------------
            // the asynchronously captured event is reported as COMM_TO
            // (V1.3 段0704: TCON catching the shared FAULT_n sets COMM_TO).
            // Without this the panel would go dark and the host would read 0.
            n_fault = fault_status | hw_fault | (err_set ? 11'h100 : 11'd0) |
                      (hw_fault_caught ? 11'h080 : 11'd0);
            if (w1c_f != 11'd0) n_fault = n_fault & ~(w1c_f & ~hw_fault);
            if (err_set)        n_fault[8] = 1'b1;

            n_int = int_status | (n_fault & ~fs_d);
            if (w1c_i != 11'd0) n_int = n_int & ~w1c_i;

            fault_status <= n_fault;
            int_status   <= n_int;
            fs_d         <= n_fault;
            if (err_set) err_reg <= 1'b1;
            else if (w1c_f[8]) err_reg <= 1'b0;

            // ---- 4. read snapshot: ONE latch per request -------------------
            if (rd_cap_o) rd_snapshot <= rval;

            // ---- 5. capture-latch clear -----------------------------------
            // Handled by `clear_caught_req` at the capture latch itself (see
            // the P0-D block above), so that the status bit and the capture
            // fall on the same edge.  Nothing is registered here on purpose:
            // a registered pulse reintroduces the one-cycle resurrection.

            // ---- 6. a live, caught or latched fault always wins -----------
            if (any_hw_fault || hw_fault_caught || disp_fault) disp_armed <= 1'b0;
        end
    end

endmodule
