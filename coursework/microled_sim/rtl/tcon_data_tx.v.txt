`timescale 1ns / 1ps
// TCON data transmitter (T2A) — 384 bytes per Data port per load window.
//
// Cycle map for the loading column N (see rtl/T2A_INTERFACE_CONTRACT.md §2/§3):
//   cycle 831, falling edge   : PRE-FETCH byte 0 onto D0/D1
//   cycle 832 .. 1599         : load_active high (768 cycles = 7.68 us)
//   DCLK rising at 832,834,.. : 384 edges; edge k samples byte k
//   D0 = even byte 2k (row 2k), D1 = odd byte 2k+1 (row 2k+1)
//
// DCLK is derived as `load_run && !load_cnt[0]`, so the FIRST rising edge is
// hard-anchored at cycle 832 regardless of the slot length (976 or 977).
// Data is launched on the falling edge of CLK_REF, which gives 5 ns setup and
// 5 ns hold around every sampling edge (V1.3 targets: 3 ns / 1 ns).
//
// KNOWN DEVIATION: a real Data interface normally receives a continuously
// running DCLK. Here DCLK only toggles inside the load window so that the first
// sampling edge is deterministic and testable. Documented, not hidden.
//
// If the selected bank is not valid the whole window transmits 8'h00 (the
// column stays dark) and the frame time base is NOT disturbed.
module tcon_data_tx #(
    parameter integer LOAD_START = 832,
    parameter integer LOAD_LEN   = 768,
    parameter integer NB_PORT    = 384
) (
    input  wire        clk,
    input  wire        reset_n,
    input  wire        safe,

    input  wire [10:0] slot_cycle,
    input  wire [9:0]  col_index,
    input  wire        col_valid,   // bank holds THIS column and is valid

    // bank read port
    output wire        rd_bank,
    output wire [8:0]  rd_idx,
    input  wire [7:0]  bk_d0,
    input  wire [7:0]  bk_d1,

    // Data interface
    output wire        load_active,
    output wire        dclk,
    output reg  [7:0]  d0,
    output reg  [7:0]  d1,

    // observability
    output reg  [8:0]  bytes_sent,   // 1..384
    output reg         load_blank    // loaded column had no valid bank
);

    // start_pt is high during cycle LOAD_START-1 (=831)
    wire start_pt = (slot_cycle == (LOAD_START - 1));

    reg        load_run;
    reg [9:0]  load_cnt;
    reg        parity_q;    // bank latched at start (col_index changes mid-window)
    reg        valid_q;
    reg [8:0]  tx_k;

    // ------------------------------------------------------------------
    // window counter (posedge domain)
    // ------------------------------------------------------------------
    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            load_run <= 1'b0;
            load_cnt <= 10'd0;
        end else if (!safe) begin
            load_run <= 1'b0;
            load_cnt <= 10'd0;
        end else if (start_pt) begin
            load_run <= 1'b1;
            load_cnt <= 10'd0;
        end else if (load_run) begin
            if (load_cnt == (LOAD_LEN - 1)) begin
                load_run <= 1'b0;
                load_cnt <= 10'd0;
            end else begin
                load_cnt <= load_cnt + 10'd1;
            end
        end
    end

    assign load_active = load_run && safe;
    assign dclk        = load_run && !load_cnt[0] && safe;

    // ------------------------------------------------------------------
    // byte launch (negedge domain: 5 ns before every DCLK sampling edge)
    // ------------------------------------------------------------------
    assign rd_bank = start_pt ? col_index[0] : parity_q;
    assign rd_idx  = start_pt ? 9'd0
                              : ((tx_k == (NB_PORT - 1)) ? (NB_PORT - 1) : (tx_k + 9'd1));

    always @(negedge clk or negedge reset_n) begin
        if (!reset_n) begin
            tx_k       <= 9'd0;
            d0         <= 8'h00;
            d1         <= 8'h00;
            parity_q   <= 1'b0;
            valid_q    <= 1'b0;
            bytes_sent <= 9'd0;
            load_blank <= 1'b0;
        end else if (!safe) begin
            tx_k       <= 9'd0;
            d0         <= 8'h00;
            d1         <= 8'h00;
            bytes_sent <= 9'd0;
            load_blank <= 1'b1;
        end else if (start_pt) begin
            parity_q   <= col_index[0];
            valid_q    <= col_valid;
            load_blank <= !col_valid;
            tx_k       <= 9'd0;
            bytes_sent <= 9'd1;
            d0         <= col_valid ? bk_d0 : 8'h00;
            d1         <= col_valid ? bk_d1 : 8'h00;
        end else if (load_run && load_cnt[0] && (tx_k != (NB_PORT - 1))) begin
            tx_k       <= tx_k + 9'd1;
            bytes_sent <= tx_k + 9'd2;
            d0         <= valid_q ? bk_d0 : 8'h00;
            d1         <= valid_q ? bk_d1 : 8'h00;
        end
    end

endmodule
