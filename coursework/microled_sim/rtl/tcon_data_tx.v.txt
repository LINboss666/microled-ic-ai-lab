`timescale 1ns / 1ps
// TCON data transmitter (T2A/T2AR) — 384 bytes per Data port per load window.
//
// DCLK is a CONTINUOUS 50 MHz clock (100 MHz / 2). T2A gated DCLK to the load
// window to make the first sampling edge deterministic; that was an unnecessary
// deviation from the real Data interface and it also invalidated the T1
// continuous-DCLK assertion. T2AR removes it: DCLK runs all the time and the
// byte launch is aligned to the DCLK phase instead.
//
// PHASE HANDLING (why the first edge is 832 OR 833)
//   A 768-cycle window always contains exactly 384 DCLK rising edges, whatever
//   the phase. Slot lengths alternate 976/977, so the absolute parity of cycle
//   832 flips once per long slot; the first rising edge inside the window is
//   therefore at 832 or at 833, and both cases still give 384 edges.
//   The byte-0 pre-fetch happens on the CLK_REF falling edge whose NEXT rising
//   edge is that first sampling edge, i.e. on the falling edge of cycle 831
//   (DCLK low during 831) or of cycle 832 (DCLK high during 831).
//   Every later byte is launched on the falling edges where DCLK is low, which
//   is exactly one launch per sampling edge: 1 + 383 = 384.
//
// SETUP / HOLD: launching on the CLK_REF falling edge gives 5 ns / 5 ns around
// every sampling edge in this ideal RTL phase model. That is a *model* result,
// not a static timing sign-off (STA stays NOT_RUN).
module tcon_data_tx #(
    parameter integer LOAD_START = 832,
    parameter integer LOAD_LEN   = 768,
    parameter integer NB_PORT    = 384
) (
    input  wire        clk,
    input  wire        reset_n,
    input  wire        safe,

    input  wire [10:0] slot_cycle,
    input  wire [9:0]  col_index,
    input  wire        col_valid,   // bank holds THIS column and is valid

    // bank read port
    output wire        rd_bank,
    output wire [8:0]  rd_idx,
    input  wire [7:0]  bk_d0,
    input  wire [7:0]  bk_d1,

    // Data interface
    output wire        load_active,
    output wire        dclk,          // continuous 50 MHz
    output reg  [7:0]  d0,
    output reg  [7:0]  d1,

    // observability
    output reg  [8:0]  bytes_sent,    // 1..384
    output reg         load_blank,    // loaded column had no valid bank
    output wire [9:0]  load_cnt_o
);

    // ------------------------------------------------------------------
    // continuous 50 MHz DCLK
    // ------------------------------------------------------------------
    reg dclk_r;
    always @(posedge clk or negedge reset_n)
        if (!reset_n) dclk_r <= 1'b0;
        else          dclk_r <= ~dclk_r;
    assign dclk = dclk_r;

    // ------------------------------------------------------------------
    // window counter
    // ------------------------------------------------------------------
    wire start_pt = (slot_cycle == (LOAD_START - 1));   // cycle 831

    reg        load_run;
    reg [9:0]  load_cnt;
    reg        parity_q;
    reg        valid_q;
    reg [8:0]  tx_k;
    reg        prefetched;
    reg [8:0]  launched;

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            load_run <= 1'b0;
            load_cnt <= 10'd0;
        end else if (!safe) begin
            load_run <= 1'b0;
            load_cnt <= 10'd0;
        end else if (start_pt) begin
            load_run <= 1'b1;
            load_cnt <= 10'd0;
        end else if (load_run) begin
            if (load_cnt == (LOAD_LEN - 1)) begin
                load_run <= 1'b0;
                load_cnt <= 10'd0;
            end else begin
                load_cnt <= load_cnt + 10'd1;
            end
        end
    end

    assign load_active = load_run && safe;
    assign load_cnt_o  = load_cnt;

    // ------------------------------------------------------------------
    // pre-fetch point: the falling edge that precedes the first sampling edge
    // ------------------------------------------------------------------
    wire pre_pt = ((slot_cycle == (LOAD_START - 1)) ||
                   (slot_cycle ==  LOAD_START)) && !dclk_r;
    wire do_pre = pre_pt && !prefetched;

    assign rd_bank = do_pre ? col_index[0] : parity_q;
    assign rd_idx  = do_pre ? 9'd0
                            : ((tx_k == (NB_PORT - 1)) ? (NB_PORT - 1) : (tx_k + 9'd1));

    // ------------------------------------------------------------------
    // byte launch (falling-edge domain: 5 ns before every sampling edge)
    // ------------------------------------------------------------------
    localparam [8:0] NB_LAST = NB_PORT - 1;

    always @(negedge clk or negedge reset_n) begin
        if (!reset_n) begin
            tx_k       <= 9'd0;
            d0         <= 8'h00;
            d1         <= 8'h00;
            parity_q   <= 1'b0;
            valid_q    <= 1'b0;
            bytes_sent <= 9'd0;
            load_blank <= 1'b0;
            prefetched <= 1'b0;
            launched   <= 9'd0;
        end else if (!safe) begin
            tx_k       <= 9'd0;
            d0         <= 8'h00;
            d1         <= 8'h00;
            bytes_sent <= 9'd0;
            load_blank <= 1'b1;
            prefetched <= 1'b0;
            launched   <= 9'd0;
        end else if (slot_cycle == (LOAD_START - 2)) begin
            // one cycle before the pre-fetch window: arm a fresh transfer
            prefetched <= 1'b0;
            launched   <= 9'd0;
            tx_k       <= 9'd0;
        end else if (do_pre) begin
            prefetched <= 1'b1;
            parity_q   <= col_index[0];
            valid_q    <= col_valid;
            load_blank <= !col_valid;
            tx_k       <= 9'd0;
            launched   <= 9'd1;
            bytes_sent <= 9'd1;
            d0         <= col_valid ? bk_d0 : 8'h00;
            d1         <= col_valid ? bk_d1 : 8'h00;
        end else if (load_run && !dclk_r && prefetched && (launched <= NB_LAST) &&
                     (tx_k != NB_LAST)) begin
            tx_k       <= tx_k + 9'd1;
            launched   <= launched + 9'd1;
            bytes_sent <= launched + 9'd1;
            d0         <= valid_q ? bk_d0 : 8'h00;
            d1         <= valid_q ? bk_d1 : 8'h00;
        end
    end

endmodule
