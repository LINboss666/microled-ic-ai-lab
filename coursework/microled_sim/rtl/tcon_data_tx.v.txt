`timescale 1ns / 1ps
// TCON data transmitter — 384 bytes per Data port per load window (T2B).
//
// DCLK is a CONTINUOUS 50 MHz clock (100 MHz / 2). The byte launch is aligned to
// the DCLK phase: byte 0 is pre-fetched on the CLK_REF falling edge whose next
// rising edge is the first sampling edge (cycle 831 or 832), every later byte on
// the following falling edges with DCLK low, so 1 + 383 = 384 launches and every
// launch has 5 ns setup / 15 ns hold around its sampling edge in this ideal RTL
// phase model (STA stays NOT_RUN).
//
// T2B: the transmitter also publishes the label of what it just forwarded
// (`fwd_col`, `fwd_frame`, `fwd_poison`, `fwd_valid`) and a one-cycle
// `load_done` when the 768-cycle window closes. tcon_top uses those to track the
// label that the Data display bank will hold after LATCH - it never computes
// "s-1" by hand, because precharge, a missed LATCH, a frame wrap or a fault all
// make that formula wrong.
//
// `fault` is the latched safe state from tcon_top: it closes the window and
// forces the payload to zero, so a mixed (partly received) column cannot keep
// being clocked out.
module tcon_data_tx #(
    parameter integer LOAD_START = 832,
    parameter integer LOAD_LEN   = 768,
    parameter integer NB_PORT    = 384
) (
    input  wire        clk,
    input  wire        reset_n,
    input  wire        safe,
    input  wire        fault,

    input  wire [10:0] slot_cycle,
    input  wire [9:0]  col_index,
    input  wire        col_valid,   // bank holds THIS column and it is clean
    input  wire        col_poison,
    input  wire [5:0]  frame_in,

    // bank read port
    output wire        rd_bank,
    output wire [8:0]  rd_idx,
    input  wire [7:0]  bk_d0,
    input  wire [7:0]  bk_d1,

    // Data interface
    output wire        load_active,
    output wire        dclk,
    output reg  [7:0]  d0,
    output reg  [7:0]  d1,

    // forward label (the column this window is carrying)
    output reg  [9:0]  fwd_col,
    output reg  [5:0]  fwd_frame,
    output reg         fwd_poison,
    output reg         fwd_valid,
    output reg         load_done,

    // observability
    output reg  [8:0]  bytes_sent,
    output reg         load_blank,
    output wire [9:0]  load_cnt_o
);

    // ------------------------------------------------------------------
    // continuous 50 MHz DCLK
    // ------------------------------------------------------------------
    reg dclk_r;
    always @(posedge clk or negedge reset_n)
        if (!reset_n) dclk_r <= 1'b0;
        else          dclk_r <= ~dclk_r;
    assign dclk = dclk_r;

    // ------------------------------------------------------------------
    // window counter
    // ------------------------------------------------------------------
    wire start_pt = (slot_cycle == (LOAD_START - 1));   // cycle 831

    reg        load_run;
    reg [9:0]  load_cnt;
    reg        parity_q;
    reg        valid_q;
    reg        poisoned_q;
    reg [8:0]  tx_k;
    reg        prefetched;
    reg [8:0]  launched;

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            load_run  <= 1'b0;
            load_cnt  <= 10'd0;
            load_done <= 1'b0;
        end else if (!safe || fault) begin
            load_run  <= 1'b0;
            load_cnt  <= 10'd0;
            load_done <= 1'b0;
        end else if (start_pt) begin
            load_run  <= 1'b1;
            load_cnt  <= 10'd0;
            load_done <= 1'b0;
        end else if (load_run) begin
            // one-cycle pulse on the cycle that closes the 768-cycle window
            load_done <= (load_cnt == (LOAD_LEN - 1));
            if (load_cnt == (LOAD_LEN - 1)) begin
                load_run <= 1'b0;
                load_cnt <= 10'd0;
            end else begin
                load_cnt <= load_cnt + 10'd1;
            end
        end else begin
            load_done <= 1'b0;
        end
    end

    assign load_active = load_run && safe && !fault;
    assign load_cnt_o  = load_cnt;

    // ------------------------------------------------------------------
    // pre-fetch point: the falling edge that precedes the first sampling edge
    // ------------------------------------------------------------------
    wire pre_pt = ((slot_cycle == (LOAD_START - 1)) ||
                   (slot_cycle ==  LOAD_START)) && !dclk_r;
    wire do_pre = pre_pt && !prefetched;

    assign rd_bank = do_pre ? col_index[0] : parity_q;
    assign rd_idx  = do_pre ? 9'd0
                            : ((tx_k == (NB_PORT - 1)) ? (NB_PORT - 1) : (tx_k + 9'd1));

    // ------------------------------------------------------------------
    // byte launch (falling-edge domain: 5 ns before every sampling edge)
    // ------------------------------------------------------------------
    localparam [8:0] NB_LAST = NB_PORT - 1;

    always @(negedge clk or negedge reset_n) begin
        if (!reset_n) begin
            tx_k       <= 9'd0;
            d0         <= 8'h00;
            d1         <= 8'h00;
            parity_q   <= 1'b0;
            valid_q    <= 1'b0;
            poisoned_q <= 1'b0;
            bytes_sent <= 9'd0;
            load_blank <= 1'b0;
            prefetched <= 1'b0;
            launched   <= 9'd0;
            fwd_col    <= 10'd0;
            fwd_frame  <= 6'd0;
            fwd_poison <= 1'b0;
            fwd_valid  <= 1'b0;
        end else if (!safe || fault) begin
            tx_k       <= 9'd0;
            d0         <= 8'h00;
            d1         <= 8'h00;
            bytes_sent <= 9'd0;
            load_blank <= 1'b1;
            prefetched <= 1'b0;
            launched   <= 9'd0;
        end else if (slot_cycle == (LOAD_START - 2)) begin
            // one cycle before the pre-fetch window: arm a fresh transfer
            prefetched <= 1'b0;
            launched   <= 9'd0;
            tx_k       <= 9'd0;
        end else if (do_pre) begin
            prefetched <= 1'b1;
            parity_q   <= col_index[0];
            valid_q    <= col_valid;
            poisoned_q <= col_poison;
            load_blank <= !col_valid;
            tx_k       <= 9'd0;
            launched   <= 9'd1;
            bytes_sent <= 9'd1;
            d0         <= col_valid ? bk_d0 : 8'h00;
            d1         <= col_valid ? bk_d1 : 8'h00;
            // label of the column this window is carrying
            fwd_col    <= col_index;
            fwd_frame  <= frame_in;
            fwd_poison <= col_poison;
            fwd_valid  <= col_valid;
        end else if (load_run && !dclk_r && prefetched && (launched <= NB_LAST) &&
                     (tx_k != NB_LAST)) begin
            tx_k       <= tx_k + 9'd1;
            launched   <= launched + 9'd1;
            bytes_sent <= launched + 9'd1;
            d0         <= valid_q ? bk_d0 : 8'h00;
            d1         <= valid_q ? bk_d1 : 8'h00;
            load_blank <= !valid_q;
        end
    end

endmodule
