`timescale 1ns / 1ps
// data_pin_mon — pin-faithful Data-port monitor (T2B-BR).
//
// WHY THIS EXISTS (T2B-B independent review, finding B):
//   data_model used to detect DCLK edges from the 100 MHz reference clock
//   (`posedge clk` + a registered dclk_d). That works in a synchronous ideal
//   world but it is NOT the V1.3 interface: V1.3 says the Data driver samples
//   DIN on the DCLK rising edge (摘录 0447-0466 / 0716-0717). This monitor uses
//   **only pins**: posedge dclk, the latch level, and din. There is no reference
//   clock and no use of the DUT's internal slot_cycle.
//
// PHASE RULE (validated offline against an archived real waveform, see
// t2b/PIN_MONITOR_OFFLINE_REPLAY.md):
//   * latch_d is only updated ON a dclk rising edge, so a LATCH edge is
//     "seen" by the Data chip at the first sampling edge that carries it;
//   * the falling LATCH edge arms the window AND takes byte 0 at that same
//     edge (the first sampling edge of the transfer);
//   * 384 edges later the buffer is complete; the next rising LATCH edge
//     closes the window and is compared.
// The 1 ns `#1` after the DCLK edge is a delta-cycle settle so that LATCH /
// DIN (both change in the same simulation instant as the edge) are read at
// their post-edge value. It is NOT a second clock: nothing is sampled on any
// other event.
//
// CONTENT RULE: the payload is self-identifying. byte0 = col[7:0],
// byte1 = {col[9:8], frame[5:0]}, byte i (i>=2) = pix(frame, col, 2i+CHIP).
// The monitor decodes (frame, col) from the bytes it captured itself and then
// checks the other 382 against the formula, so a torn / zero-filled column
// shows up in `byte_bad` instead of being silently accepted.
module data_pin_mon #(
    parameter integer CHIP = 0
) (
    input  wire        reset_n,
    input  wire        dclk,
    input  wire [7:0]  din,
    input  wire        latch,

    // observation only - not a chip interface
    output reg  [15:0] n_win,        // completed load windows
    output reg  [15:0] n_win_badlen, // completed windows whose byte count != 384
    output reg  [15:0] n_byte_bad,   // payload bytes that disagree with the tags
    output reg  [15:0] n_edge,       // dclk rising edges captured
    output reg  [9:0]  last_bytes,   // bytes in the most recent window
    output reg  [9:0]  last_col,
    output reg  [5:0]  last_frame,
    output reg         any_x         // X/Z ever seen on din
);

    reg [7:0] cap [0:383];
    reg       armed;
    reg       latch_d;
    integer   wcnt;
    integer   i;
    reg [7:0] exp;
    reg [9:0] col;
    reg [5:0] frm;

    function automatic [7:0] pix(input integer f, input integer c, input integer r);
        pix = ((c & 255) + 13 * (r & 255) + 7 * (f & 255) + 8'h5A) & 255;
    endfunction

    initial begin
        n_win = 16'd0; n_win_badlen = 16'd0; n_byte_bad = 16'd0; n_edge = 16'd0;
        last_bytes = 10'd0; last_col = 10'd0; last_frame = 6'd0; any_x = 1'b0;
        armed = 1'b0; latch_d = 1'b0; wcnt = 0;
        for (i = 0; i < 384; i = i + 1) cap[i] = 8'h00;
    end

    always @(posedge dclk or negedge reset_n) begin
        #1;
        if (!reset_n) begin
            armed     = 1'b0;
            latch_d   = 1'b0;
            wcnt      = 0;
            n_win     = 16'd0;
            n_win_badlen = 16'd0;
            n_byte_bad   = 16'd0;
            n_edge    = 16'd0;
            last_bytes = 10'd0;
            last_col   = 10'd0;
            last_frame = 6'd0;
            any_x      = 1'b0;
        end else begin
            // ---- LATCH edges, seen on the DCLK grid ----------------------
            if (latch && !latch_d) begin
                // rising: the Data chip commits what it captured
                if (armed) begin
                    n_win      = n_win + 16'd1;
                    last_bytes = wcnt;
                    if (wcnt != 384) begin
                        n_win_badlen = n_win_badlen + 16'd1;
                    end else begin
                        col = {cap[1][7:6], cap[0]};
                        frm = cap[1][5:0];
                        last_col   = col;
                        last_frame = frm;
                        for (i = 2; i < 384; i = i + 1) begin
                            exp = pix(frm, col, (i * 2) + CHIP);
                            if (cap[i] !== exp) n_byte_bad = n_byte_bad + 16'd1;
                        end
                    end
                end
                armed = 1'b0;
                wcnt  = 0;
            end else if (!latch && latch_d) begin
                // falling: a new 384-byte transfer starts at this very edge
                armed = 1'b1;
                wcnt  = 0;
            end

            // ---- capture on the real DCLK rising edge --------------------
            if (armed && (wcnt < 384)) begin
                cap[wcnt] = din;
                wcnt      = wcnt + 1;
                n_edge    = n_edge + 16'd1;
                if (^din === 1'bx) any_x = 1'b1;
            end

            latch_d = latch;
        end
    end

endmodule
