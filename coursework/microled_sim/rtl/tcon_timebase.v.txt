`timescale 1ns / 1ps
// TCON column time base — 100 Hz: 1024 column slots per 10 ms frame, each slot
// is 976 or 977 cycles of the 100 MHz CLK_REF, distributed so that exactly
// 576 slots are 977 and 448 are 976  (576*977 + 448*976 = 1,000,000).
//
// T1R fixes (three defects found by independent review):
//   1) frac is initialised to FRAC_INIT (=576). With frac=0 the FIRST frame
//      after reset contains only 575 long slots (449 short) = 999,999 cycles,
//      i.e. 10 ns short. With 576 it is exactly 576/448 = 1,000,000 for every
//      frame, including the first one after reset.
//   2) The hold state (enable=0) parked slot_cycle at 0, so on the first
//      enabled edge it incremented to 1 and slot 0 lasted only 975 cycles.
//      The rising edge of enable now RESTARTS cycle 0, so the first slot is a
//      full BASE_LEN slot.
//   3) Short-slot-first order is preserved: the first slot after reset is 976.
//
// SCOPE: T1 skeleton. No 768-byte receive, no 2x768 B ping-pong (that is T2).
module tcon_timebase #(
    parameter integer BASE_LEN  = 976,     // 100 Hz short slot
    parameter integer LONG_LEN  = 977,     // 100 Hz long slot
    parameter integer FRAC_STEP = 576,     // long slots per 1024 slots
    parameter integer FRAC_MOD  = 1024,
    parameter integer FRAC_INIT = 576,     // T1R: makes the first frame exact
    parameter integer N_COL     = 1024
) (
    input  wire        clk,
    input  wire        reset_n,
    input  wire        enable,        // 0 = hold at slot start, column frozen
    output reg  [10:0] slot_cycle,    // 0 .. slot_len-1
    output reg  [10:0] slot_len,      // 976 or 977
    output reg  [9:0]  col_index,     // 0 .. 1023
    output reg         slot_start,    // high on the first cycle of a slot
    output reg         slot_last,     // high on the last cycle of a slot
    output reg         frame_wrap,    // 1-cycle pulse when col_index wraps
    output wire        slot_long      // current slot is a 977-cycle slot
);

    reg [9:0] frac;                   // fractional accumulator, 0 .. FRAC_MOD-1
    reg       enable_d;               // detects the rising edge of enable

    assign slot_long = (slot_len == LONG_LEN);

    // next-length decision, evaluated when the current slot ends
    wire        add_frac = (frac + FRAC_STEP) >= FRAC_MOD;
    wire [9:0]  frac_nxt = add_frac ? (frac + FRAC_STEP - FRAC_MOD) : (frac + FRAC_STEP);
    wire [10:0] len_nxt  = add_frac ? LONG_LEN : BASE_LEN;

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            slot_cycle <= 11'd0;
            slot_len   <= BASE_LEN;
            col_index  <= 10'd0;
            slot_start <= 1'b1;
            slot_last  <= 1'b0;
            frame_wrap <= 1'b0;
            frac       <= FRAC_INIT;
            enable_d   <= 1'b0;
        end else begin
            enable_d <= enable;
            if (!enable) begin
                // Hold at the start of the slot; the fractional phase (frac) is
                // preserved so a pause does not shift the 576/448 distribution.
                slot_cycle <= 11'd0;
                slot_start <= 1'b1;
                slot_last  <= 1'b0;
                frame_wrap <= 1'b0;
            end else if (!enable_d) begin
                // Rising edge of enable: restart cycle 0 of the first slot.
                // Without this the held cycle 0 is consumed and slot 0 would
                // last 975 instead of 976 cycles (a 10 ns short first frame).
                slot_cycle <= 11'd0;
                slot_len   <= BASE_LEN;
                slot_start <= 1'b1;
                slot_last  <= 1'b0;
                frame_wrap <= 1'b0;
            end else begin
                frame_wrap <= 1'b0;
                if (slot_cycle == (slot_len - 11'd1)) begin
                    slot_cycle <= 11'd0;
                    slot_len   <= len_nxt;
                    frac       <= frac_nxt;
                    slot_start <= 1'b1;
                    slot_last  <= 1'b0;
                    col_index  <= (col_index == (N_COL - 1)) ? 10'd0 : (col_index + 10'd1);
                    frame_wrap <= (col_index == (N_COL - 1));
                end else begin
                    slot_cycle <= slot_cycle + 11'd1;
                    slot_start <= 1'b0;
                    slot_last  <= (slot_cycle == (slot_len - 11'd2));
                end
            end
        end
    end

endmodule
