`timescale 1ns / 1ps
// TCON video receiver (T2A / T2AR) — 768 bytes per column, 1 byte per CLK_REF.
//
// A column is committed to its bank (bank = col_index[0]) only when exactly
// RX_LEN bytes were seen AND COL_SYNC arrives inside the acceptance window.
// Everything else is classified and the bank is invalidated, which makes the
// data transmitter emit zeros for that column (black, no optical output)
// without disturbing the frame time base.
//
// ---- ACCEPTANCE WINDOW (T2AR, review P0-A) --------------------------------
//   The last legal COL_SYNC cycle is 830, not 831.
//   Reason: tcon_data_tx pre-fetches byte 0 on the CLK_REF FALLING edge that
//   precedes the first DCLK sampling edge, and it latches col_valid there. That
//   falling edge is at cycle 831 (or 832 when DCLK happens to be high during
//   831). A commit takes one posedge, so the latest usable sync is the one
//   present during cycle 830 (commit at posedge 831, 5 ns before the latch).
//   A sync during 831 or later would commit at posedge 832, after byte 0 has
//   already been launched -> the column cannot be used. It is therefore
//   REJECTED and flagged as late_sync (err_flags[5]) instead of being silently
//   accepted with a zero payload or silently dropped.
//   The V1.3 "768 B before the load at 832" wording is thus tightened by one
//   cycle by the pre-fetch edge; this is a contract correction, not a test
//   threshold change (50 MHz, 64 ticks, LOAD_START=832 are untouched).
//
// ---- ERROR BITS (sticky until reset) --------------------------------------
//   [0] underrun      COL_SYNC before RX_LEN bytes (inside the window)
//   [1] overflow      a (RX_LEN+1)-th byte arrived -> column revoked
//   [2] duplicate     a second COL_SYNC after a legal commit (column kept)
//   [3] missing       no COL_SYNC at the deadline (end of cycle 830)
//   [4] vsync phase   VSYNC while COL_ID != 0
//   [5] late sync     COL_SYNC after the acceptance window -> column rejected
//
// ---- REVIEW FINDINGS FIXED HERE ------------------------------------------
//   P0-A non-blocking collision: a sync exactly on the deadline edge used to
//        commit (col_ok<=1, okX<=1) and then be invalidated by the deadline
//        branch reading the OLD col_ok. The deadline now uses the same-cycle
//        commit condition, and the window moved to 830 (see above).
//   P0-B overflow after a successful commit used to set the flag but leave the
//        bank valid; it now revokes ok/col_ok while keeping had_sync, so the
//        deadline branch does not mis-report "missing sync" afterwards.
//
// SCOPE: receive + classify only. No frame buffer, no transpose (V1.3 §3/§9).
module tcon_video_rx #(
    parameter integer RX_LEN      = 768,
    parameter integer ACCEPT_LAST = 830,   // last cycle a COL_SYNC can be used
    parameter integer LOAD_START  = 832
) (
    input  wire        clk,
    input  wire        reset_n,
    input  wire        safe,

    input  wire        slot_last,
    input  wire [10:0] slot_cycle,
    input  wire [9:0]  col_index,

    input  wire        vsync,
    input  wire        col_sync,
    input  wire [7:0]  video_d,
    input  wire        de,

    output wire        wr_en,
    output wire [9:0]  wr_idx,
    output wire [7:0]  wr_data,
    output wire        wr_bank,

    output wire [1:0]  bank_valid,
    output wire        col_valid,
    output reg  [5:0]  err_flags,
    output reg  [5:0]  err_now,     // bits set on THIS cycle (per-column attribution)
    output reg  [9:0]  bytes_rcvd,
    output reg         col_ok
);

    reg [9:0] idx;
    reg       sync_d;
    reg       ovf;
    reg       had_sync;      // a legal commit happened for this column

    reg [9:0] owner0, owner1;
    reg       ok0, ok1;

    reg        wr_en_r;
    reg [9:0]  wr_idx_r;
    reg [7:0]  wr_data_r;

    wire sync_p       = col_sync && !sync_d;
    wire full         = (idx == RX_LEN);
    wire past_deadline = (slot_cycle > ACCEPT_LAST);
    // a sync captured on this edge is usable if the column is complete, not
    // already committed, not overflowed and not past the acceptance window
    wire sync_commit  = sync_p && !col_ok && !had_sync && full && !ovf && !past_deadline;

    assign wr_bank    = col_index[0];
    assign wr_en      = wr_en_r;
    assign wr_idx     = wr_idx_r;
    assign wr_data    = wr_data_r;
    assign bank_valid = {ok1, ok0};
    assign col_valid  = col_index[0] ? (ok1 && (owner1 == col_index))
                                     : (ok0 && (owner0 == col_index));

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            idx        <= 10'd0;
            sync_d     <= 1'b0;
            ovf        <= 1'b0;
            had_sync   <= 1'b0;
            wr_en_r    <= 1'b0;
            wr_idx_r   <= 10'd0;
            wr_data_r  <= 8'h00;
            owner0     <= 10'd0;
            owner1     <= 10'd0;
            ok0        <= 1'b0;
            ok1        <= 1'b0;
            err_flags  <= 6'b000000;
            err_now    <= 6'b000000;
            bytes_rcvd <= 10'd0;
            col_ok     <= 1'b0;
        end else begin
            sync_d  <= col_sync;
            wr_en_r <= 1'b0;
            err_now <= 6'b000000;

            if (!safe) begin
                idx        <= 10'd0;
                col_ok     <= 1'b0;
                ovf        <= 1'b0;
                had_sync   <= 1'b0;
                bytes_rcvd <= 10'd0;
                if (col_index[0]) ok1 <= 1'b0;
                else              ok0 <= 1'b0;
            end else begin
                // ---------------- column boundary (priority) ----------------
                if (slot_last) begin
                    col_ok   <= 1'b0;
                    ovf      <= 1'b0;
                    had_sync <= 1'b0;
                    if (de) begin
                        wr_en_r    <= 1'b1;
                        wr_idx_r   <= 10'd0;
                        wr_data_r  <= video_d;
                        idx        <= 10'd1;
                        bytes_rcvd <= 10'd1;
                    end else begin
                        idx        <= 10'd0;
                        bytes_rcvd <= 10'd0;
                    end
                end else if (de) begin
                    if (!full) begin
                        wr_en_r    <= 1'b1;
                        wr_idx_r   <= idx;
                        wr_data_r  <= video_d;
                        idx        <= idx + 10'd1;
                        bytes_rcvd <= idx + 10'd1;
                    end else begin
                        // 769th byte: the column no longer matches the format,
                        // so a bank that was already committed must be revoked.
                        err_flags[1] <= 1'b1;
                        err_now[1]   <= 1'b1;
                        ovf          <= 1'b1;
                        col_ok       <= 1'b0;
                        if (col_index[0]) ok1 <= 1'b0;
                        else              ok0 <= 1'b0;
                    end
                end

                // ---------------- COL_SYNC classification ---------------------
                if (sync_p) begin
                    if (col_ok) begin
                        err_flags[2] <= 1'b1;              // duplicate, column kept
                        err_now[2]   <= 1'b1;
                    end else if (sync_commit) begin
                        if (col_index[0]) begin
                            owner1 <= col_index;
                            ok1    <= 1'b1;
                        end else begin
                            owner0 <= col_index;
                            ok0    <= 1'b1;
                        end
                        col_ok   <= 1'b1;
                        had_sync <= 1'b1;
                    end else if (past_deadline) begin
                        err_flags[5] <= 1'b1;              // too late to be used
                        err_now[5]   <= 1'b1;
                        if (col_index[0]) ok1 <= 1'b0;
                        else              ok0 <= 1'b0;
                    end else begin
                        err_flags[0] <= 1'b1;              // underrun / overflow
                        err_now[0]   <= 1'b1;
                        if (col_index[0]) ok1 <= 1'b0;
                        else              ok0 <= 1'b0;
                    end
                end

                // ---------------- acceptance deadline --------------------------
                // Evaluated on the last usable cycle. `sync_commit` keeps a legal
                // sync captured on this very edge from being invalidated by the
                // pre-edge value of had_sync (review P0-A).
                if ((slot_cycle == ACCEPT_LAST) && !had_sync && !sync_commit) begin
                    err_flags[3] <= 1'b1;
                    err_now[3]   <= 1'b1;
                    if (col_index[0]) ok1 <= 1'b0;
                    else              ok0 <= 1'b0;
                end

                if (vsync && (col_index != 10'd0)) begin
                    err_flags[4] <= 1'b1;
                    err_now[4]   <= 1'b1;
                end
            end
        end
    end

endmodule
