`timescale 1ns / 1ps
// TCON video receiver — V1.3 column-start protocol (T2B).
//
// V1.3 (SHA 6b61e4ab…afaa):
//   §12  COL_SYNC = "新列首字节边界，同PCLK采样"
//   §9   "COL_SYNC与第一个有效字节同一PCLK上升沿采样，DE=1表示有效；
//         每列必须有且只有768个有效字节。允许短暂停顿但整列必须在规定装载窗内完成"
//        "VSYNC与新帧第一个COL_SYNC同沿"
//
// So a column STARTS on the edge that carries COL_SYNC + DE + byte0. The previous
// implementation (LEGACY_TAIL_SYNC) waited for 768 bytes and then accepted a
// COL_SYNC as a completion acknowledgement; that was the interface direction
// error recorded in T2B-A and is archived under legacy_tail_sync/.
//
// CYCLE CONVENTION: `slot_cycle` still holds the PRE-EDGE value inside a posedge
// block, so "cycle 0" means "the byte that was present during slot cycle 0".
//
// DEADLINE: the 768th byte must be sampled no later than cycle 830 (DEADLINE).
//   The transmit pre-fetch happens on the falling edge of cycle 831 (or 832,
//   depending on the DCLK phase); a byte sampled at 830 is written into the bank
//   at posedge 831, i.e. 5 ns before the earliest pre-fetch.
//
// ERROR BITS (6 bits are frozen; see t2b/T2B_V13_FROZEN_INTERFACE.md §7):
//   [0] underrun  : legal start but fewer than 768 bytes by DEADLINE
//   [1] overflow  : 769th byte
//   [2] duplicate : second COL_SYNC in the same slot OR input column label
//                   disagrees with the local column (column-order error)
//   [3] missing   : no COL_SYNC on cycle 0 of a slot (in ACTIVE)
//   [4] vsync     : VSYNC not on cycle 0, or VSYNC on a non-zero column
//   [5] misplaced : COL_SYNC on a cycle other than 0
//
// Every error also raises `poison` for the current column; tcon_top turns any
// error pulse into the latched safe state (Q3=B).
module tcon_video_rx #(
    parameter integer RX_LEN   = 768,
    parameter integer DEADLINE = 830
) (
    input  wire        clk,
    input  wire        reset_n,
    input  wire        safe,        // mode-level readiness (not per column)
    input  wire        active,      // FSM is in ACTIVE (video framing established)

    input  wire [10:0] slot_cycle,
    input  wire [9:0]  col_index,   // local (time base) column

    input  wire        vsync,
    input  wire        col_sync,
    input  wire [7:0]  video_d,
    input  wire        de,

    output wire        wr_en,
    output wire [9:0]  wr_idx,
    output wire [7:0]  wr_data,
    output wire        wr_bank,

    output wire        col_valid,   // column complete, clean and label-consistent
    output wire        col_poison,
    output reg  [9:0]  in_col,      // input column label since VSYNC
    output reg  [5:0]  frame_id,    // increments on every VSYNC (first frame = 1)

    output reg  [5:0]  err_flags,   // sticky
    output reg  [5:0]  err_now,     // one-cycle pulses
    output reg  [9:0]  bytes_rcvd
);

    reg [9:0] idx;
    reg       sync_d, vsync_d;
    reg       start_seen;      // legal start sync seen in this slot
    reg       poison_c;        // current column is not displayable

    reg        wr_en_r;
    reg [9:0]  wr_idx_r;
    reg [7:0]  wr_data_r;

    wire at0    = (slot_cycle == 11'd0);
    wire sync_p = col_sync && !sync_d;
    wire vs_p   = vsync && !vsync_d;
    wire full   = (idx == RX_LEN);

    // the column label this sync introduces
    wire [9:0] next_col = vs_p ? 10'd0 : (in_col + 10'd1);

    assign wr_en   = wr_en_r;
    assign wr_idx  = wr_idx_r;
    assign wr_data = wr_data_r;
    // the bank follows the LOCAL column so the receive/forward relationship stays
    // deterministic even while an error is being reported
    assign wr_bank = col_index[0];

    assign col_poison = poison_c;
    assign col_valid  = start_seen && (idx == RX_LEN) && !poison_c &&
                        (in_col == col_index);

    always @(posedge clk or negedge reset_n) begin
        if (!reset_n) begin
            idx        <= 10'd0;
            sync_d     <= 1'b0;
            vsync_d    <= 1'b0;
            start_seen <= 1'b0;
            poison_c   <= 1'b0;
            in_col     <= 10'd0;
            frame_id   <= 6'd0;
            wr_en_r    <= 1'b0;
            wr_idx_r   <= 10'd0;
            wr_data_r  <= 8'h00;
            err_flags  <= 6'b000000;
            err_now    <= 6'b000000;
            bytes_rcvd <= 10'd0;
        end else begin
            sync_d  <= col_sync;
            vsync_d <= vsync;
            wr_en_r <= 1'b0;
            err_now <= 6'b000000;

            if (!safe) begin
                idx        <= 10'd0;
                start_seen <= 1'b0;
                poison_c   <= 1'b0;
                bytes_rcvd <= 10'd0;
            end else begin
                // ---------------- cycle 0: column start -----------------------
                if (at0) begin
                    // a new column starts clean; any error re-sets it below
                    poison_c <= 1'b0;
                    if (vs_p) frame_id <= frame_id + 6'd1;

                    if (sync_p) begin
                        start_seen <= 1'b1;
                        in_col     <= next_col;
                        if (de) begin
                            wr_en_r    <= 1'b1;
                            wr_idx_r   <= 10'd0;
                            wr_data_r  <= video_d;
                            idx        <= 10'd1;
                            bytes_rcvd <= 10'd1;
                        end else begin
                            // sync without byte 0: the edge is not a byte boundary
                            idx          <= 10'd0;
                            bytes_rcvd   <= 10'd0;
                            poison_c     <= 1'b1;
                            err_now[0]   <= 1'b1;
                            err_flags[0] <= 1'b1;
                        end
                        // column-order / frame-phase checks
                        if (next_col != col_index) begin
                            poison_c     <= 1'b1;
                            err_now[2]   <= 1'b1;
                            err_flags[2] <= 1'b1;
                        end
                        if (vs_p && (col_index != 10'd0)) begin
                            poison_c     <= 1'b1;
                            err_now[4]   <= 1'b1;
                            err_flags[4] <= 1'b1;
                        end
                    end else begin
                        start_seen <= 1'b0;
                        idx        <= 10'd0;
                        bytes_rcvd <= 10'd0;
                        if (active) begin
                            poison_c     <= 1'b1;
                            err_now[3]   <= 1'b1;
                            err_flags[3] <= 1'b1;
                        end
                    end
                end

                // ---------------- misplaced / duplicate sync -------------------
                if (sync_p && !at0) begin
                    poison_c     <= 1'b1;
                    err_now[5]   <= 1'b1;
                    err_flags[5] <= 1'b1;
                end

                // ---------------- data beats -----------------------------------
                if (!at0 && de) begin
                    if (!full) begin
                        wr_en_r    <= 1'b1;
                        wr_idx_r   <= idx;
                        wr_data_r  <= video_d;
                        idx        <= idx + 10'd1;
                        bytes_rcvd <= idx + 10'd1;
                    end else begin
                        // 769th byte of this column
                        poison_c     <= 1'b1;
                        err_now[1]   <= 1'b1;
                        err_flags[1] <= 1'b1;
                    end
                end

                // ---------------- deadline -------------------------------------
                if (slot_cycle == DEADLINE) begin
                    if (start_seen && (idx != RX_LEN)) begin
                        poison_c     <= 1'b1;
                        err_now[0]   <= 1'b1;
                        err_flags[0] <= 1'b1;
                    end
                end

                // ---------------- VSYNC away from the column boundary ----------
                if (vs_p && !at0 && active) begin
                    poison_c     <= 1'b1;
                    err_now[4]   <= 1'b1;
                    err_flags[4] <= 1'b1;
                end
            end
        end
    end

endmodule
